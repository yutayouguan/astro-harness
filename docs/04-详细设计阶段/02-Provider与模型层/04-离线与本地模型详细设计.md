## 1. Ollama Sidecar 生命周期管理

### 1.1 核心结构

`OllamaSidecar` 封装 Ollama 进程的完整生命周期——安装检测、启动、健康监控、优雅关闭。它作为 Tauri 应用的 managed state 在应用启动时初始化。

```rust
// apps/desktop/src-tauri/src/sidecar.rs

use std::path::PathBuf;
use std::sync::Arc;
use tokio::sync::{RwLock, watch};
use tauri::AppHandle;
use tauri_plugin_shell::ShellExt;
use tauri_plugin_shell::process::CommandChild;

/// Ollama sidecar 进程管理器
pub struct OllamaSidecar {
    /// 当前 Ollama 进程句柄（None = 未启动）
    child: Arc<RwLock<Option<CommandChild>>>,
    /// Ollama 运行状态广播
    status_tx: watch::Sender<SidecarStatus>,
    /// Ollama 运行状态接收端（供外部订阅）
    pub status_rx: watch::Receiver<SidecarStatus>,
    /// Ollama 检测到的安装路径
    install_path: Arc<RwLock<Option<PathBuf>>>,
    /// Ollama HTTP 端口（默认 11434）
    port: u16,
    /// 健康检查间隔
    health_interval: Duration,
}

#[derive(Debug, Clone, PartialEq)]
pub enum SidecarStatus {
    /// 未检测到 Ollama 安装
    NotInstalled,
    /// 已检测到安装，但未启动
    Installed { version: String, path: PathBuf },
    /// 正在启动中
    Starting,
    /// 运行中，健康检查通过
    Running { version: String, models_loaded: usize },
    /// 运行中但健康检查失败（进程存在但 API 无响应）
    Unhealthy { reason: String },
    /// 已停止（正常关闭或崩溃）
    Stopped { exit_code: Option<i32> },
}

use std::time::Duration;

impl OllamaSidecar {
    pub fn new(port: u16) -> Self {
        let (status_tx, status_rx) = watch::channel(SidecarStatus::NotInstalled);
        Self {
            child: Arc::new(RwLock::new(None)),
            status_tx,
            status_rx,
            install_path: Arc::new(RwLock::new(None)),
            port,
            health_interval: Duration::from_secs(15),
        }
    }

    pub fn base_url(&self) -> String {
        format!("http://localhost:{}", self.port)
    }

    pub fn api_url(&self) -> String {
        format!("{}/v1", self.base_url())
    }
}
```

### 1.2 安装检测

应用启动时首先检测系统是否安装了 Ollama。按优先级依次尝试：打包的 sidecar 二进制、系统 PATH、已知安装路径。

```rust
impl OllamaSidecar {
    /// 检测 Ollama 安装位置，返回可执行文件路径
    pub async fn detect_installation(&self) -> Option<OllamaInstallInfo> {
        // 1. 检查应用内打包的 sidecar（优先级最高）
        //    Tauri 打包后的 sidecar 路径由框架管理，无需手动查找
        //    这里仅记录外部安装情况，sidecar 启动走 app.shell().sidecar()

        // 2. 检查系统 PATH
        if let Ok(output) = tokio::process::Command::new("which")
            .arg("ollama")
            .output()
            .await
        {
            if output.status.success() {
                let path = PathBuf::from(
                    String::from_utf8_lossy(&output.stdout).trim()
                );
                if let Some(version) = self.query_version_at(&path).await {
                    return Some(OllamaInstallInfo {
                        path,
                        version,
                        source: InstallSource::SystemPath,
                    });
                }
            }
        }

        // 3. 检查已知安装路径（macOS / Linux / Windows）
        let known_paths = if cfg!(target_os = "macos") {
            vec![
                PathBuf::from("/usr/local/bin/ollama"),
                PathBuf::from("/opt/homebrew/bin/ollama"),
                dirs::home_dir().map(|h| h.join(".ollama/bin/ollama"))
                    .unwrap_or_default(),
            ]
        } else if cfg!(target_os = "linux") {
            vec![
                PathBuf::from("/usr/bin/ollama"),
                PathBuf::from("/usr/local/bin/ollama"),
                PathBuf::from("/snap/bin/ollama"),
            ]
        } else {
            vec![
                PathBuf::from(r"C:\Program Files\Ollama\ollama.exe"),
                dirs::home_dir().map(|h| h.join("AppData\\Local\\Ollama\\ollama.exe"))
                    .unwrap_or_default(),
            ]
        };

        for path in known_paths {
            if path.exists() {
                if let Some(version) = self.query_version_at(&path).await {
                    return Some(OllamaInstallInfo {
                        path,
                        version,
                        source: InstallSource::KnownPath,
                    });
                }
            }
        }

        None
    }

    /// 查询指定路径的 Ollama 版本号
    async fn query_version_at(&self, path: &PathBuf) -> Option<String> {
        let output = tokio::process::Command::new(path)
            .arg("--version")
            .output()
            .await
            .ok()?;
        if output.status.success() {
            // 输出格式: "ollama version is 0.5.x"
            let stdout = String::from_utf8_lossy(&output.stdout);
            let version = stdout.trim()
                .strip_prefix("ollama version is ")
                .unwrap_or(stdout.trim())
                .to_string();
            Some(version)
        } else {
            None
        }
    }

    /// 检查版本兼容性（最低要求 0.3.0，因为需要 OpenAI 兼容 API）
    pub fn check_version_compatibility(version: &str) -> VersionCompat {
        let parts: Vec<u32> = version.split('.')
            .filter_map(|s| s.parse().ok())
            .collect();

        match (parts.get(0), parts.get(1)) {
            (Some(0), Some(minor)) if *minor < 3 => VersionCompat::TooOld {
                installed: version.to_string(),
                minimum: "0.3.0".to_string(),
                reason: "需要 OpenAI 兼容 API（/v1/chat/completions）".to_string(),
            },
            (Some(0), Some(minor)) if *minor < 5 => VersionCompat::Supported {
                warning: Some("建议升级到 0.5.0+ 以获得更好的多模态支持".into()),
            },
            _ => VersionCompat::Supported { warning: None },
        }
    }
}

#[derive(Debug, Clone)]
pub struct OllamaInstallInfo {
    pub path: PathBuf,
    pub version: String,
    pub source: InstallSource,
}

#[derive(Debug, Clone)]
pub enum InstallSource {
    Sidecar,     // 应用内打包
    SystemPath,  // 系统 PATH
    KnownPath,   // 已知安装路径
}

#[derive(Debug, Clone)]
pub enum VersionCompat {
    Supported { warning: Option<String> },
    TooOld { installed: String, minimum: String, reason: String },
}
```

### 1.3 启动流程

应用启动时自动检测 Ollama 并尝试启动。如果已有 Ollama 实例在运行（用户自行启动），直接复用。

```rust
impl OllamaSidecar {
    /// 应用启动时调用：检测安装 → 启动或复用 → 健康监控
    pub async fn initialize(&self, app: &AppHandle) -> anyhow::Result<()> {
        // 1. 先检查是否已有 Ollama 在运行（用户可能已手动启动）
        if self.is_already_running().await {
            let version = self.query_running_version().await
                .unwrap_or_else(|| "unknown".to_string());
            let models = self.count_loaded_models().await.unwrap_or(0);
            self.status_tx.send(SidecarStatus::Running {
                version,
                models_loaded: models,
            }).ok();
            tracing::info!("检测到已运行的 Ollama 实例，复用之");
            return Ok(());
        }

        // 2. 检测安装
        let install_info = self.detect_installation().await;
        match &install_info {
            None => {
                self.status_tx.send(SidecarStatus::NotInstalled).ok();
                tracing::info!("未检测到 Ollama 安装，离线功能不可用");
                return Ok(()); // 非致命错误，应用可在无离线支持的情况下运行
            }
            Some(info) => {
                // 版本兼容性检查
                match Self::check_version_compatibility(&info.version) {
                    VersionCompat::TooOld { installed, minimum, reason } => {
                        tracing::warn!(
                            "Ollama 版本过低：{} < {}，原因：{}",
                            installed, minimum, reason
                        );
                        self.status_tx.send(SidecarStatus::Installed {
                            version: info.version.clone(),
                            path: info.path.clone(),
                        }).ok();
                        return Ok(());
                    }
                    VersionCompat::Supported { warning } => {
                        if let Some(w) = warning {
                            tracing::info!("Ollama 版本提醒：{}", w);
                        }
                    }
                }
                *self.install_path.write().await = Some(info.path.clone());
                self.status_tx.send(SidecarStatus::Installed {
                    version: info.version.clone(),
                    path: info.path.clone(),
                }).ok();
            }
        }

        // 3. 启动 Ollama sidecar
        self.start(app).await?;

        Ok(())
    }

    /// 检查 Ollama API 是否已可达（用户可能自行启动了 ollama serve）
    async fn is_already_running(&self) -> bool {
        reqwest::get(format!("{}/api/version", self.base_url()))
            .await
            .map(|r| r.status().is_success())
            .unwrap_or(false)
    }

    /// 查询正在运行的 Ollama 版本
    async fn query_running_version(&self) -> Option<String> {
        let resp = reqwest::get(format!("{}/api/version", self.base_url()))
            .await.ok()?
            .json::<serde_json::Value>()
            .await.ok()?;
        resp.get("version")
            .and_then(|v| v.as_str())
            .map(|s| s.to_string())
    }

    /// 统计当前已加载的模型数量
    async fn count_loaded_models(&self) -> Option<usize> {
        let resp = reqwest::get(format!("{}/api/tags", self.base_url()))
            .await.ok()?
            .json::<serde_json::Value>()
            .await.ok()?;
        resp.get("models")
            .and_then(|m| m.as_array())
            .map(|arr| arr.len())
    }

    /// 启动 Ollama sidecar 进程
    pub async fn start(&self, app: &AppHandle) -> anyhow::Result<()> {
        self.status_tx.send(SidecarStatus::Starting).ok();

        // 通过 Tauri sidecar 机制启动
        let sidecar_cmd = app.shell()
            .sidecar("ollama")?
            .args(["serve"])
            .env("OLLAMA_HOST", format!("0.0.0.0:{}", self.port));

        let (mut rx, child) = sidecar_cmd.spawn()?;

        // 保存子进程句柄
        *self.child.write().await = Some(child);

        // 在后台监听子进程输出（用于日志）
        let status_tx = self.status_tx.clone();
        tokio::spawn(async move {
            use tauri_plugin_shell::process::CommandEvent;
            while let Some(event) = rx.recv().await {
                match event {
                    CommandEvent::Stdout(line) => {
                        tracing::debug!(target: "ollama", "{}", String::from_utf8_lossy(&line));
                    }
                    CommandEvent::Stderr(line) => {
                        tracing::warn!(target: "ollama", "{}", String::from_utf8_lossy(&line));
                    }
                    CommandEvent::Terminated(payload) => {
                        tracing::info!("Ollama 进程退出: code={:?}", payload.code);
                        status_tx.send(SidecarStatus::Stopped {
                            exit_code: payload.code,
                        }).ok();
                        break;
                    }
                    _ => {}
                }
            }
        });

        // 等待 Ollama 就绪（轮询健康检查，最多 30 秒）
        for attempt in 1..=30 {
            if self.is_already_running().await {
                let version = self.query_running_version().await
                    .unwrap_or_else(|| "unknown".to_string());
                let models = self.count_loaded_models().await.unwrap_or(0);
                self.status_tx.send(SidecarStatus::Running {
                    version: version.clone(),
                    models_loaded: models,
                }).ok();
                tracing::info!(
                    "Ollama sidecar 启动完成（v{}，耗时 {}s，已有 {} 个模型）",
                    version, attempt, models
                );
                return Ok(());
            }
            tokio::time::sleep(Duration::from_secs(1)).await;
        }

        Err(anyhow::anyhow!("Ollama 启动超时（30s），请检查日志"))
    }
}
```

### 1.4 健康检查循环

Ollama 启动后持续进行健康检查。如果进程意外退出，自动尝试重启（最多 3 次）。

```rust
impl OllamaSidecar {
    /// 启动后台健康检查循环
    pub fn start_health_loop(&self, app: AppHandle) {
        let status_tx = self.status_tx.clone();
        let child = self.child.clone();
        let base_url = self.base_url();
        let interval = self.health_interval;
        let port = self.port;

        tokio::spawn(async move {
            let mut consecutive_failures = 0u32;
            let max_restart_attempts = 3u32;
            let mut restart_count = 0u32;

            loop {
                tokio::time::sleep(interval).await;

                let health = Self::probe_health(&base_url).await;

                match health {
                    HealthProbe::Ok { version, models } => {
                        consecutive_failures = 0;
                        status_tx.send(SidecarStatus::Running {
                            version,
                            models_loaded: models,
                        }).ok();
                    }
                    HealthProbe::Unreachable(reason) => {
                        consecutive_failures += 1;
                        tracing::warn!(
                            "Ollama 健康检查失败（连续第 {} 次）：{}",
                            consecutive_failures, reason
                        );

                        if consecutive_failures >= 3 {
                            // 进程可能已崩溃，检查并尝试重启
                            let has_child = child.read().await.is_some();
                            if has_child && restart_count < max_restart_attempts {
                                restart_count += 1;
                                tracing::warn!(
                                    "Ollama 似乎已崩溃，尝试重启（第 {}/{} 次）",
                                    restart_count, max_restart_attempts
                                );
                                status_tx.send(SidecarStatus::Starting).ok();

                                // 清理旧进程
                                if let Some(mut old_child) = child.write().await.take() {
                                    let _ = old_child.kill();
                                }

                                // 尝试重新启动（复用 sidecar 命令）
                                match app.shell()
                                    .sidecar("ollama")
                                    .and_then(|cmd| Ok(cmd.args(["serve"])
                                        .env("OLLAMA_HOST", format!("0.0.0.0:{}", port))))
                                {
                                    Ok(cmd) => match cmd.spawn() {
                                        Ok((_, new_child)) => {
                                            *child.write().await = Some(new_child);
                                            consecutive_failures = 0;
                                            // 等待几秒后下次循环会检查健康状态
                                            tokio::time::sleep(Duration::from_secs(5)).await;
                                        }
                                        Err(e) => {
                                            tracing::error!("Ollama 重启失败：{}", e);
                                            status_tx.send(SidecarStatus::Stopped {
                                                exit_code: None,
                                            }).ok();
                                        }
                                    },
                                    Err(e) => {
                                        tracing::error!("Ollama sidecar 命令构建失败：{}", e);
                                    }
                                }
                            } else {
                                status_tx.send(SidecarStatus::Unhealthy {
                                    reason: reason.clone(),
                                }).ok();
                            }
                        }
                    }
                }
            }
        });
    }

    /// 探测 Ollama 健康状态
    async fn probe_health(base_url: &str) -> HealthProbe {
        match tokio::time::timeout(
            Duration::from_secs(5),
            reqwest::get(format!("{}/api/tags", base_url)),
        ).await {
            Ok(Ok(resp)) if resp.status().is_success() => {
                let body = resp.json::<serde_json::Value>().await
                    .unwrap_or_default();
                let models = body.get("models")
                    .and_then(|m| m.as_array())
                    .map(|a| a.len())
                    .unwrap_or(0);
                // 顺便拿版本
                let version = reqwest::get(format!("{}/api/version", base_url))
                    .await.ok()
                    .and_then(|r| r.json::<serde_json::Value>().ok().flatten())
                    .and_then(|v| v.get("version")?.as_str().map(String::from))
                    .unwrap_or_else(|| "unknown".to_string());
                HealthProbe::Ok { version, models }
            }
            Ok(Ok(resp)) => {
                HealthProbe::Unreachable(format!("HTTP {}", resp.status()))
            }
            Ok(Err(e)) => {
                HealthProbe::Unreachable(format!("连接失败: {}", e))
            }
            Err(_) => {
                HealthProbe::Unreachable("健康检查超时（5s）".to_string())
            }
        }
    }
}

enum HealthProbe {
    Ok { version: String, models: usize },
    Unreachable(String),
}
```

### 1.5 优雅关闭

应用退出时向 Ollama 发送终止信号，等待其完成当前推理请求后退出。

```rust
impl OllamaSidecar {
    /// 优雅关闭 Ollama 进程
    pub async fn shutdown(&self) -> anyhow::Result<()> {
        let mut child_guard = self.child.write().await;
        if let Some(mut child) = child_guard.take() {
            tracing::info!("正在关闭 Ollama sidecar...");
            self.status_tx.send(SidecarStatus::Stopped { exit_code: None }).ok();

            // 先发送 SIGTERM（macOS/Linux）让 Ollama 优雅退出
            // Tauri CommandChild::kill() 在 Unix 上发送 SIGKILL
            // 这里先通过 API 请求 Ollama 停止当前推理
            let _ = reqwest::Client::new()
                .delete(format!("{}/api/generate", self.base_url()))
                .send()
                .await;

            // 等待最多 10 秒让 Ollama 完成清理
            let shutdown_timeout = tokio::time::timeout(
                Duration::from_secs(10),
                async {
                    loop {
                        if !self.is_already_running().await {
                            break;
                        }
                        tokio::time::sleep(Duration::from_millis(500)).await;
                    }
                },
            ).await;

            if shutdown_timeout.is_err() {
                // 超时，强制终止
                tracing::warn!("Ollama 优雅关闭超时，强制终止");
                let _ = child.kill();
            }

            tracing::info!("Ollama sidecar 已关闭");
        }
        Ok(())
    }
}
```

## 2. 本地模型自动发现与注册

### 2.1 模型扫描

定期扫描 Ollama 已安装的模型列表，与 Provider `Registry` 同步。当用户通过 `ollama pull` 拉取新模型或删除旧模型时，注册表自动更新。

```rust
// crates/agent-providers/src/providers/ollama/discovery.rs

use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::RwLock;
use serde::{Deserialize, Serialize};

/// 本地模型发现与注册管理器
pub struct LocalModelDiscovery {
    /// Ollama API 地址
    base_url: String,
    /// 已发现的本地模型（model_name → LocalModelInfo）
    models: Arc<RwLock<HashMap<String, LocalModelInfo>>>,
    /// 模型变更通知
    on_change: tokio::sync::broadcast::Sender<ModelChangeEvent>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct LocalModelInfo {
    /// 模型名称（如 "qwen2.5:7b"）
    pub name: String,
    /// 模型大小（字节）
    pub size: u64,
    /// 模型参数量描述（如 "7B"）
    pub parameter_size: String,
    /// 量化类型（如 "Q4_K_M"）
    pub quantization: String,
    /// 模型家族（如 "qwen2.5"）
    pub family: String,
    /// 模型能力
    pub capabilities: ModelCapabilities,
    /// 模型修改时间
    pub modified_at: String,
    /// 模型摘要（Modelfile 中的描述）
    pub digest: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ModelCapabilities {
    /// 是否支持文本对话
    pub text_chat: bool,
    /// 是否支持多模态输入（图片）
    pub vision: bool,
    /// 是否支持工具调用（function calling）
    pub tool_use: bool,
    /// 是否支持嵌入向量生成
    pub embedding: bool,
    /// 是否支持代码生成
    pub code: bool,
    /// 上下文窗口大小（token 数）
    pub context_length: u32,
}

#[derive(Debug, Clone)]
pub enum ModelChangeEvent {
    /// 发现新模型
    Added(LocalModelInfo),
    /// 模型被删除
    Removed { name: String },
    /// 模型列表完整刷新
    Refreshed { count: usize },
}
```

### 2.2 扫描实现

调用 Ollama 的 `/api/tags` 端点获取已安装模型列表，并通过 `/api/show` 获取模型详细信息以判定能力。

```rust
impl LocalModelDiscovery {
    pub fn new(base_url: String) -> Self {
        let (on_change, _) = tokio::sync::broadcast::channel(32);
        Self {
            base_url,
            models: Arc::new(RwLock::new(HashMap::new())),
            on_change,
        }
    }

    /// 执行一次完整扫描，返回发现的模型列表
    pub async fn scan(&self) -> anyhow::Result<Vec<LocalModelInfo>> {
        let resp = reqwest::get(format!("{}/api/tags", self.base_url))
            .await?
            .json::<serde_json::Value>()
            .await?;

        let raw_models = resp.get("models")
            .and_then(|m| m.as_array())
            .cloned()
            .unwrap_or_default();

        let mut discovered = Vec::new();
        for raw in &raw_models {
            let name = raw.get("name")
                .and_then(|n| n.as_str())
                .unwrap_or_default()
                .to_string();

            if name.is_empty() { continue; }

            let size = raw.get("size")
                .and_then(|s| s.as_u64())
                .unwrap_or(0);

            let modified_at = raw.get("modified_at")
                .and_then(|m| m.as_str())
                .unwrap_or_default()
                .to_string();

            let digest = raw.get("digest")
                .and_then(|d| d.as_str())
                .unwrap_or_default()
                .to_string();

            // 获取模型详细信息以判定能力
            let capabilities = self.detect_capabilities(&name).await;
            let (parameter_size, quantization, family) =
                self.extract_model_metadata(&name).await;

            discovered.push(LocalModelInfo {
                name,
                size,
                parameter_size,
                quantization,
                family,
                capabilities,
                modified_at,
                digest,
            });
        }

        // 与当前已知模型列表做差集，发送变更事件
        let mut models = self.models.write().await;
        let old_names: std::collections::HashSet<_> = models.keys().cloned().collect();
        let new_names: std::collections::HashSet<_> =
            discovered.iter().map(|m| m.name.clone()).collect();

        for added_name in new_names.difference(&old_names) {
            if let Some(info) = discovered.iter().find(|m| &m.name == added_name) {
                self.on_change.send(ModelChangeEvent::Added(info.clone())).ok();
            }
        }
        for removed_name in old_names.difference(&new_names) {
            self.on_change.send(ModelChangeEvent::Removed {
                name: removed_name.clone(),
            }).ok();
        }

        *models = discovered.iter()
            .map(|m| (m.name.clone(), m.clone()))
            .collect();

        self.on_change.send(ModelChangeEvent::Refreshed {
            count: discovered.len(),
        }).ok();

        Ok(discovered)
    }
}
```

### 2.3 模型能力检测

不同本地模型的能力差异很大。通过 `/api/show` 接口获取 Modelfile 和模型元数据，推断其支持的功能。

```rust
impl LocalModelDiscovery {
    /// 通过 Ollama /api/show 接口探测模型能力
    async fn detect_capabilities(&self, model_name: &str) -> ModelCapabilities {
        let client = reqwest::Client::new();
        let resp = client
            .post(format!("{}/api/show", self.base_url))
            .json(&serde_json::json!({ "name": model_name }))
            .send()
            .await;

        let info = match resp {
            Ok(r) => r.json::<serde_json::Value>().await.unwrap_or_default(),
            Err(_) => return ModelCapabilities::text_only_default(),
        };

        // 从模型信息中提取能力
        let modelfile = info.get("modelfile")
            .and_then(|m| m.as_str())
            .unwrap_or_default();

        let template = info.get("template")
            .and_then(|t| t.as_str())
            .unwrap_or_default();

        let details = info.get("details").cloned().unwrap_or_default();
        let families: Vec<String> = details.get("families")
            .and_then(|f| f.as_array())
            .map(|arr| arr.iter()
                .filter_map(|v| v.as_str().map(String::from))
                .collect())
            .unwrap_or_default();

        // 视觉能力检测：模型家族中包含 "clip" 或模型名含 "vision" / "llava"
        let vision = families.iter().any(|f| f.contains("clip"))
            || model_name.contains("vision")
            || model_name.contains("llava")
            || model_name.contains("minicpm-v");

        // 工具调用检测：模板或已知模型支持原生结构化工具接口
        let tool_use = template.contains("{{.Tools}}")
            || template.contains("<|plugin|>")
            || KNOWN_TOOL_MODELS.iter().any(|m| model_name.starts_with(m));

        // 代码能力：模型名或家族包含 code 相关标识
        let code = model_name.contains("code")
            || model_name.contains("coder")
            || model_name.contains("starcoder")
            || model_name.contains("deepseek-coder");

        // 嵌入能力：模型名包含 embed 标识
        let embedding = model_name.contains("embed")
            || model_name.contains("nomic-embed")
            || model_name.contains("bge-");

        // 上下文长度：从 modelfile 中解析 num_ctx，默认 4096
        let context_length = Self::parse_context_length(modelfile);

        ModelCapabilities {
            text_chat: true,  // 所有 Ollama 模型都支持文本对话
            vision,
            tool_use,
            embedding,
            code,
            context_length,
        }
    }

    fn parse_context_length(modelfile: &str) -> u32 {
        for line in modelfile.lines() {
            let trimmed = line.trim();
            if trimmed.starts_with("PARAMETER num_ctx") {
                if let Some(val) = trimmed.split_whitespace().last() {
                    return val.parse().unwrap_or(4096);
                }
            }
        }
        4096 // Ollama 默认值
    }

    /// 提取模型参数量、量化方式、家族
    async fn extract_model_metadata(&self, model_name: &str) -> (String, String, String) {
        let client = reqwest::Client::new();
        let resp = client
            .post(format!("{}/api/show", self.base_url))
            .json(&serde_json::json!({ "name": model_name }))
            .send()
            .await;

        match resp.and_then(|r| Ok(r.json::<serde_json::Value>())) {
            Ok(fut) => match fut.await {
                Ok(info) => {
                    let details = info.get("details").cloned().unwrap_or_default();
                    let param_size = details.get("parameter_size")
                        .and_then(|p| p.as_str())
                        .unwrap_or("unknown")
                        .to_string();
                    let quant = details.get("quantization_level")
                        .and_then(|q| q.as_str())
                        .unwrap_or("unknown")
                        .to_string();
                    let family = details.get("family")
                        .and_then(|f| f.as_str())
                        .unwrap_or("unknown")
                        .to_string();
                    (param_size, quant, family)
                }
                Err(_) => ("unknown".into(), "unknown".into(), "unknown".into()),
            },
            Err(_) => ("unknown".into(), "unknown".into(), "unknown".into()),
        }
    }
}

impl ModelCapabilities {
```

### 3.1 NetworkMonitor 设计

`NetworkMonitor` 同时使用两种机制检测网络状态：周期性主动探测和系统网络事件监听。两者互为补充——主动探测确保不遗漏，系统事件监听提供亚秒级响应速度。

```rust
// crates/agent-core/src/network/monitor.rs

use std::sync::Arc;
use std::time::Duration;
use tokio::sync::{watch, RwLock};

/// 网络连接状态监控器
pub struct NetworkMonitor {
    /// 当前网络状态
    state: Arc<RwLock<NetworkState>>,
    /// 状态变更广播
    state_tx: watch::Sender<NetworkState>,
    /// 状态变更接收（供外部订阅）
    pub state_rx: watch::Receiver<NetworkState>,
    /// 探测配置
    config: NetworkMonitorConfig,
    /// 变更回调列表
    on_change_callbacks: Arc<RwLock<Vec<Box<dyn Fn(&NetworkState, &NetworkState) + Send + Sync>>>>,
}

#[derive(Debug, Clone, PartialEq)]
pub enum NetworkState {
    /// 网络正常
    Online,
    /// 完全离线
    Offline,
    /// 网络质量差（高延迟或丢包）
    Degraded { latency_ms: u64 },
}

impl NetworkState {
    pub fn is_available(&self) -> bool {
        matches!(self, NetworkState::Online | NetworkState::Degraded { .. })
    }

    pub fn is_offline(&self) -> bool {
        matches!(self, NetworkState::Offline)
    }
}

pub struct NetworkMonitorConfig {
    /// 在线时的探测间隔（较长，减少开销）
    pub online_check_interval: Duration,
    /// 离线时的探测间隔（较短，快速检测恢复）
    pub offline_check_interval: Duration,
    /// 降级延迟阈值（超过此值视为 Degraded）
    pub degraded_threshold_ms: u64,
    /// 探测目标 URL 列表（按顺序尝试，任一可达即视为在线）
    pub probe_urls: Vec<String>,
    /// 探测超时
    pub probe_timeout: Duration,
    /// 连续失败多少次才切换到 Offline（防抖）
    pub offline_threshold: u32,
    /// 连续成功多少次才从 Offline 切换到 Online（防抖）
    pub online_threshold: u32,
}

impl Default for NetworkMonitorConfig {
    fn default() -> Self {
```

### 3.2 探测逻辑

```rust
impl NetworkMonitor {
    pub fn new(config: NetworkMonitorConfig) -> Self {
        let (state_tx, state_rx) = watch::channel(NetworkState::Online);
        Self {
            state: Arc::new(RwLock::new(NetworkState::Online)),
            state_tx,
            state_rx,
            config,
            on_change_callbacks: Arc::new(RwLock::new(Vec::new())),
        }
    }

    /// 启动主动探测循环
    pub async fn start(&self, app: tauri::AppHandle) {
        let state = self.state.clone();
        let state_tx = self.state_tx.clone();
        let config = self.config.clone();
        let callbacks = self.on_change_callbacks.clone();

        tokio::spawn(async move {
            let mut consecutive_failures = 0u32;
            let mut consecutive_successes = 0u32;

            loop {
                // 根据当前状态选择探测间隔
                let current = state.read().await.clone();
                let interval = if current.is_offline() {
                    config.offline_check_interval
                } else {
                    config.online_check_interval
                };

                tokio::time::sleep(interval).await;

                let probe_result = Self::probe_connectivity(&config).await;
                let old_state = state.read().await.clone();

                let new_state = match probe_result {
                    ProbeResult::Reachable { latency_ms } => {
                        consecutive_failures = 0;
                        consecutive_successes += 1;

                        if latency_ms > config.degraded_threshold_ms {
                            NetworkState::Degraded { latency_ms }
                        } else if old_state.is_offline()
                            && consecutive_successes < config.online_threshold
                        {
                            // 从离线恢复需要连续成功 N 次（防抖）
                            old_state.clone()
                        } else {
                            NetworkState::Online
                        }
                    }
                    ProbeResult::Unreachable => {
                        consecutive_successes = 0;
                        consecutive_failures += 1;

                        if consecutive_failures >= config.offline_threshold {
                            NetworkState::Offline
                        } else {
                            old_state.clone()
                        }
                    }
                };

                if new_state != old_state {
                    tracing::info!(
                        "网络状态变更：{:?} → {:?}",
                        old_state, new_state
                    );
                    *state.write().await = new_state.clone();
                    state_tx.send(new_state.clone()).ok();

                    // 通知前端
                    app.emit("network-state-changed", &new_state).ok();

                    // 执行回调
                    let cbs = callbacks.read().await;
                    for cb in cbs.iter() {
                        cb(&old_state, &new_state);
                    }
                }
            }
        });
    }

    /// 对配置的目标 URL 列表做连通性探测
    async fn probe_connectivity(config: &NetworkMonitorConfig) -> ProbeResult {
        let client = reqwest::Client::builder()
            .timeout(config.probe_timeout)
            .build()
            .unwrap_or_default();

        for url in &config.probe_urls {
            let start = tokio::time::Instant::now();
            match client.head(url).send().await {
                Ok(resp) if resp.status().is_success()
                    || resp.status().as_u16() == 204
                    || resp.status().as_u16() == 401 =>
                {
                    // 401 也算可达（说明网络通了，只是没认证）
                    return ProbeResult::Reachable {
                        latency_ms: start.elapsed().as_millis() as u64,
                    };
                }
                _ => continue,
            }
        }

        ProbeResult::Unreachable
    }

    /// 获取当前网络状态（无阻塞读）
    pub async fn current_state(&self) -> NetworkState {
        self.state.read().await.clone()
    }

    /// 注册状态变更回调
    pub async fn on_state_change<F>(&self, callback: F)
    where
        F: Fn(&NetworkState, &NetworkState) + Send + Sync + 'static,
    {
        self.on_change_callbacks.write().await.push(Box::new(callback));
    }
}

enum ProbeResult {
    Reachable { latency_ms: u64 },
    Unreachable,
}
```

### 3.5 自动切换执行器

将 `NetworkMonitor` 和 `OfflinePolicy` 连接起来，在状态变更时执行实际的 Provider 切换。

```rust
// crates/agent-core/src/network/auto_switch.rs

use crate::network::monitor::{NetworkMonitor, NetworkState};
use crate::network::offline_policy::OfflinePolicy;

/// 网络状态变更时自动切换 Provider
pub struct AutoSwitchController {
    network_monitor: Arc<NetworkMonitor>,
    offline_policy: OfflinePolicy,
    /// 当前活跃的 Provider 模型标识
    active_model: Arc<RwLock<String>>,
    /// 切换前的云端模型（用于恢复）
    cloud_model_backup: Arc<RwLock<Option<String>>>,
}

impl AutoSwitchController {
    /// 监听网络状态并执行自动切换
    pub async fn start(
        &self,
        registry: Arc<Registry>,
        sidecar: Arc<OllamaSidecar>,
        discovery: Arc<LocalModelDiscovery>,
        app: AppHandle,
    ) {
        let mut rx = self.network_monitor.state_rx.clone();
        let policy = self.offline_policy.clone();
        let active_model = self.active_model.clone();
        let cloud_backup = self.cloud_model_backup.clone();

        tokio::spawn(async move {
            while rx.changed().await.is_ok() {
                let new_state = rx.borrow().clone();

                match new_state {
                    NetworkState::Offline => {
                        if !policy.auto_switch_enabled { continue; }
                        Self::switch_to_offline(
                            &policy, &registry, &sidecar, &discovery,
                            &active_model, &cloud_backup, &app,
                        ).await;
                    }
                    NetworkState::Degraded { latency_ms } => {
                        if policy.switch_on_degraded {
                            Self::switch_to_offline(
                                &policy, &registry, &sidecar, &discovery,
                                &active_model, &cloud_backup, &app,
                            ).await;
                        } else {
                            tracing::info!(
                                "网络延迟 {}ms（阈值 {}ms），保持云端模型",
                                latency_ms, policy.switch_latency_target.as_millis()
                            );
                        }
                    }
                    NetworkState::Online => {
                        // 恢复在线，切回云端模型
                        Self::switch_to_online(
                            &policy, &registry, &active_model,
                            &cloud_backup, &app,
                        ).await;
                    }
                }
            }
        });
    }

    /// 切换到离线模式
    async fn switch_to_offline(
        policy: &OfflinePolicy,
        registry: &Registry,
        sidecar: &OllamaSidecar,
        discovery: &LocalModelDiscovery,
        active_model: &RwLock<String>,
        cloud_backup: &RwLock<Option<String>>,
        app: &AppHandle,
    ) {
        let start = tokio::time::Instant::now();

        // 检查 Ollama 是否可用
        let sidecar_status = sidecar.status_rx.borrow().clone();
        if !matches!(sidecar_status, SidecarStatus::Running { .. }) {
            tracing::warn!("离线切换失败：Ollama sidecar 未运行");
            app.emit("offline-switch-failed", serde_json::json!({
                "reason": "本地模型运行时（Ollama）未就绪"
            })).ok();
            return;
        }

        // 获取可用的本地模型
        let available_models = discovery.list_models().await;

        // 按策略选择最佳本地模型
        let selected = policy.fallback_models.iter()
            .find(|fb| available_models.iter().any(|m| m.name == fb.model))
            .map(|fb| fb.model.clone());

        let Some(local_model) = selected else {
            tracing::warn!("离线切换失败：无可用的本地模型");
            app.emit("offline-switch-failed", serde_json::json!({
                "reason": "未安装任何本地模型，请先下载"
            })).ok();
            return;
        };

        // 备份当前云端模型
        let current = active_model.read().await.clone();
        *cloud_backup.write().await = Some(current);

        // 执行切换
        *active_model.write().await = local_model.clone();
        registry.set_active_model(&local_model).await;

        let elapsed = start.elapsed();
        tracing::info!(
            "已切换到本地模型 {}（耗时 {}ms，目标 {}ms）",
            local_model, elapsed.as_millis(),
            policy.switch_latency_target.as_millis()
        );

        // 通知前端
        if policy.notification.show_on_switch {
            let message = policy.notification.switch_message
                .replace("{model}", &local_model);
            app.emit("offline-mode-activated", serde_json::json!({
                "model": local_model,
                "message": message,
                "switch_latency_ms": elapsed.as_millis() as u64,
            })).ok();
        }
    }

    /// 切换回在线模式
    async fn switch_to_online(
        policy: &OfflinePolicy,
        registry: &Registry,
        active_model: &RwLock<String>,
        cloud_backup: &RwLock<Option<String>>,
        app: &AppHandle,
    ) {
        let backup = cloud_backup.read().await.clone();
        if let Some(cloud_model) = backup {
            *active_model.write().await = cloud_model.clone();
            registry.set_active_model(&cloud_model).await;
            *cloud_backup.write().await = None;

            tracing::info!("已恢复云端模型 {}", cloud_model);

            if policy.notification.show_on_recovery {
                app.emit("online-mode-restored", serde_json::json!({
                    "model": cloud_model,
                    "message": &policy.notification.recovery_message,
                })).ok();
            }
        }
    }
}
```

## 4. 离线可用功能范围

### 4.1 功能可用性矩阵

离线状态下各功能模块的可用性取决于其是否依赖网络请求。以下矩阵详细列出每项功能在不同网络状态下的行为：

| 功能模块 | 在线 | Degraded | 离线 | 离线替代方案 |
|---------|:----:|:--------:|:----:|------------|
| **文本对话** | 云端 LLM | 云端 LLM（可能慢） | 本地 Ollama 模型 | 自动切换，能力受限 |
| **工具执行** | 全部可用 | 全部可用 | 仅本地工具 | 见下方工具分类 |
| **嵌入向量** | 云端 Embedding API | 云端 API | 本地 ONNX 模型 | 使用 `ort`（ONNX Runtime）运行本地嵌入模型 |
| **知识库搜索** | 云端嵌入 + 本地向量库 | 云端嵌入 | 本地嵌入 + 本地向量库 | 精度略降但可用 |
| **浏览历史对话** | 本地 SQLite | 本地 SQLite | 本地 SQLite | 完全不受影响 |
| **Skill 管理** | 本地文件 | 本地文件 | 本地文件 | 完全不受影响 |
| **进化日志** | 本地文件 | 本地文件 | 本地文件 | 完全不受影响 |
| **本地备份** | 本地文件 | 本地文件 | 本地文件 | 完全不受影响 |
| **web_search** | 可用 | 可用（可能慢） | 不可用 | 返回明确错误提示 |
| **web_fetch** | 可用 | 可用（可能慢） | 不可用 | 返回明确错误提示 |
| **媒体生成**（TTS/图像/视频/音乐） | 云端 API | 云端 API | 不可用 | 返回明确错误提示 |
| **ASR 语音识别** | 云端 API | 云端 API | Whisper.cpp 本地 | 本地模型精度低于云端 |
| **插件市场** | 可用 | 可用 | 不可用 | 展示缓存的已安装插件列表 |
| **Model Insights** | 实时更新 | 实时更新 | 展示缓存数据 | 上次在线时的快照 |
| **设置同步** | 可用 | 可用 | 本地修改暂存 | 恢复在线后同步 |

### 4.2 工具离线分类

Agent 工具按网络依赖分为三类，离线时仅可执行本地工具：

```rust
// crates/agent-core/src/tools/offline_filter.rs

/// 工具的网络依赖分类
#[derive(Debug, Clone, PartialEq)]
pub enum ToolNetworkRequirement {
    /// 完全本地执行，不需要网络
    Local,
    /// 需要网络，离线时不可用
    RequiresNetwork,
    /// 部分功能需要网络，离线时降级运行
    Degradable,
}

/// 根据网络状态过滤可用工具列表
pub fn filter_tools_for_network_state(
    tools: &[ToolDefinition],
    network: &NetworkState,
) -> Vec<ToolDefinition> {
    if network.is_available() {
        return tools.to_vec(); // 在线时全部可用
    }

    // 离线时仅保留本地工具和可降级工具
    tools.iter()
        .filter(|t| {
            let req = classify_tool_network(t.name.as_str());
            matches!(req, ToolNetworkRequirement::Local | ToolNetworkRequirement::Degradable)
        })
        .cloned()
        .collect()
}

/// 分类已知工具的网络依赖
fn classify_tool_network(tool_name: &str) -> ToolNetworkRequirement {
    match tool_name {
        // 纯本地工具
        "file_read" | "file_write" | "file_search"
        | "code_execute" | "shell_command"
        | "calculator" | "calendar"
        | "knowledge_search"   // 使用本地向量库
        | "memory_read" | "memory_write"
            => ToolNetworkRequirement::Local,

        // 需要网络的工具
        "web_search" | "web_fetch" | "web_browse"
        | "image_generate" | "video_generate"
        | "tts_synthesize" | "music_generate"
        | "plugin_install" | "plugin_search"
            => ToolNetworkRequirement::RequiresNetwork,

        // 可降级工具
        "asr_transcribe"   // 可降级到本地 Whisper
        | "embedding"      // 可降级到本地 ONNX
            => ToolNetworkRequirement::Degradable,

        // 未知工具默认需要网络
        _ => ToolNetworkRequirement::RequiresNetwork,
    }
}
```

### 4.3 本地嵌入（ONNX Runtime）

知识库搜索的嵌入向量在离线时通过本地 ONNX 模型生成，确保知识库搜索始终可用。

```rust
// crates/agent-core/src/embedding/local_onnx.rs

use ort::{Session, SessionBuilder, Value};

/// 本地 ONNX 嵌入模型客户端
pub struct LocalOnnxEmbedding {
    session: Session,
    tokenizer: tokenizers::Tokenizer,
    /// 嵌入维度
    dim: usize,
    /// 模型名称（用于日志和调试）
    model_name: String,
}

impl LocalOnnxEmbedding {
    /// 加载预打包的 ONNX 嵌入模型
    pub fn load(model_dir: &Path) -> anyhow::Result<Self> {
        let model_path = model_dir.join("model.onnx");
        let tokenizer_path = model_dir.join("tokenizer.json");

        let session = SessionBuilder::new()?
            .with_optimization_level(ort::GraphOptimizationLevel::Level3)?
            .with_intra_threads(4)?
            .commit_from_file(&model_path)?;

        let tokenizer = tokenizers::Tokenizer::from_file(&tokenizer_path)
            .map_err(|e| anyhow::anyhow!("加载 tokenizer 失败: {}", e))?;

        // 推断嵌入维度
        let dim = session.outputs[0].dimensions()
            .last()
            .and_then(|d| *d)
            .unwrap_or(384) as usize;

        Ok(Self {
            session,
            tokenizer,
            dim,
            model_name: "bge-small-zh-v1.5".to_string(), // 默认打包模型
        })
    }

    /// 生成嵌入向量
    pub fn embed(&self, texts: &[String]) -> anyhow::Result<Vec<Vec<f32>>> {
        let mut all_embeddings = Vec::with_capacity(texts.len());

        for text in texts {
            let encoding = self.tokenizer.encode(text.as_str(), true)
                .map_err(|e| anyhow::anyhow!("tokenize 失败: {}", e))?;

            let input_ids: Vec<i64> = encoding.get_ids()
                .iter().map(|&id| id as i64).collect();
            let attention_mask: Vec<i64> = encoding.get_attention_mask()
                .iter().map(|&m| m as i64).collect();
            let token_type_ids: Vec<i64> = encoding.get_type_ids()
                .iter().map(|&t| t as i64).collect();

            let seq_len = input_ids.len();
            let inputs = vec![
                Value::from_array(([1, seq_len], input_ids.as_slice()))?,
                Value::from_array(([1, seq_len], attention_mask.as_slice()))?,
                Value::from_array(([1, seq_len], token_type_ids.as_slice()))?,
            ];

            let outputs = self.session.run(inputs)?;
            let embedding: Vec<f32> = outputs[0].extract_tensor::<f32>()?
                .view()
                .iter()
                .take(self.dim)
                .copied()
                .collect();

            // L2 归一化
            let norm: f32 = embedding.iter().map(|x| x * x).sum::<f32>().sqrt();
            let normalized: Vec<f32> = if norm > 0.0 {
                embedding.iter().map(|x| x / norm).collect()
            } else {
                embedding
            };

            all_embeddings.push(normalized);
        }

        Ok(all_embeddings)
    }
}

impl EmbeddingClient for LocalOnnxEmbedding {
    async fn embed(&self, req: EmbeddingRequest) -> Result<Vec<Vec<f32>>, ProviderError> {
        // ONNX 推理是 CPU 同步操作，放到 blocking 线程池
        let texts = req.texts.clone();
        let session = self.clone(); // 需要 Session 支持 Clone 或使用 Arc
        tokio::task::spawn_blocking(move || {
            session.embed(&texts)
                .map_err(|e| ProviderError::InternalServerError {
                    code: 500,
                    message: format!("本地嵌入失败: {}", e),
                })
        }).await.map_err(|e| ProviderError::InternalServerError {
            code: 500,
            message: format!("spawn_blocking 失败: {}", e),
        })?
    }

    fn embedding_dim(&self) -> usize {
        self.dim
    }
}
```

## 5. 在线恢复与同步

### 5.1 恢复检测

`NetworkMonitor` 通过防抖机制（连续 N 次探测成功）确认网络真正恢复后，触发在线恢复流程。避免网络抖动导致频繁切换。

```rust

/// 在线恢复控制器
pub struct OnlineRecoveryController {

    registry: Arc<Registry>,

/// 离线期间累积的待同步操作

    /// 创建时间
```

### 5.2 恢复流程

网络恢复后的处理是有序的，避免瞬间大量请求导致再次触发限流。

```rust
impl OnlineRecoveryController {
    /// 网络恢复时执行的完整恢复流程
    pub async fn on_network_restored(&self, app: &AppHandle) {
        tracing::info!("网络已恢复，开始在线恢复流程");

        // 阶段 1：重新验证云端 Provider 可用性（关键路径，优先执行）
        self.revalidate_cloud_providers().await;

        // 阶段 2：切换回云端模型（AutoSwitchController 已处理，此处确认）
        // 由 AutoSwitchController 的 switch_to_online 负责

        // 阶段 3：重放累积的待同步操作（非关键路径，后台执行）
        self.replay_pending_operations(app).await;

        // 阶段 4：刷新缓存数据
        self.refresh_cached_data(app).await;

        tracing::info!("在线恢复流程完成");
    }

    /// 重新验证所有云端 Provider 的 API Key 和健康状态
    async fn revalidate_cloud_providers(&self) {
        let providers = self.registry.list_cloud_providers().await;

        for provider_id in providers {
            match self.registry.health_check(&provider_id).await {
                Ok(health) => {
                    tracing::info!(
                        "Provider {} 恢复可用（延迟 {}ms）",
                        provider_id, health.latency_p50_ms
                    );
                    // 重置该 Provider 的熔断器
                    self.registry.reset_circuit_breaker(&provider_id).await;
                }
                Err(e) => {
                    tracing::warn!(
                        "Provider {} 恢复检查失败：{}（保持离线状态）",
                        provider_id, e
                    );
                }
            }
        }
    }

    /// 重放离线期间累积的待同步操作
    async fn replay_pending_operations(&self, app: &AppHandle) {
        let mut ops = self.pending_operations.write().await;
        if ops.is_empty() { return; }

        tracing::info!("重放 {} 个待同步操作", ops.len());

        // 按创建时间排序
        ops.sort_by_key(|op| op.created_at);

        let mut succeeded = 0usize;
        let mut failed = 0usize;

        for op in ops.drain(..) {
            // 操作间插入短暂延迟，避免突发请求
            tokio::time::sleep(Duration::from_millis(200)).await;

            match self.execute_pending_op(&op).await {
                Ok(()) => succeeded += 1,
                Err(e) => {
                    failed += 1;
                    tracing::warn!(
                        "待同步操作执行失败（{:?}）：{}",
                        op.op_type, e
                    );
                }
            }
        }

        if succeeded > 0 || failed > 0 {
            tracing::info!(
                "待同步操作重放完成：{} 成功，{} 失败",
                succeeded, failed
            );
        }
    }

    async fn execute_pending_op(&self, op: &PendingOperation) -> anyhow::Result<()> {
        match &op.op_type {
            PendingOpType::SettingsSync => {
                // 将本地设置变更推送到云端
                tracing::debug!("同步设置变更");
                Ok(())
            }
            PendingOpType::ModelInsightsRefresh => {
                // 拉取最新的模型信息
                tracing::debug!("刷新 Model Insights");
                Ok(())
            }
            PendingOpType::PluginUpdateCheck => {
                // 检查插件更新
                tracing::debug!("检查插件更新");
                Ok(())
            }
            PendingOpType::ConversationBackup { conversation_id } => {
                // 备份指定对话到云端
                tracing::debug!("备份对话 {}", conversation_id);
                Ok(())
            }
        }
    }

    /// 刷新离线期间可能过期的缓存数据
    async fn refresh_cached_data(&self, app: &AppHandle) {
        // Model Insights 缓存
        app.emit("refresh-model-insights", ()).ok();

        // 插件市场列表
        app.emit("refresh-plugin-market", ()).ok();
    }
}
```

### 5.3 平滑过渡 — 不中断当前对话

在线恢复时最关键的用户体验要求是：**不中断正在进行的对话**。切换过程对用户透明。

```rust
/// 对话中途的 Provider 平滑切换
pub async fn seamless_provider_switch(
    conversation: &mut ConversationContext,
    old_model: &str,
    new_model: &str,
    registry: &Registry,
) -> anyhow::Result<()> {
    // 1. 对话历史保持不变——所有消息继续保留
    //    ChatCompletionRequest 的 messages 字段是模型无关的标准格式

    // 2. 切换活跃模型
    conversation.active_model = new_model.to_string();

    // 3. 在对话元数据中记录切换事件（用于审计和调试）
    conversation.metadata.push(ConversationEvent::ProviderSwitch {
        from: old_model.to_string(),
        to: new_model.to_string(),
        timestamp: chrono::Utc::now(),
        reason: if new_model.contains(':') {
            "offline_fallback".to_string()
        } else {
            "online_recovery".to_string()
        },
    });

    // 4. 如果新模型的上下文窗口小于当前历史长度，执行压缩
    let new_client = registry.chat_completion_model(new_model)?;
    let max_tokens = new_client.max_context_tokens();
    let current_tokens = conversation.estimate_token_count();

    if current_tokens > max_tokens as usize {
        tracing::info!(
            "模型切换后上下文超限（{} > {}），执行消息压缩",
            current_tokens, max_tokens
        );
        conversation.compress_history(max_tokens as usize).await?;
    }

    // 5. 如果有正在进行的流式响应，标记当前 chunk 为最后一个
    //    后续消息将使用新模型重新生成（不自动续传——避免上下文不一致）

    Ok(())
}
```

### 6.1 OllamaProvider 实现 ChatCompletionModel

```rust
    }
}

impl ChatCompletionModel for OllamaProvider {
    async fn chat(&self, req: ChatCompletionRequest) -> Result<ChatCompletionResponse, ProviderError> {
        // 使用 OpenAI 兼容端点
        let url = format!("{}/v1/chat/completions", self.base_url);

        let body = serde_json::json!({
            "model": self.model,
            "messages": req.messages.iter().map(|m| m.to_openai_format()).collect::<Vec<_>>(),
            "stream": false,
            "temperature": req.temperature.unwrap_or(0.7),
            "max_tokens": req.max_tokens.unwrap_or(4096),
            // 如果模型支持工具调用且请求中有工具定义
            "tools": if self.capabilities.tool_use && !req.tools.is_empty() {
                serde_json::to_value(&req.tools).unwrap_or_default()
            } else {
                serde_json::Value::Null
            },
        });

        let resp = self.http.post(&url)
            .json(&body)
            .send()
            .await
            .map_err(|e| {
                if e.is_connect() {
                    ProviderError::NetworkTimeout
                } else {
                    ProviderError::ServerError {
                        code: 500,
                        message: format!("Ollama 请求失败: {}", e),
                    }
                }
            })?;

        if !resp.status().is_success() {
            let status = resp.status().as_u16();
            let body_text = resp.text().await.unwrap_or_default();

            // 404 可能意味着模型未安装或 API 路径不支持
            if status == 404 {
                return Err(ProviderError::ModelNotFound(self.model.clone()));
            }

            return Err(ProviderError::ServerError {
                code: status,
                message: body_text,
            });
        }

        let response_body: serde_json::Value = resp.json().await
            .map_err(|e| ProviderError::InvalidRequest(format!("响应解析失败: {}", e)))?;

        // 解析 OpenAI 兼容格式的响应
        Ok(ChatCompletionResponse::from_openai_format(&response_body, "ollama"))
    }

    fn chat_stream(
        &self,
        req: ChatCompletionRequest,
    ) -> BoxStream<'static, Result<StreamChunk, ProviderError>> {
        let url = format!("{}/v1/chat/completions", self.base_url);
        let model = self.model.clone();
        let capabilities = self.capabilities.clone();
        let http = self.http.clone();

        Box::pin(async_stream::try_stream! {
            let body = serde_json::json!({
                "model": model,
                "messages": req.messages.iter()
                    .map(|m| m.to_openai_format()).collect::<Vec<_>>(),
                "stream": true,
                "temperature": req.temperature.unwrap_or(0.7),
                "max_tokens": req.max_tokens.unwrap_or(4096),
                "tools": if capabilities.tool_use && !req.tools.is_empty() {
                    serde_json::to_value(&req.tools).unwrap_or_default()
                } else {
                    serde_json::Value::Null
                },
            });

            let resp = http.post(&url)
                .json(&body)
                .send()
                .await
                .map_err(|e| ProviderError::ServerError {
                    code: 500,
                    message: format!("Ollama 流式请求失败: {}", e),
                })?;

            let mut stream = resp.bytes_stream();
            let mut buffer = String::new();

            while let Some(chunk) = stream.next().await {
                let chunk = chunk.map_err(|e| ProviderError::StreamInterrupted {
                    received_chunks: 0,
                    partial_content: buffer.clone(),
                })?;

                let text = String::from_utf8_lossy(&chunk);
                for line in text.lines() {
                    let line = line.trim();
                    if line.is_empty() || line == "data: [DONE]" { continue; }
                    let data = line.strip_prefix("data: ").unwrap_or(line);

                    if let Ok(parsed) = serde_json::from_str::<serde_json::Value>(data) {
                        if let Some(delta) = parsed
                            .get("choices")
                            .and_then(|c| c.get(0))
                            .and_then(|c| c.get("delta"))
                            .and_then(|d| d.get("content"))
                            .and_then(|c| c.as_str())
                        {
                            buffer.push_str(delta);
                            yield StreamChunk::TextDelta(delta.to_string());
                        }
                    }
                }
            }
        })
    }

    fn provider_id(&self) -> &str {
        "ollama"
    }

    fn max_context_tokens(&self) -> u32 {
        self.capabilities.context_length
    }
}
```

### 6.2 FailoverChain 中的优先级排列

本地 Ollama 模型在 FailoverChain 中作为**最后兜底**，仅在所有云端 Provider 和降级模型都失败后启用。

```rust
// crates/agent-providers/src/failover.rs 中的集成

/// 构建包含本地模型兜底的 FailoverClient
pub fn build_failover_chain(
    config: &ProvidersConfig,
    registry: &Registry,
    ollama: Option<Arc<OllamaProvider>>,
) -> FailoverClient {
    // 优先级顺序：
    // 1. 用户首选的云端模型（primary）
    // 2. 同 Provider 降级模型（如 claude-sonnet-5 → claude-haiku-4-5）
    // 3. 跨 Provider 备选（如 Claude 不可用 → OpenAI）
    // 4. 本地 Ollama 模型（最后兜底）

    let primary = registry.chat_completion_model(&config.default_text)
        .expect("默认模型必须已注册");

    let mut fallbacks: Vec<Arc<dyn ChatCompletionModel>> = Vec::new();

    // 同 Provider 降级
    if let Some(downgrade) = FALLBACK_MODEL_MAP.get(config.default_text.as_str()) {
        if let Ok(client) = registry.chat_completion_model(downgrade) {
            fallbacks.push(client);
        }
    }

    // 跨 Provider 备选
    for alt_model in &config.cross_provider_fallbacks {
        if let Ok(client) = registry.chat_completion_model(alt_model) {
            fallbacks.push(client);
        }
    }

    // Ollama 本地兜底
    if let Some(ollama_client) = ollama {
        fallbacks.push(ollama_client);
    }

    FailoverClient {
        primary,
        fallbacks,
        circuit_breakers: build_circuit_breakers(&config),
        retry_policy: RetryPolicy::default(),
        app: config.app_handle.clone(),
    }
}
```

### 6.3 费用追踪 — 本地模型零成本

本地模型的调用不产生 API 费用。费用追踪系统对 Ollama Provider 记录用量统计但费用归零。

```rust
// crates/agent-core/src/cost/tracker.rs 中的集成

impl CostTracker {
    /// 记录一次模型调用的费用
    pub async fn record_usage(
        &self,
        provider_id: &str,
        model_id: &str,
        input_tokens: u32,
        output_tokens: u32,
    ) {
        let (input_cost, output_cost) = if provider_id == "ollama" {
            // 本地模型：零费用，但仍记录 token 用量（用于统计）
            (0.0, 0.0)
        } else {
            let pricing = self.pricing_table.get(model_id);
            (
                input_tokens as f64 * pricing.cost_per_input_token,
                output_tokens as f64 * pricing.cost_per_output_token,
            )
        };

        sqlx::query!(
            "INSERT INTO usage_logs (id, provider, model, input_tokens, output_tokens, \
             input_cost, output_cost, is_local, created_at)
             VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
            Uuid::new_v4().to_string(),
            provider_id,
            model_id,
            input_tokens,
            output_tokens,
            input_cost,
            output_cost,
            provider_id == "ollama",
            chrono::Utc::now().timestamp_millis(),
        ).execute(&self.pool).await.ok();
    }
}
```

### 6.4 模型能力映射

不同本地模型的能力不同，需要映射到统一的能力体系中，以便 Provider `Registry` 正确路由请求。

```rust
// crates/agent-providers/src/providers/ollama/capability_map.rs

/// 将本地模型映射到 Provider Registry 的能力体系
pub fn map_ollama_to_registry_capabilities(
    model_info: &LocalModelInfo,
) -> RegistryModelCapabilities {
    RegistryModelCapabilities {
        text: true,
        vision: model_info.capabilities.vision,
        tool_use: model_info.capabilities.tool_use,
        embedding: model_info.capabilities.embedding,
        // 本地模型不支持以下云端能力
        tts: false,
        asr: false,
        image_generation: false,
        video_generation: false,
        music_generation: false,
        // 性能特征
        max_context: model_info.capabilities.context_length,
        typical_latency_ms: estimate_local_latency(&model_info.parameter_size),
        // 质量评级（相对于云端模型的粗略估计）
        quality_tier: estimate_quality_tier(
            &model_info.family,
            &model_info.parameter_size,
        ),
    }
}

/// 根据参数量估算本地推理延迟（首 token）
fn estimate_local_latency(parameter_size: &str) -> u64 {
    match parameter_size {
        s if s.contains("1B") || s.contains("3B") => 200,   // 小模型，快
        s if s.contains("7B") || s.contains("8B") => 500,   // 中等模型
        s if s.contains("13B") || s.contains("14B") => 1500, // 大模型
        s if s.contains("32B") || s.contains("34B") => 3000, // 很大
        s if s.contains("70B") => 8000,                       // 超大
        _ => 1000,
    }
}

/// 根据模型家族和参数量估算质量等级
fn estimate_quality_tier(family: &str, parameter_size: &str) -> QualityTier {
    let is_large = parameter_size.contains("14B")
        || parameter_size.contains("32B")
        || parameter_size.contains("70B");

    match (family, is_large) {
        ("qwen2.5", true) | ("llama3.3", true) | ("phi4", true) => {
            QualityTier::Good // 接近中等云端模型
        }
        ("qwen2.5", false) | ("llama3.2", _) | ("mistral", _) => {
            QualityTier::Adequate // 基本可用
        }
        _ => QualityTier::Basic, // 仅供应急
    }
}

#[derive(Debug, Clone, PartialEq)]
pub enum QualityTier {
    /// 接近中等云端模型质量
    Good,
    /// 基本可用，适合简单对话
    Adequate,
    /// 仅供应急，能力有限
    Basic,
}
```
