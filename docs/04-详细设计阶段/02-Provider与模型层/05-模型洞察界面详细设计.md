# 模型洞察界面详细设计

> 版本：v1.0 | 日期：2026-08-09 | 状态：草稿
> 对应需求：F-26 模型洞察界面（使用统计、性能排行、成本分析、模型推荐）
> 系统设计：[08-模型洞察界面.md](../../03-系统设计阶段/06-桌面端/08-模型洞察界面.md)

模型洞察界面（Model Insights）是桌面端独立页面，综合展示用户在本地积累的模型使用数据：使用统计、性能排行榜、成本分析、模型推荐。数据来源为 `usage_metrics`、`cost_records` 两张 SQLite 表以及 Eval 评估结果，所有计算与展示均在本地完成，不依赖云端基础设施。

> **边界说明**：多模型对比（并发发送 + 并排查看）已在 [02-多模型对比系统设计.md](./02-多模型对比系统设计.md) 中完整覆盖，本文不重复。本文聚焦"事后洞察"——基于历史使用数据做统计、排行、分析与推荐。

## 1. 洞察面板概述

### 1.1 面板定位

模型洞察面板回答三个核心问题：

1. **我在用什么？** — 各模型的使用量、调用频次、Token 消耗趋势
2. **哪个模型更好？** — 基于质量（Eval 分数）、速度（TTFT + tokens/s）、成本效率的综合排行
3. **如何优化支出？** — 成本拆解、趋势预测、省钱建议、最优模型推荐

### 1.2 数据源

| 数据源 | 表/来源 | 提供信息 |
|--------|---------|---------|
| 使用指标 | `usage_metrics` | 每轮 Token 计数、延迟、TTFT、LLM/Tool 调用次数 |
| 成本记录 | `cost_records` | 每次 LLM 调用的精细费用（按 provider + model 拆分） |
| Eval 结果 | `eval_runs` + `eval_case_results` | 模型质量分数（pass_rate、task_completion） |
| 模型目录 | OpenRouter API / 本地缓存 | 模型价格、能力、基准测试数据 |
| 内存指标 | `MetricsCollector` | 实时性能百分位数（P50/P95/P99） |

### 2.2 聚合查询

按模型分组的使用统计查询，从 `usage_metrics` 表聚合。支持 7 天 / 30 天 / 自定义时间范围。

```sql
-- 按模型分组的使用统计（7 天窗口）
SELECT
    um.model,
    COUNT(*)                             AS call_count,
    SUM(um.prompt_tokens)                AS total_prompt_tokens,
    SUM(um.completion_tokens)            AS total_completion_tokens,
    COALESCE(SUM(um.cost_usd), 0.0)     AS total_cost_usd,
    ROUND(AVG(um.latency_ms))           AS avg_latency_ms,
    ROUND(AVG(um.ttft_ms))             AS avg_ttft_ms,
    -- 成功率：无 error status 的轮次 / 总轮次
    ROUND(1.0 * SUM(CASE WHEN s.status = 'ok' THEN 1 ELSE 0 END) / COUNT(*), 3)
                                         AS success_rate,
    -- 平均输出速度
    ROUND(AVG(
        CASE WHEN um.latency_ms > 0
             THEN um.completion_tokens * 1000.0 / um.latency_ms
             ELSE 0 END
    ), 1)                                AS avg_tokens_per_sec
FROM usage_metrics um
LEFT JOIN spans s
    ON s.conversation_id = um.conversation_id
   AND s.turn_index = um.turn_index
   AND s.parent_span_id IS NULL            -- 根 span
JOIN conversations c ON um.conversation_id = c.id
WHERE c.workspace_id = ?
  AND um.created_at >= ?                   -- from_ms
  AND um.created_at <= ?                   -- to_ms
GROUP BY um.model
ORDER BY total_cost_usd DESC;
```

```sql
-- 按天的趋势数据（概览面板折线图）
SELECT
    date(um.created_at / 1000, 'unixepoch') AS date,
    COUNT(*)                                 AS call_count,
    SUM(um.prompt_tokens)                    AS prompt_tokens,
    SUM(um.completion_tokens)                AS completion_tokens,
    COALESCE(SUM(um.cost_usd), 0.0)         AS cost_usd,
    ROUND(AVG(um.latency_ms))               AS avg_latency_ms
FROM usage_metrics um
JOIN conversations c ON um.conversation_id = c.id
WHERE c.workspace_id = ?
  AND um.created_at >= ?
GROUP BY date
ORDER BY date ASC;
```

```sql
-- 同比对比：当前周期 vs 上一周期（计算 delta 百分比）
WITH current_period AS (
    SELECT SUM(cost_usd) AS cost, COUNT(*) AS calls,
           SUM(prompt_tokens + completion_tokens) AS tokens,
           AVG(latency_ms) AS latency
    FROM usage_metrics um
    JOIN conversations c ON um.conversation_id = c.id
    WHERE c.workspace_id = ? AND um.created_at BETWEEN ? AND ?
),
previous_period AS (
    SELECT SUM(cost_usd) AS cost, COUNT(*) AS calls,
           SUM(prompt_tokens + completion_tokens) AS tokens,
           AVG(latency_ms) AS latency
    FROM usage_metrics um
    JOIN conversations c ON um.conversation_id = c.id
    WHERE c.workspace_id = ? AND um.created_at BETWEEN ? AND ?
)
SELECT
    c.cost, c.calls, c.tokens, c.latency,
    CASE WHEN p.cost > 0 THEN (c.cost - p.cost) / p.cost ELSE NULL END AS cost_delta,
    CASE WHEN p.calls > 0 THEN (c.calls - p.calls) * 1.0 / p.calls ELSE NULL END AS calls_delta,
    CASE WHEN p.tokens > 0 THEN (c.tokens - p.tokens) * 1.0 / p.tokens ELSE NULL END AS tokens_delta,
    CASE WHEN p.latency > 0 THEN (c.latency - p.latency) / p.latency ELSE NULL END AS latency_delta
FROM current_period c, previous_period p;
```

```rust
// crates/agent-core/src/storage/repos/insights_repo.rs

pub struct InsightsRepo {
    pool: SqlitePool,
}

impl InsightsRepo {
    pub fn new(pool: &SqlitePool) -> Self {
        Self { pool: pool.clone() }
    }

    /// 按模型分组的使用统计
    pub async fn get_model_usage_stats(
        &self,
        workspace_id: &str,
        from_ms: i64,
        to_ms: i64,
    ) -> anyhow::Result<Vec<ModelUsageStats>> {
        let rows = sqlx::query_as::<_, ModelUsageStatsRow>(
            "SELECT um.model, COUNT(*) AS call_count,
                    SUM(um.prompt_tokens) AS total_prompt_tokens,
                    SUM(um.completion_tokens) AS total_completion_tokens,
                    COALESCE(SUM(um.cost_usd), 0.0) AS total_cost_usd,
                    ROUND(AVG(um.latency_ms)) AS avg_latency_ms,
                    ROUND(AVG(um.ttft_ms)) AS avg_ttft_ms,
                    ROUND(AVG(CASE WHEN um.latency_ms > 0
                        THEN um.completion_tokens * 1000.0 / um.latency_ms
                        ELSE 0 END), 1) AS avg_tokens_per_sec
             FROM usage_metrics um
             JOIN conversations c ON um.conversation_id = c.id
             WHERE c.workspace_id = ? AND um.created_at BETWEEN ? AND ?
             GROUP BY um.model
             ORDER BY total_cost_usd DESC"
        )
        .bind(workspace_id)
        .bind(from_ms)
        .bind(to_ms)
        .fetch_all(&self.pool)
        .await?;
        Ok(rows.into_iter().map(Into::into).collect())
    }

    /// 概览汇总 + 同比 delta
    pub async fn get_overview(
        &self,
        workspace_id: &str,
        from_ms: i64,
        to_ms: i64,
    ) -> anyhow::Result<InsightsOverviewData> {
        let period_len = to_ms - from_ms;
        let prev_from = from_ms - period_len;
        let prev_to = from_ms;

        let current = self.get_period_summary(workspace_id, from_ms, to_ms).await?;
        let previous = self.get_period_summary(workspace_id, prev_from, prev_to).await?;

        Ok(InsightsOverviewData {
            total_calls: current.calls,
            total_tokens: current.tokens,
            total_cost_usd: current.cost,
            avg_latency_ms: current.latency,
            calls_delta: safe_delta(current.calls as f64, previous.calls as f64),
            tokens_delta: safe_delta(current.tokens as f64, previous.tokens as f64),
            cost_delta: safe_delta(current.cost, previous.cost),
            latency_delta: safe_delta(current.latency, previous.latency),
        })
    }

    /// 按天趋势
    pub async fn get_trend(
        &self,
        workspace_id: &str,
        from_ms: i64,
        to_ms: i64,
    ) -> anyhow::Result<Vec<TrendPoint>> {
        let rows = sqlx::query_as::<_, TrendPointRow>(
            "SELECT date(um.created_at / 1000, 'unixepoch') AS date,
                    COUNT(*) AS call_count,
                    SUM(um.prompt_tokens) AS prompt_tokens,
                    SUM(um.completion_tokens) AS completion_tokens,
                    COALESCE(SUM(um.cost_usd), 0.0) AS cost_usd,
                    ROUND(AVG(um.latency_ms)) AS avg_latency_ms
             FROM usage_metrics um
             JOIN conversations c ON um.conversation_id = c.id
             WHERE c.workspace_id = ? AND um.created_at BETWEEN ? AND ?
             GROUP BY date ORDER BY date ASC"
        )
        .bind(workspace_id)
        .bind(from_ms)
        .bind(to_ms)
        .fetch_all(&self.pool)
        .await?;
        Ok(rows.into_iter().map(Into::into).collect())
    }

    // ── 内部辅助 ──

    async fn get_period_summary(
        &self,
        workspace_id: &str,
        from_ms: i64,
        to_ms: i64,
    ) -> anyhow::Result<PeriodSummary> {
        let row = sqlx::query_as::<_, PeriodSummaryRow>(
            "SELECT COUNT(*) AS calls,
                    COALESCE(SUM(prompt_tokens + completion_tokens), 0) AS tokens,
                    COALESCE(SUM(cost_usd), 0.0) AS cost,
                    COALESCE(AVG(latency_ms), 0.0) AS latency
             FROM usage_metrics um
             JOIN conversations c ON um.conversation_id = c.id
             WHERE c.workspace_id = ? AND um.created_at BETWEEN ? AND ?"
        )
        .bind(workspace_id)
        .bind(from_ms)
        .bind(to_ms)
        .fetch_one(&self.pool)
        .await?;
        Ok(row.into())
    }
}

fn safe_delta(current: f64, previous: f64) -> Option<f64> {
    if previous > 0.0 {
        Some((current - previous) / previous)
    } else {
        None
    }
}
```

## 3. 性能排行榜

### 3.1 排行维度

ModelLeaderboard 对用户实际使用过的模型进行排名，支持三个维度：

| 维度 | 排名依据 | 数据来源 |
|------|---------|---------|
| **质量** | Eval pass_rate + task_completion 加权 | `eval_runs` / `eval_case_results` |
| **速度** | TTFT (P50) + tokens/s (P50) 综合 | `usage_metrics` + `MetricsCollector` |
| **成本效率** | quality_score / cost_per_1m_output_tokens | 质量分 + `cost_records` 单价 |

### 3.2 评分计算

```typescript
// apps/desktop/src/lib/leaderboard.ts

interface LeaderboardEntry {
  model: string;
  provider: string;
  qualityScore: number;       // 0-100，来自 Eval
  speedScore: number;         // 0-100，综合 TTFT + tokens/s
  costEfficiencyScore: number; // 0-100，quality / cost 归一化
  overallScore: number;        // 三维加权综合分
  rank: number;
  // 明细
  evalPassRate: number | null;
  avgTtftMs: number;
  avgTokensPerSec: number;
  costPer1mOutput: number;
  callCount: number;
}

/** 计算质量分（0-100） */
function computeQualityScore(
  evalPassRate: number | null,
  taskCompletion: number | null,
): number {
  // 有 Eval 数据：pass_rate 占 60%，task_completion 占 40%
  if (evalPassRate != null && taskCompletion != null) {
    return (evalPassRate * 0.6 + taskCompletion * 0.4) * 100;
  }
  // 仅有 pass_rate
  if (evalPassRate != null) {
    return evalPassRate * 100;
  }
  // 无 Eval 数据：返回 -1 表示无评分
  return -1;
}

/** 计算速度分（0-100），基于 TTFT 和 tokens/s 的归一化 */
function computeSpeedScore(
  ttftMs: number,
  tokensPerSec: number,
  allModels: { ttftMs: number; tokensPerSec: number }[],
): number {
  // TTFT 越低越好，归一化到 0-1（反向）
  const ttfts = allModels.map(m => m.ttftMs);
  const minTtft = Math.min(...ttfts);
  const maxTtft = Math.max(...ttfts);
  const ttftNorm = maxTtft > minTtft
    ? 1 - (ttftMs - minTtft) / (maxTtft - minTtft)
    : 1;

  // tokens/s 越高越好，归一化到 0-1
  const speeds = allModels.map(m => m.tokensPerSec);
  const minSpeed = Math.min(...speeds);
  const maxSpeed = Math.max(...speeds);
  const speedNorm = maxSpeed > minSpeed
    ? (tokensPerSec - minSpeed) / (maxSpeed - minSpeed)
    : 1;

  // TTFT 权重 40%，tokens/s 权重 60%
  return (ttftNorm * 0.4 + speedNorm * 0.6) * 100;
}

/** 计算成本效率分（0-100） */
function computeCostEfficiencyScore(
  qualityScore: number,
  costPer1mOutput: number,
  allModels: { qualityScore: number; costPer1mOutput: number }[],
): number {
  if (qualityScore < 0 || costPer1mOutput <= 0) return -1;

  // quality / cost 比值，归一化
  const ratio = qualityScore / costPer1mOutput;
  const allRatios = allModels
    .filter(m => m.qualityScore >= 0 && m.costPer1mOutput > 0)
    .map(m => m.qualityScore / m.costPer1mOutput);

  if (allRatios.length === 0) return -1;

  const minR = Math.min(...allRatios);
  const maxR = Math.max(...allRatios);
  return maxR > minR
    ? ((ratio - minR) / (maxR - minR)) * 100
    : 50;
}

/** 计算综合排行 */
function computeOverallScore(entry: LeaderboardEntry): number {
  const scores: number[] = [];
  const weights: number[] = [];

  if (entry.qualityScore >= 0) {
    scores.push(entry.qualityScore);
    weights.push(0.4);
  }
  scores.push(entry.speedScore);
  weights.push(entry.qualityScore >= 0 ? 0.3 : 0.5);

  if (entry.costEfficiencyScore >= 0) {
    scores.push(entry.costEfficiencyScore);
    weights.push(0.3);
  } else {
    // 无成本效率分时，速度权重增加
    weights[weights.length - 1] += 0.3;
  }

  const totalWeight = weights.reduce((a, b) => a + b, 0);
  return scores.reduce((sum, s, i) => sum + s * weights[i], 0) / totalWeight;
}
```

### 3.3 排行榜数据查询

```sql
-- 从 Eval 结果获取各模型质量分
SELECT
    er.model,
    AVG(er.pass_rate)          AS avg_pass_rate,
    AVG(er.task_completion)    AS avg_task_completion,
    COUNT(DISTINCT er.id)      AS eval_run_count
FROM eval_runs er
WHERE er.workspace_id = ?
  AND er.created_at >= ?
GROUP BY er.model;

-- 从 usage_metrics 获取各模型速度指标
SELECT
    model,
    AVG(ttft_ms) FILTER (WHERE ttft_ms IS NOT NULL) AS avg_ttft_ms,
    AVG(
        CASE WHEN latency_ms > 0
             THEN completion_tokens * 1000.0 / latency_ms
             ELSE NULL END
    ) AS avg_tokens_per_sec,
    COUNT(*) AS call_count
FROM usage_metrics um
JOIN conversations c ON um.conversation_id = c.id
WHERE c.workspace_id = ?
  AND um.created_at >= ?
GROUP BY model
HAVING call_count >= 5;   -- 至少 5 次调用才纳入排行

-- 从 cost_records 获取各模型单价（最近一次记录的实际单价）
SELECT DISTINCT model,
    cost_usd * 1000000.0 / NULLIF(output_tokens, 0) AS cost_per_1m_output
FROM cost_records
WHERE workspace_id = ?
  AND output_tokens > 0
GROUP BY model
HAVING recorded_at = MAX(recorded_at);
```

## 4. 成本分析面板

### 4.1 核心功能

CostAnalysis 面板提供四个子视图：

1. **按模型拆分** — 饼图 + 表格，展示各模型的费用占比
2. **按时间趋势** — 折线图，展示每日/每周/每月的费用走势
3. **月度预估** — 基于近期使用量线性外推本月预估总费用
4. **省钱建议** — 基于使用模式，推荐更经济的替代模型

### 4.2 数据结构


### 4.3 月度费用预估

```rust
// crates/agent-core/src/insights/cost_projector.rs

pub struct CostProjector;

impl CostProjector {
    /// 基于最近 N 天的平均日费用，线性外推本月总费用
    pub fn project_monthly_cost(
        daily_costs: &[DailyCostPoint],
        today: NaiveDate,
    ) -> ProjectionResult {
        if daily_costs.is_empty() {
            return ProjectionResult::insufficient_data();
        }

        // 使用最近 7 天（或可用天数）计算日均费用
        let recent_days = daily_costs.iter().rev().take(7).collect::<Vec<_>>();
        let avg_daily = recent_days.iter()
            .map(|d| d.total_cost)
            .sum::<f64>() / recent_days.len() as f64;

        // 本月已花费（到今天）
        let month_start = today.with_day(1).unwrap();
        let spent_this_month: f64 = daily_costs.iter()
            .filter(|d| {
                NaiveDate::parse_from_str(&d.date, "%Y-%m-%d")
                    .map(|nd| nd >= month_start && nd <= today)
                    .unwrap_or(false)
            })
            .map(|d| d.total_cost)
            .sum();

        // 本月剩余天数
        let days_in_month = month_start
            .with_month(today.month() % 12 + 1)
            .unwrap_or(month_start)
            .signed_duration_since(month_start)
            .num_days() as u32;
        let remaining_days = days_in_month - today.day();

        let projected = spent_this_month + avg_daily * remaining_days as f64;

        ProjectionResult {

            avg_daily_cost: avg_daily,
            spent_this_month,
            remaining_days,
            basis: format!("基于最近 {} 天平均日费用 ${:.2}", recent_days.len(), avg_daily),
        }
    }
}
```

```rust
// crates/agent-core/src/insights/saving_advisor.rs

pub struct SavingAdvisor {
    pool: SqlitePool,
}

impl SavingAdvisor {
    /// 分析用户使用模式，生成省钱建议
    pub async fn generate_suggestions(
        &self,
        workspace_id: &str,
        from_ms: i64,
        to_ms: i64,
        model_catalog: &ModelCatalog,
    ) -> anyhow::Result<Vec<SavingSuggestion>> {
        let mut suggestions = Vec::new();

        // 获取各模型使用统计
        let stats = InsightsRepo::new(&self.pool)
            .get_model_usage_stats(workspace_id, from_ms, to_ms)
            .await?;

        for stat in &stats {
            // 规则 1：高费用模型有更便宜的同族替代
            if let Some(suggestion) = self.check_cheaper_alternative(stat, model_catalog) {
                suggestions.push(suggestion);
            }

            // 规则 2：简单任务使用了昂贵模型（平均 output tokens < 200）
            if stat.avg_completion_tokens() < 200.0 && stat.cost_per_1m_output() > 10.0 {
                if let Some(cheaper) = model_catalog.find_cheapest_capable(&stat.model) {
                    suggestions.push(SavingSuggestion {
                        current_model: stat.model.clone(),
                        suggested_model: cheaper.id.clone(),
                        estimated_saving_pct: 1.0 - cheaper.price_output / stat.cost_per_1m_output(),
                        reason: format!(
                            "该模型的平均输出仅 {:.0} tokens/次，属于简单任务，\
                             可考虑切换到 {} 以降低成本",
                            stat.avg_completion_tokens(), cheaper.id
                        ),
                        confidence: Confidence::Medium,
                    });
                }
            }

            // 规则 3：有 Eval 数据时，对比质量相近但更便宜的模型
            // 质量差异 < 5% 且价格差异 > 30% 则建议切换
        }

        // 按预估节省金额降序
        suggestions.sort_by(|a, b| b.estimated_saving_usd
            .partial_cmp(&a.estimated_saving_usd)
            .unwrap_or(std::cmp::Ordering::Equal));

        Ok(suggestions)
    }

    fn check_cheaper_alternative(
        &self,
        stat: &ModelUsageStats,
        catalog: &ModelCatalog,
    ) -> Option<SavingSuggestion> {
        let current_price = catalog.get_output_price(&stat.model)?;
        let provider = catalog.get_provider(&stat.model)?;

        // 同 Provider 的更便宜模型
        let alternatives = catalog.models_by_provider(&provider)
            .filter(|m| {
                let alt_price = catalog.get_output_price(&m.id).unwrap_or(f64::MAX);
                alt_price < current_price * 0.7  // 至少便宜 30%
            })
            .collect::<Vec<_>>();

        let best = alternatives.iter()
            .max_by(|a, b| {
                let sa = catalog.get_context_length(&a.id).unwrap_or(0);
                let sb = catalog.get_context_length(&b.id).unwrap_or(0);
                sa.cmp(&sb)  // 同族中上下文最长的
            })?;

        let alt_price = catalog.get_output_price(&best.id)?;
        let saving_pct = 1.0 - alt_price / current_price;
        let saving_usd = stat.total_cost_usd * saving_pct;

        Some(SavingSuggestion {
            current_model: stat.model.clone(),
            suggested_model: best.id.clone(),
            estimated_saving_pct: saving_pct,
            estimated_saving_usd: saving_usd,
            reason: format!(
                "切换到 {} 可节省约 {:.0}% 费用（${:.2}/月），\
                 上下文长度 {}",
                best.id, saving_pct * 100.0, saving_usd,
                format_context_length(catalog.get_context_length(&best.id).unwrap_or(0))
            ),
            confidence: Confidence::High,
        })
    }
}
```

```rust
                cost_weight: 0.35,
                preferred_capabilities: vec![],
                min_context_length: 16_000,
            },
        }
    }

impl ModelRecommender {
    /// 生成模型推荐
    pub async fn recommend(
        &self,
        workspace_id: &str,
        request: &RecommendRequest,
        model_catalog: &ModelCatalog,
    ) -> anyhow::Result<RecommendResult> {
        let profile = TaskProfile::for_task(&request.task_type);
        let now = now_ms();
        let from = now - 30 * 86_400_000; // 最近 30 天数据

        // 1. 获取候选模型（用户用过的 + 目录中价格合适的）
        let used_models = InsightsRepo::new(&self.pool)
            .get_model_usage_stats(workspace_id, from, now)
            .await?;
        let catalog_models = model_catalog.all_text_models();

        // 2. 合并去重，过滤不符合基本条件的
        let mut candidates = self.build_candidates(
            &used_models, &catalog_models, &profile, &request.budget_constraint,
        );

        // 3. 打分排序
        for candidate in &mut candidates {
            candidate.score = self.score_candidate(
                candidate,
                &profile,
                &request.quality_requirement,
            );
        }
        candidates.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap());

        // 4. 构建结果
        let primary = candidates.first()
            .ok_or_else(|| anyhow::anyhow!("无可推荐模型"))?;
        let alternatives = candidates.iter().skip(1).take(3).cloned().collect();

        let reasoning = self.generate_reasoning(
            &request.task_type, primary, &profile,
        );

        Ok(RecommendResult {
            primary: primary.clone(),
            alternatives,
            reasoning,
        })
    }

    fn score_candidate(
        &self,
        candidate: &ModelRecommendation,
        profile: &TaskProfile,
        quality_req: &Option<QualityRequirement>,
    ) -> f64 {
        let mut quality_score = candidate.eval_pass_rate.unwrap_or(0.5) * 100.0;
        let mut speed_score = candidate.avg_tokens_per_sec
            .map(|tps| (tps / 200.0 * 100.0).min(100.0))  // 200 tok/s = 满分
            .unwrap_or(50.0);
        let cost_score = (1.0 - (candidate.estimated_cost_per_1m_output / 30.0).min(1.0)) * 100.0;
        // $30/1M = 0 分，$0 = 100 分

        // 应用用户偏好
        let (qw, sw, cw) = match quality_req.as_ref().map(|q| &q.priority) {
            Some(Priority::Quality) => (0.6, 0.15, 0.25),
            Some(Priority::Speed)   => (0.2, 0.55, 0.25),
            Some(Priority::Cost)    => (0.15, 0.2, 0.65),
            _ => (profile.quality_weight, profile.speed_weight, profile.cost_weight),
        };

        quality_score * qw + speed_score * sw + cost_score * cw
    }

    fn generate_reasoning(
        &self,
        task_type: &TaskType,
        primary: &ModelRecommendation,
        profile: &TaskProfile,
    ) -> String {
        let task_label = match task_type {
            TaskType::Coding => "编码任务",
            TaskType::Writing => "写作任务",
            TaskType::Analysis => "分析任务",
            TaskType::Chat => "日常对话",
            TaskType::Translation => "翻译任务",
            TaskType::General => "通用任务",
        };

        let mut parts = vec![
            format!(
                "针对{}，推荐 {}（综合得分 {:.0}）。",
                task_label, primary.model, primary.score
            ),
        ];

        if let Some(pass_rate) = primary.eval_pass_rate {
            parts.push(format!("Eval 通过率 {:.0}%。", pass_rate * 100.0));
        }

        if let Some(tps) = primary.avg_tokens_per_sec {
            parts.push(format!("输出速度 {:.0} tokens/s。", tps));
        }

        parts.push(format!(
            "费用 ${:.2}/1M output tokens。",
            primary.estimated_cost_per_1m_output
        ));

        parts.join("")
    }
}
```

```rust
// crates/agent-core/src/insights/aggregator.rs

pub struct InsightsAggregator {
    pool: SqlitePool,
}

impl InsightsAggregator {
    /// 刷新每日汇总表（增量更新：只重算今天和昨天的数据）
    pub async fn refresh_daily_summary(&self, workspace_id: &str) -> anyhow::Result<()> {
        let today = chrono::Utc::now().format("%Y-%m-%d").to_string();
        let yesterday = (chrono::Utc::now() - chrono::Duration::days(1))
            .format("%Y-%m-%d").to_string();

        for date in [&yesterday, &today] {
            sqlx::query(
                "INSERT OR REPLACE INTO daily_model_summary
                    (date, workspace_id, model, provider, call_count,
                     prompt_tokens, completion_tokens, cost_usd,
                     avg_latency_ms, avg_ttft_ms, success_count, failure_count)
                 SELECT
                    date(um.created_at / 1000, 'unixepoch') AS date,
                    c.workspace_id,
                    um.model,
                    COALESCE(cr.provider, 'unknown') AS provider,
                    COUNT(*) AS call_count,
                    SUM(um.prompt_tokens) AS prompt_tokens,
                    SUM(um.completion_tokens) AS completion_tokens,
                    COALESCE(SUM(um.cost_usd), 0.0) AS cost_usd,
                    ROUND(AVG(um.latency_ms)) AS avg_latency_ms,
                    ROUND(AVG(um.ttft_ms)) AS avg_ttft_ms,
                    SUM(CASE WHEN um.latency_ms > 0 THEN 1 ELSE 0 END),
                    SUM(CASE WHEN um.latency_ms = 0 THEN 1 ELSE 0 END)
                 FROM usage_metrics um
                 JOIN conversations c ON um.conversation_id = c.id
                 LEFT JOIN (
                     SELECT DISTINCT model, provider
                     FROM cost_records WHERE workspace_id = ?
                 ) cr ON cr.model = um.model
                 WHERE c.workspace_id = ?
                   AND date(um.created_at / 1000, 'unixepoch') = ?
                 GROUP BY date, c.workspace_id, um.model"
            )
            .bind(workspace_id)
            .bind(workspace_id)
            .bind(date)
            .execute(&self.pool)
            .await?;
        }

        Ok(())
    }

    /// 清理超过 90 天的汇总数据
    pub async fn cleanup(&self, retention_days: u32) -> anyhow::Result<u64> {
        let cutoff = (chrono::Utc::now() - chrono::Duration::days(retention_days as i64))
            .format("%Y-%m-%d").to_string();
        let result = sqlx::query("DELETE FROM daily_model_summary WHERE date < ?")
            .bind(&cutoff)
            .execute(&self.pool)
            .await?;
        Ok(result.rows_affected())
    }
}
```

### 6.2 查询性能优化

```sql
    SUM(prompt_tokens) AS prompt_tokens,

    SUM(success_count) AS success_count,
```

```rust
    let cache_key = format!("{}:{}:{:?}:{:?}",
        workspace_id, task_type, max_cost_per_1m, priority);
    if let Some(cached) = state.recommender_cache.get(&cache_key) {
        return Ok(cached);
    }

    let recommender = ModelRecommender::new(&state.db_pool);
    let result = recommender.recommend(
        &workspace_id, &request, &state.model_catalog,
    ).await?;

    state.recommender_cache.put(cache_key, result.clone());
    Ok(result)
}

/// 导出洞察数据
#[tauri::command]
pub async fn export_insights(
    state: State<'_, AppState>,
    workspace_id: String,
    window: String,
    output_path: String,
) -> Result<String, AppError> {
    let (from, to) = window_to_range(&window);
    let repo = InsightsRepo::new(&state.db_pool);

    let overview = repo.get_overview(&workspace_id, from, to).await?;
    let model_stats = repo.get_model_usage_stats(&workspace_id, from, to).await?;
    let cost_data = repo.get_cost_by_model_detail(&workspace_id, from, to).await?;
    let trend = repo.get_trend(&workspace_id, from, to).await?;

    let export = serde_json::json!({
        "workspace_id": workspace_id,
        "window": window,
        "exported_at": chrono::Utc::now().to_rfc3339(),
        "overview": overview,
        "model_stats": model_stats,
        "cost_breakdown": cost_data,
        "trend": trend,
    });

    let content = serde_json::to_string_pretty(&export)?;
    tokio::fs::write(&output_path, &content).await?;
    Ok(output_path)
}

fn window_to_range(window: &str) -> (i64, i64) {
    let now = now_ms();
    let from = match window {
        "7d"  => now - 7 * 86_400_000,
        "30d" => now - 30 * 86_400_000,
        "90d" => now - 90 * 86_400_000,
        _     => now - 7 * 86_400_000,
    };
    (from, now)
}
```
