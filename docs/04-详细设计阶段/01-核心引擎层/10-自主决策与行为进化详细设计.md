## 2. 行为策略引擎（BehaviorEngine）

### 2.1 核心数据结构

```rust
// crates/agent-core/src/behavior/mod.rs

use std::sync::Arc;
use tokio::sync::{mpsc, RwLock};

/// 行为策略引擎：决定 Agent HOW to respond，而非 WHAT to respond
pub struct BehaviorEngine {
    feedback_collector: FeedbackCollector,
    preference_learner: PreferenceLearner,
    tool_pattern_learner: ToolPatternLearner,
    prompt_optimizer: PromptOptimizer,
    proactive_engine: ProactiveDecisionEngine,
    behavior_repo: BehaviorRepo,
    config: BehaviorConfig,
    /// 接收来自 AgentRuntime 的任务完成事件（与 EvolutionEngine 共享）
    task_rx: mpsc::Receiver<TaskCompletedEvent>,
    /// 接收来自 HookPipeline 的交互信号
    interaction_rx: mpsc::Receiver<InteractionSignal>,
}

/// 行为引擎配置
pub struct BehaviorConfig {
    /// 行为分析周期（默认每日一次）
    pub analysis_interval: Duration,
    /// 每周期最大调整数量（防震荡）
    pub max_adjustments_per_cycle: u32,
    /// 单维度最大调整幅度（0.0~1.0 范围内每次最多偏移 0.15）
    pub max_adjustment_magnitude: f32,
    /// 偏好学习的指数衰减系数（越新的信号权重越高，默认 0.95）
    pub decay_factor: f32,
    /// 是否启用行为学习（用户可全局关闭）
    pub learning_enabled: bool,
    /// 是否启用主动行为
    pub proactive_enabled: bool,
    /// 回归检测：连续 N 次负反馈触发回滚（默认 3）
    pub regression_threshold: u32,
}

impl Default for BehaviorConfig {
    fn default() -> Self {
        Self {
            analysis_interval: Duration::from_secs(24 * 3600),
            max_adjustments_per_cycle: 2,
            max_adjustment_magnitude: 0.15,
            decay_factor: 0.95,
            learning_enabled: true,
            proactive_enabled: true,
            regression_threshold: 3,
        }
    }
}
```

### 2.2 后台服务启动

```rust
impl BehaviorEngine {
    pub fn spawn(mut self) -> tokio::task::JoinHandle<()> {
        tokio::spawn(async move {
            let mut analysis_interval = tokio::time::interval(self.config.analysis_interval);

            loop {
                tokio::select! {
                    // 即时模式：收到任务完成事件后采集反馈信号
                    Some(event) = self.task_rx.recv() => {
                        if self.config.learning_enabled {
                            self.on_task_completed(&event).await;
                        }
                    }

                    // 即时模式：收到交互信号（编辑、重新生成等）
                    Some(signal) = self.interaction_rx.recv() => {
                        if self.config.learning_enabled {
                            self.on_interaction_signal(&signal).await;
                        }
                    }

                    // 定期模式：行为进化分析
                    _ = analysis_interval.tick() => {
                        if self.config.learning_enabled {
                            if let Err(e) = self.run_evolution_cycle().await {
                                tracing::warn!("行为进化周期执行失败: {e}");
                            }
                        }
                    }
                }
            }
        })
    }
}
```

### 2.3 四维行为策略

BehaviorEngine 在四个维度上学习和调整 Agent 行为：

| 维度 | 学习信号 | 调整目标 |
|------|---------|---------|
| 沟通风格 | 用户追问频率、消息编辑、明确反馈 | 响应长度、解释深度、代码注释密度 |
| 主动建议 | 建议被采纳率 vs 被忽略率 | 建议频率、建议时机、建议类型 |
| 工具偏好 | 工具链成功率、用户对工具结果的修改 | 优先选择成功率高的工具组合 |
| 风险容忍 | 权限请求的批准率、自动执行后的回滚率 | L0/L1 动作是否自动执行 |

## 3. 用户反馈信号收集（FeedbackCollector）

### 3.1 信号类型

```rust
// crates/agent-core/src/behavior/feedback.rs

use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};

/// 反馈信号类型
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum FeedbackSignal {
    // ── 隐式信号（从用户行为推断） ──

    /// 用户在 Agent 响应后立即编辑自己的消息并重新发送
    /// 推断：Agent 误解了意图，原始回答方向错误
    MessageEdited {
        original_hash: u64,
        edited_hash: u64,
        edit_distance_ratio: f32, // 编辑距离 / 原始长度，越大说明改动越大
    },

    /// 用户对同一话题立即追问
    /// 推断：回答不够完整或不够清晰
    FollowUpQuestion {
        time_gap_ms: u64,     // 与上一条回复的间隔
        is_clarification: bool, // 是否为澄清性质（"你的意思是..."、"能不能详细说..."）
    },

    /// 用户重新生成回答
    /// 推断：对回答质量不满意
    ResponseRegenerated {
        attempt_number: u32,
    },

    /// 对话在 Agent 回复后被放弃（用户没有后续交互，session 超时关闭）
    /// 推断：可能满意（已得到答案），也可能不满意（放弃）
    ConversationAbandoned {
        last_message_role: String, // "assistant" = Agent 最后说话
        session_duration_ms: u64,
        total_turns: u32,
    },

    /// 用户对 Agent 提议的操作点击了"批准"或"拒绝"
    PermissionDecision {
        tool_name: String,
        approved: bool,
        response_time_ms: u64, // 用户决策耗时（快速批准 = 信任）
    },

    /// Agent 建议被用户采纳（用户点击了建议的操作）
    SuggestionAccepted {
        suggestion_type: String,
        suggestion_id: String,
    },

    /// Agent 建议被用户忽略（显示后超过 30 秒未操作，或用户发送了不相关的新消息）
    SuggestionIgnored {
        suggestion_type: String,
        suggestion_id: String,
    },

    // ── 显式信号（用户主动给出） ──

    /// 点赞/点踩
    ThumbsRating {
        message_id: String,
        is_positive: bool,
    },

    /// 用户明确纠正（"不要这样做"、"以后用 X 代替 Y"）
    ExplicitCorrection {
        correction_text: String,
        category: CorrectionCategory,
    },

    /// 用户明确表扬（"很好"、"完美"、"就是这样"）
    ExplicitPraise {
        praise_text: String,
    },
}

/// 纠正类别
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum CorrectionCategory {
    StyleCorrection,    // "太啰嗦了"、"简洁点"
    ToolCorrection,     // "不要用 X 工具"、"用 Y 代替"
    BehaviorCorrection, // "不要主动建议"、"先问我"
    ContentCorrection,  // "这个答案不对"
    Unknown,
}

/// 反馈事件（持久化单元）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FeedbackEvent {
    pub id: String,
    pub workspace_id: String,
    pub conversation_id: String,
    pub signal: FeedbackSignal,
    pub context: FeedbackContext,
    pub created_at: i64,
}

/// 反馈发生时的上下文快照
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FeedbackContext {
    /// 当时 Agent 的回复长度（token 数）
    pub response_length_tokens: u32,
    /// 当时使用的工具列表
    pub tools_used: Vec<String>,
    /// 当时的任务类型标签（coding / qa / creative / ...）
    pub task_type: String,
    /// 当时 Agent 是否给出了建议
    pub had_suggestion: bool,
    /// 当时的 turn index
    pub turn_index: u32,
}
```

### 3.3 信号采集实现

隐式信号的采集通过三总线 Hook 系统实现。BehaviorEngine 计划注册一个 Plugin Hook，监听 `UserPromptSubmit`、`PostLlmCall` 等 canonical 事件（当前 Hook 事件名遵循 Codex 对齐命名，完整列表见 `crates/agent-hooks/src/event.rs` 的 `HookEvent` 枚举）：

```rust
impl FeedbackCollector {
    /// 从 HookContext 中提取隐式反馈信号
    pub fn extract_implicit_signals(
        &self,
        event: &HookEvent,
        ctx: &HookContext,
        history: &ConversationHistory,
    ) -> Vec<FeedbackSignal> {
        let mut signals = Vec::new();

        match event {
            HookEvent::UserPromptSubmit => {
                // 检测追问模式
                if let Some(prev_assistant) = history.last_assistant_message() {
                    let gap = ctx.metadata.get("message_timestamp")
                        .and_then(|v| v.as_i64())
                        .unwrap_or(0) - prev_assistant.timestamp;

                    if gap < 30_000 { // 30 秒内追问
                        let is_clarification = self.is_clarification_question(
                            &ctx.payload, &prev_assistant.content,
                        );
                        signals.push(FeedbackSignal::FollowUpQuestion {
                            time_gap_ms: gap as u64,
                            is_clarification,
                        });
                    }
                }

                // 检测消息编辑（用户修改上一条消息重新发送）
                if let Some(edit_info) = ctx.metadata.get("edit_of_message_id") {
                    if let HookPayload::MessageData { content, .. } = &ctx.payload {
                        let original = history.get_message(edit_info.as_str().unwrap_or(""));
                        if let Some(orig) = original {
                            let ratio = edit_distance(&orig.content, content) as f32
                                / orig.content.len().max(1) as f32;
                            signals.push(FeedbackSignal::MessageEdited {
                                original_hash: hash(&orig.content),
                                edited_hash: hash(content),
                                edit_distance_ratio: ratio,
                            });
                        }
                    }
                }
            }
            _ => {}
        }

        signals
    }

    /// 检测是否为澄清性质的追问
    fn is_clarification_question(&self, payload: &HookPayload, _prev: &str) -> bool {
        if let HookPayload::MessageData { content, .. } = payload {
            const CLARIFICATION_PATTERNS: &[&str] = &[
                "你的意思是", "能不能详细", "什么意思", "不太明白",
                "能否解释", "具体来说", "换句话说", "也就是说",
                "could you explain", "what do you mean", "can you elaborate",
            ];
            CLARIFICATION_PATTERNS.iter().any(|p| content.contains(p))
        } else {
            false
        }
    }
}
```

## 4. 偏好学习模型（PreferenceLearner）

### 4.1 偏好维度定义

```rust
// crates/agent-core/src/behavior/preference.rs

/// 用户偏好维度（每个维度是 0.0~1.0 的加权分数）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct UserPreferences {
    /// 响应长度偏好：0.0=极简（一行）→ 1.0=详尽（多段落）
    pub response_length: WeightedScore,

    /// 代码风格偏好：0.0=精简无注释 → 1.0=防御式编程+详细注释
    pub code_style: WeightedScore,

    /// 主动性偏好：0.0=被动（只回答问题）→ 1.0=高度主动（预测需求、自动行动）
    pub proactivity: WeightedScore,

    /// 解释深度偏好：0.0=直接给结论 → 1.0=分步推理过程全展示
    pub explanation_depth: WeightedScore,

    /// 冻结的维度列表（用户明确要求不再自动调整的维度）
    pub frozen_dimensions: Vec<String>,

    /// 上次更新时间
    pub updated_at: i64,
}

/// 带衰减权重的分数
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct WeightedScore {
    /// 当前分值（0.0~1.0）
    pub value: f32,
    /// 置信度：样本越多越高（0.0~1.0）
    pub confidence: f32,
    /// 累计样本数
    pub sample_count: u32,
}

impl WeightedScore {
    pub fn new(initial: f32) -> Self {
        Self {
            value: initial,
            confidence: 0.0,
            sample_count: 0,
        }
    }

    /// 用指数衰减更新分值
    ///
    /// new_value = decay * old_value + (1 - decay) * observation
    /// confidence 随样本数递增，上限 1.0
    pub fn update(&mut self, observation: f32, decay: f32) {
        self.value = decay * self.value + (1.0 - decay) * observation;
        self.value = self.value.clamp(0.0, 1.0);
        self.sample_count += 1;
        // 置信度：10 个样本达到 0.5，50 个样本达到 0.9
        self.confidence = 1.0 - (-0.05 * self.sample_count as f32).exp();
    }
}

impl Default for UserPreferences {
    fn default() -> Self {
        Self {
            response_length: WeightedScore::new(0.5),   // 默认中等
            code_style: WeightedScore::new(0.5),
            proactivity: WeightedScore::new(0.5),
            explanation_depth: WeightedScore::new(0.5),
            frozen_dimensions: Vec::new(),
            updated_at: 0,
        }
    }
}
```

### 4.2 信号到偏好的映射规则

```rust
impl PreferenceLearner {
    /// 将一批反馈信号聚合为偏好更新
    pub fn process_signals(
        &self,
        prefs: &mut UserPreferences,
        signals: &[FeedbackEvent],
        decay: f32,
    ) {
        for event in signals {
            // 跳过已冻结的维度
            match &event.signal {
                FeedbackSignal::FollowUpQuestion { is_clarification, .. } => {
                    if *is_clarification && !prefs.is_frozen("explanation_depth") {
                        // 用户追问要求解释 → 增加解释深度
                        prefs.explanation_depth.update(0.8, decay);
                    }
                    if !prefs.is_frozen("response_length") {
                        // 追问 = 回答不够完整 → 增加响应长度
                        prefs.response_length.update(0.7, decay);
                    }
                }

                FeedbackSignal::MessageEdited { edit_distance_ratio, .. } => {
                    // 大幅编辑 = Agent 误解意图，不直接影响偏好维度
                    // 但如果编辑是缩短消息，说明用户觉得太啰嗦
                    // edit_distance_ratio 高 → 可能不满意但方向不确定，不更新
                }

                FeedbackSignal::ResponseRegenerated { attempt_number } => {
                    // 重新生成 = 全面不满意，但无法判断具体维度
                    // 仅记录为负面信号，由 EvolutionLoop 综合分析
                    if *attempt_number >= 2 && !prefs.is_frozen("response_length") {
                        prefs.response_length.update(0.6, decay);
                    }
                }

                FeedbackSignal::ExplicitCorrection { category, .. } => {
                    match category {
                        CorrectionCategory::StyleCorrection => {
                            // "太啰嗦了" → 降低响应长度
                            // "太简短了" → 增加响应长度
                            // 具体方向由 correction_text 中的关键词判断
                            let direction = self.parse_style_direction(
                                &event.signal,
                            );
                            if !prefs.is_frozen("response_length") {
                                prefs.response_length.update(direction, decay);
                            }
                        }
                        CorrectionCategory::BehaviorCorrection => {
                            if !prefs.is_frozen("proactivity") {
                                // "不要主动建议" → 降低主动性
                                prefs.proactivity.update(0.1, decay);
                            }
                        }
                        _ => {}
                    }
                }

                FeedbackSignal::ExplicitPraise { .. } => {
                    // 表扬 → 当前回复的所有参数都正确，强化当前偏好值
                    // 不改变方向，只增加置信度
                    prefs.response_length.confidence =
                        (prefs.response_length.confidence + 0.05).min(1.0);
                    prefs.explanation_depth.confidence =
                        (prefs.explanation_depth.confidence + 0.05).min(1.0);
                }

                FeedbackSignal::ThumbsRating { is_positive, .. } => {
                    if !is_positive {
                        // 点踩：需要结合上下文分析具体维度
                        // 此处仅记录，由 EvolutionLoop 做深度分析
                    }
                    // 点赞同理，增加置信度
                }

                FeedbackSignal::SuggestionAccepted { .. } => {
                    if !prefs.is_frozen("proactivity") {
                        prefs.proactivity.update(0.8, decay);
                    }
                }

                FeedbackSignal::SuggestionIgnored { .. } => {
                    if !prefs.is_frozen("proactivity") {
                        prefs.proactivity.update(0.3, decay);
                    }
                }

                FeedbackSignal::PermissionDecision { approved, response_time_ms, .. } => {
                    // 快速批准（< 2秒）= 高信任 → 可适度提高自动执行
                    // 拒绝 = 保守 → 降低自动执行
                    if *approved && *response_time_ms < 2000 {
                        // 高信任信号，但不在偏好维度中直接体现
                        // 影响 ProactiveDecisionEngine 的风险阈值
                    }
                }

                _ => {}
            }
        }

        prefs.updated_at = now_ms();
    }

    /// 解析风格纠正方向（"太啰嗦" → 0.2，"太简短" → 0.8）
    fn parse_style_direction(&self, signal: &FeedbackSignal) -> f32 {
        if let FeedbackSignal::ExplicitCorrection { correction_text, .. } = signal {
            let text = correction_text.to_lowercase();
            const CONCISE_KEYWORDS: &[&str] = &[
                "啰嗦", "太长", "简洁", "简短", "废话", "直接",
                "too long", "verbose", "concise", "brief", "shorter",
            ];
            const VERBOSE_KEYWORDS: &[&str] = &[
                "太短", "详细", "多说", "展开", "解释",
                "too short", "more detail", "elaborate", "explain more",
            ];
            if CONCISE_KEYWORDS.iter().any(|k| text.contains(k)) {
                return 0.2; // 用户想要更简洁
            }
            if VERBOSE_KEYWORDS.iter().any(|k| text.contains(k)) {
                return 0.8; // 用户想要更详细
            }
        }
        0.5 // 无法判断方向，保持中性
    }
}

impl UserPreferences {
    fn is_frozen(&self, dimension: &str) -> bool {
        self.frozen_dimensions.iter().any(|d| d == dimension)
    }
}
```

UserPreferences 写入 L3 记忆层的 USER.md，并在下次 Session 启动时通过 Slot [4] 注入 System Prompt。注入格式：

## 5. 行为进化循环（EvolutionLoop）

### 5.1 进化周期流程

```text
累积反馈信号（24 小时窗口）
    │
    ▼
① 聚合：将未处理的 FeedbackEvent 按维度分组统计
    │
    ▼
② 分析：计算各维度的调整方向和幅度
    │
    ▼
③ 提案：生成 BehaviorAdjustment（最多 2 个/周期）
    │
    ▼
④ 约束检查：幅度 <= max_adjustment_magnitude，
            不触碰已冻结维度，不回退已收敛维度
    │
    ▼
⑤ 应用：更新 UserPreferences → 写入 USER.md
    │
    ▼
⑥ 监控：下一周期检查调整后的反馈趋势，
        连续 3 次负反馈 → 自动回滚
```

### 5.4 回归检测与回滚

```rust
impl EvolutionLoop {
    /// 检测最近的调整是否引起了回归
    async fn check_regression(
        &self,
        workspace_id: &str,
        dimension: &str,
    ) -> anyhow::Result<bool> {
        // 找到该维度最近一次 applied 的调整
        let last_adjustment = self.behavior_repo
            .get_latest_adjustment(workspace_id, dimension, AdjustmentStatus::Applied)
            .await?;

        let Some(adj) = last_adjustment else {
            return Ok(false);
        };

        // 计算调整后收到的负反馈数量
        let negative_count = self.behavior_repo
            .count_negative_feedback_since(workspace_id, adj.applied_at.unwrap_or(0))
            .await?;

        if negative_count >= self.config.regression_threshold {
            tracing::warn!(
                "维度 '{}' 调整后连续 {} 次负反馈，触发回滚",
                dimension, negative_count
            );

            // 回滚到调整前的值
            let mut prefs = self.behavior_repo
                .get_preferences(workspace_id).await?;

            match dimension {
                "response_length" => prefs.response_length.value = adj.old_value,
                "code_style" => prefs.code_style.value = adj.old_value,
                "proactivity" => prefs.proactivity.value = adj.old_value,
                "explanation_depth" => prefs.explanation_depth.value = adj.old_value,
                _ => {}
            }

            self.behavior_repo.save_preferences(workspace_id, &prefs).await?;
            self.behavior_repo.update_adjustment_status(
                &adj.id, AdjustmentStatus::Reverted,
            ).await?;

            return Ok(true);
        }

        Ok(false)
    }
}

## 6. Prompt 自优化（PromptOptimizer）

### 6.1 设计思路

Agent 的 System Prompt 中有多处可优化的措辞。PromptOptimizer 跟踪不同 prompt 变体在实际任务中的效果差异，自动选择表现最好的变体。优化范围限定在 SOUL.md 的可变段落，不触碰安全相关的固定指令。

```rust
// crates/agent-core/src/behavior/prompt_optimizer.rs

/// Prompt 变体
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PromptVariant {
    pub id: String,
    /// 变体所属的 SOUL.md 段落标识（如 "communication_style"）
    pub section_id: String,
    /// 变体文本
    pub content: String,
    /// 累计试用次数
    pub trial_count: u32,
    /// 累计正面反馈次数
    pub positive_count: u32,
    /// 是否为当前活跃变体
    pub is_active: bool,
    pub created_at: i64,
}

impl PromptVariant {
    /// 正面反馈率
    pub fn success_rate(&self) -> f32 {
        if self.trial_count == 0 { return 0.5; }
        self.positive_count as f32 / self.trial_count as f32
    }
}

/// Prompt 变体表
/// 存储 SOUL.md 各段落的候选措辞，跟踪每个变体的效果
pub struct PromptOptimizer {
    behavior_repo: BehaviorRepo,
    provider: Arc<dyn TextClient>,
}

impl PromptOptimizer {
    /// 基于近期反馈生成新的 prompt 候选变体
    pub async fn generate_variants(
        &self,
        section_id: &str,
        current_content: &str,
        recent_feedback: &[FeedbackEvent],
        count: usize,
    ) -> anyhow::Result<Vec<String>> {
        let feedback_summary: Vec<String> = recent_feedback.iter()
            .take(10)
            .map(|f| format!("- {:?}", f.signal))
            .collect();

        let prompt = format!(
            r#"你是 Prompt 优化专家。当前 System Prompt 的 "{section}" 段落如下：

{current}

近期用户反馈信号：
{feedback}

请生成 {n} 个改进版本的段落文本，用 "===VARIANT===" 分隔。
每个版本应尝试不同的优化方向（更具体、更灵活、不同措辞等）。
不要改变段落的核心意图，只优化表达方式。"#,
            section = section_id,
            current = current_content,
            feedback = feedback_summary.join("\n"),
            n = count,
        );

        let response = self.provider.generate(&prompt).await?;
        Ok(response.split("===VARIANT===")
            .map(|s| s.trim().to_string())
            .filter(|s| !s.is_empty())
            .collect())
    }

    /// 选择下一周期使用的变体（Thompson Sampling）
    pub fn select_variant(&self, variants: &[PromptVariant]) -> Option<&PromptVariant> {
        if variants.is_empty() { return None; }

        // Thompson Sampling：根据 Beta 分布采样
        // 优先探索试用次数少的变体，同时利用已知的高分变体
        let mut best_score = f32::MIN;
        let mut best = None;

        for v in variants {
            let alpha = v.positive_count as f32 + 1.0;
            let beta = (v.trial_count - v.positive_count) as f32 + 1.0;
            // 简化 Beta 分布采样为均值 + 随机扰动
            let mean = alpha / (alpha + beta);
            let noise = (rand::random::<f32>() - 0.5) * 2.0 / (v.trial_count as f32 + 5.0);
            let score = mean + noise;

            if score > best_score {
                best_score = score;
                best = Some(v);
            }
        }

        best
    }

    /// 评估是否有变体已收敛（试用 >= 20 次且优于当前活跃版本 10%+）
    pub async fn check_promotion(
        &self,
        section_id: &str,
        workspace_id: &str,
    ) -> anyhow::Result<Option<String>> {
        let variants = self.behavior_repo
            .list_variants(workspace_id, section_id).await?;

        let active = variants.iter().find(|v| v.is_active);
        let active_rate = active.map(|v| v.success_rate()).unwrap_or(0.5);

        for v in &variants {
            if !v.is_active
                && v.trial_count >= 20
                && v.success_rate() > active_rate + 0.10
            {
                return Ok(Some(v.id.clone()));
            }
        }

        Ok(None)
    }
}
```

## 7. 工具使用模式学习（ToolPatternLearner）

### 7.1 核心数据结构

```rust
// crates/agent-core/src/behavior/tool_pattern.rs

/// 工具使用模式（从成功任务中提取的工具调用序列模板）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ToolPattern {
    pub id: String,
    /// 任务类型标签（如 "code_review"、"file_refactor"）
    pub task_type: String,
    /// 工具调用序列（有序）
    pub tool_chain: Vec<String>,
    /// 该模式的成功次数
    pub success_count: u32,
    /// 该模式的总使用次数
    pub total_count: u32,
    /// 平均满意度评分
    pub avg_satisfaction: f32,
    /// 最近使用时间
    pub last_used_at: i64,
}

impl ToolPattern {
    pub fn success_rate(&self) -> f32 {
        if self.total_count == 0 { return 0.0; }
        self.success_count as f32 / self.total_count as f32
    }
}

pub struct ToolPatternLearner {
    behavior_repo: BehaviorRepo,
}

impl ToolPatternLearner {
    /// 从任务完成事件中提取工具链并更新模式库
    pub async fn record_pattern(
        &self,
        event: &TaskCompletedEvent,
    ) -> anyhow::Result<()> {
        if !event.outcome.is_success() || event.tools_used.len() < 2 {
            return Ok(());
        }

        let chain = event.tools_used.clone();
        let task_type = classify_task_type(&event.task_description);

        // 查找是否已存在匹配的模式
        let existing = self.behavior_repo
            .find_matching_pattern(&task_type, &chain)
            .await?;

        match existing {
            Some(mut pattern) => {
                pattern.total_count += 1;
                if event.outcome.is_success() {
                    pattern.success_count += 1;
                }
                if let Some(rating) = event.user_rating {
                    pattern.avg_satisfaction = (
                        pattern.avg_satisfaction * (pattern.total_count - 1) as f32
                        + rating
                    ) / pattern.total_count as f32;
                }
                pattern.last_used_at = now_ms();
                self.behavior_repo.update_pattern(&pattern).await?;
            }
            None => {
                let pattern = ToolPattern {
                    id: Uuid::new_v4().to_string(),
                    task_type,
                    tool_chain: chain,
                    success_count: if event.outcome.is_success() { 1 } else { 0 },
                    total_count: 1,
                    avg_satisfaction: event.user_rating.unwrap_or(0.5),
                    last_used_at: now_ms(),
                };
                self.behavior_repo.insert_pattern(&pattern).await?;
            }
        }

        Ok(())
    }

    /// 为给定任务类型推荐最佳工具链
    pub async fn recommend_chain(
        &self,
        task_type: &str,
        workspace_id: &str,
    ) -> anyhow::Result<Option<Vec<String>>> {
        let patterns = self.behavior_repo
            .list_patterns_by_task_type(workspace_id, task_type)
            .await?;

        // 选择成功率最高且样本量 >= 3 的模式
        let best = patterns.iter()
            .filter(|p| p.total_count >= 3)
            .max_by(|a, b| {
                a.success_rate().partial_cmp(&b.success_rate())
                    .unwrap_or(std::cmp::Ordering::Equal)
            });

        Ok(best.map(|p| p.tool_chain.clone()))
    }
}

/// 工具模式表
/// ```sql
/// CREATE TABLE IF NOT EXISTS tool_patterns (
///     id              TEXT PRIMARY KEY,
///     workspace_id    TEXT NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
///     task_type       TEXT NOT NULL,
///     tool_chain      TEXT NOT NULL,        -- JSON 数组
///     success_count   INTEGER NOT NULL DEFAULT 0,
///     total_count     INTEGER NOT NULL DEFAULT 0,
///     avg_satisfaction REAL NOT NULL DEFAULT 0.5,
///     last_used_at    INTEGER NOT NULL DEFAULT (unixepoch('now','subsec')*1000)
/// );
/// CREATE INDEX IF NOT EXISTS idx_tp_ws_task
///     ON tool_patterns(workspace_id, task_type);

/// 简单任务分类（基于关键词）
fn classify_task_type(description: &str) -> String {
    let d = description.to_lowercase();
    if d.contains("review") || d.contains("审查") { return "code_review".into(); }
    if d.contains("refactor") || d.contains("重构") { return "refactor".into(); }
    if d.contains("fix") || d.contains("修复") || d.contains("bug") { return "bug_fix".into(); }
    if d.contains("test") || d.contains("测试") { return "testing".into(); }
    if d.contains("deploy") || d.contains("部署") { return "deployment".into(); }
    if d.contains("write") || d.contains("create") || d.contains("新建") { return "creation".into(); }
    "general".into()
}

## 8. 主动行为决策（ProactiveDecisionEngine）

### 8.1 主动行为触发条件

```rust
// crates/agent-core/src/behavior/proactive.rs

/// 主动行为类型
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum ProactiveAction {
    /// 建议创建 Skill：检测到同一任务模式重复 3+ 次
    SuggestSkillCreation {
        pattern_description: String,
        occurrence_count: u32,
    },

    /// 建议更新记忆：检测到记忆中的信息可能已过时
    SuggestMemoryUpdate {
        memory_id: String,
        reason: String,
    },

    /// 主动继续多步骤任务：上次会话中未完成的任务
    ContinueMultiStepTask {
        task_id: String,
        remaining_steps: Vec<String>,
    },

    /// 提醒定时任务结果
    ScheduledTaskResult {
        task_name: String,
        result_summary: String,
    },

    /// 发现代码中的潜在问题并主动提醒
    CodeInsight {
        file_path: String,
        insight: String,
        severity: String,
    },
}

/// 主动决策引擎
pub struct ProactiveDecisionEngine {
    behavior_repo: BehaviorRepo,
    memory_repo: MemoryRepo,
    config: ProactiveConfig,
}

pub struct ProactiveConfig {
    /// 置信度阈值：低于此值不触发主动行为（默认 0.7）
    pub confidence_threshold: f32,
    /// 同一类型主动行为的最小间隔（默认 1 小时）
    pub cooldown_per_type: Duration,
    /// 每个 session 最多触发的主动行为次数（默认 3）
    pub max_per_session: u32,
    /// 静默时段：用户设定的不打扰时间范围
    pub quiet_hours: Option<(u8, u8)>, // (start_hour, end_hour)
}

impl ProactiveDecisionEngine {
    /// 评估当前上下文是否应触发某个主动行为
    pub async fn evaluate(
        &self,
        workspace_id: &str,
        session_context: &SessionContext,
    ) -> anyhow::Result<Vec<ProactiveAction>> {
        let prefs = self.behavior_repo.get_preferences(workspace_id).await?;

        // 如果用户的主动性偏好低于 0.3，跳过所有主动行为
        if prefs.proactivity.value < 0.3 {
            return Ok(Vec::new());
        }

        // 检查静默时段
        if self.is_quiet_hours() {
            return Ok(Vec::new());
        }

        let mut actions = Vec::new();
        let session_count = session_context.proactive_action_count;

        if session_count >= self.config.max_per_session {
            return Ok(Vec::new());
        }

        // 检查各类主动行为触发条件
        self.check_skill_creation_opportunity(workspace_id, &mut actions).await?;
        self.check_stale_memories(workspace_id, &mut actions).await?;
        self.check_incomplete_tasks(workspace_id, &mut actions).await?;

        // 按置信度过滤，按用户主动性偏好限制数量
        let max_actions = if prefs.proactivity.value > 0.7 { 3 } else { 1 };
        actions.truncate(max_actions);

        Ok(actions)
    }

    /// 检查是否存在重复任务模式（可提炼为 Skill）
    async fn check_skill_creation_opportunity(
        &self,
        workspace_id: &str,
        actions: &mut Vec<ProactiveAction>,
    ) -> anyhow::Result<()> {
        let patterns = self.behavior_repo
            .find_repeated_unmatched_patterns(workspace_id, 3) // >= 3 次重复
            .await?;

        for pattern in patterns.into_iter().take(1) {
            // 检查冷却期
            if self.is_in_cooldown(workspace_id, "suggest_skill_creation").await? {
                continue;
            }

            actions.push(ProactiveAction::SuggestSkillCreation {
                pattern_description: pattern.description,
                occurrence_count: pattern.count,
            });
        }

        Ok(())
    }

    /// 检查是否有过时的记忆条目
    async fn check_stale_memories(
        &self,
        workspace_id: &str,
        actions: &mut Vec<ProactiveAction>,
    ) -> anyhow::Result<()> {
        // 查找 90 天以上未检索且 importance < 5 的记忆
        let stale = self.memory_repo
            .find_stale_entries(workspace_id, 90, 5)
            .await?;

        for entry in stale.into_iter().take(1) {
            actions.push(ProactiveAction::SuggestMemoryUpdate {
                memory_id: entry.id,
                reason: format!(
                    "记忆条目 '{}' 已 {} 天未被引用，可能需要更新或清理",
                    entry.key, entry.days_since_retrieval
                ),
            });
        }

        Ok(())
    }

    /// 检查是否有未完成的多步骤任务
    async fn check_incomplete_tasks(
        &self,
        workspace_id: &str,
        actions: &mut Vec<ProactiveAction>,
    ) -> anyhow::Result<()> {
        let incomplete = self.behavior_repo
            .find_incomplete_multi_step_tasks(workspace_id, 24 * 3600 * 1000) // 24h 内
            .await?;

        for task in incomplete.into_iter().take(1) {
            actions.push(ProactiveAction::ContinueMultiStepTask {
                task_id: task.id,
                remaining_steps: task.remaining_steps,
            });
        }

        Ok(())
    }

    fn is_quiet_hours(&self) -> bool {
        if let Some((start, end)) = self.config.quiet_hours {
            let hour = chrono::Local::now().hour() as u8;
            if start <= end {
                hour >= start && hour < end
            } else {
                hour >= start || hour < end
            }
        } else {
            false
        }
    }

    async fn is_in_cooldown(
        &self,
        workspace_id: &str,
        action_type: &str,
    ) -> anyhow::Result<bool> {
        let last = self.behavior_repo
            .get_last_proactive_action_time(workspace_id, action_type)
            .await?;
        match last {
            Some(ts) => Ok(now_ms() - ts < self.config.cooldown_per_type.as_millis() as i64),
            None => Ok(false),
        }
    }
}
```

## 9. 安全边界与护栏

### 9.1 护栏规则总览

| 风险 | 护栏措施 | 实现位置 |
|------|---------|---------|
| 偏好震荡 | 单维度每周期最大调整 0.15、每日最多 2 次调整 | `BehaviorConfig` |
| 偏好误学习 | 连续 3 次负反馈自动回滚到调整前 | `EvolutionLoop::check_regression()` |
| 用户失控感 | 冻结维度（`freeze_preference` 命令） | `UserPreferences::frozen_dimensions` |
| 隐私风险 | 反馈数据仅存本地 SQLite，不上传 | 架构约束 |
| 过度主动 | 每 session 最多 3 次主动行为、静默时段 | `ProactiveConfig` |
| Prompt 退化 | 变体晋升需 20 次试用 + 10% 优势 | `PromptOptimizer::check_promotion()` |
| 全局关闭 | `learning_enabled = false` 禁用所有行为学习 | `BehaviorConfig` |

### 9.2 行为变更的可逆性

所有行为调整记录在 `adjustment_history` 表中，每条记录包含调整前后的完整值。用户或系统可在任何时刻回滚到历史上任意一次调整的状态：

```rust
impl BehaviorRepo {
    /// 回滚到指定时间点的行为配置
    pub async fn rollback_to(
        &self,
        workspace_id: &str,
        target_timestamp: i64,
    ) -> anyhow::Result<UserPreferences> {
        // 从 adjustment_history 重建该时间点的偏好值
        let adjustments = sqlx::query_as!(
            BehaviorAdjustment,
            "SELECT * FROM adjustment_history
             WHERE workspace_id = ? AND status = 'applied' AND applied_at <= ?
             ORDER BY applied_at ASC",
            workspace_id, target_timestamp,
        )
        .fetch_all(&self.pool)
        .await?;

        // 从默认值开始，按时间顺序重放所有调整
        let mut prefs = UserPreferences::default();
        for adj in adjustments {
            match adj.dimension.as_str() {
                "response_length" => prefs.response_length.value = adj.new_value,
                "code_style" => prefs.code_style.value = adj.new_value,
                "proactivity" => prefs.proactivity.value = adj.new_value,
                "explanation_depth" => prefs.explanation_depth.value = adj.new_value,
                _ => {}
            }
        }

        self.save_preferences(workspace_id, &prefs).await?;
        Ok(prefs)
    }
}
```

// crates/agent-core/src/behavior/profile.rs

/// 行为档案：Agent 学到的关于用户的完整行为画像
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BehaviorProfile {
    /// 当前偏好值（四个维度）

    pub recent_adjustments: Vec<BehaviorAdjustment>,

/// 反馈统计摘要
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FeedbackStats {
    pub total_events: u32,
    pub positive_count: u32,
    pub negative_count: u32,
    pub implicit_count: u32,
    pub explicit_count: u32,
    /// 最常见的反馈类型 Top-3
    pub top_signal_types: Vec<(String, u32)>,
}

/// 主动行为统计
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ProactiveStats {
    pub total_suggestions: u32,
    pub accepted_count: u32,
    pub dismissed_count: u32,
    pub ignored_count: u32,
    pub acceptance_rate: f32,
}

/// Prompt 变体摘要（脱敏后的展示）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PromptVariantSummary {
    pub section_id: String,
    pub variant_count: u32,
    pub active_success_rate: f32,
    pub best_variant_success_rate: f32,
}

impl BehaviorProfile {
    /// 从各数据源聚合生成完整的行为档案
    pub async fn build(
        repo: &BehaviorRepo,
        workspace_id: &str,
    ) -> anyhow::Result<Self> {
        let preferences = repo.get_preferences(workspace_id).await?;
        let top_patterns = repo.list_top_patterns(workspace_id, 10).await?;
        let adjustments = repo.list_recent_adjustments(workspace_id, 30).await?;
        let feedback_stats = repo.compute_feedback_stats(workspace_id).await?;
        let proactive_stats = repo.compute_proactive_stats(workspace_id).await?;
        let variants = repo.list_variant_summaries(workspace_id).await?;

        Ok(Self {
            preferences,
            top_tool_patterns: top_patterns,
            recent_adjustments: adjustments,
            feedback_stats,
            proactive_stats,
            prompt_variants: variants,
        })
    }
}

    Ok(())

/// 冻结/解冻指定偏好维度
#[tauri::command]
pub async fn freeze_preference(
    state: tauri::State<'_, AppState>,
    workspace_id: String,
    dimension: String,
    frozen: bool,
) -> Result<(), String> {
    let repo = &state.behavior_repo;
    let mut prefs = repo.get_preferences(&workspace_id)
        .await.map_err(|e| e.to_string())?;

    if frozen {
        if !prefs.frozen_dimensions.contains(&dimension) {
            prefs.frozen_dimensions.push(dimension.clone());
        }
    } else {
        prefs.frozen_dimensions.retain(|d| d != &dimension);
    }

    repo.save_preferences(&workspace_id, &prefs)
        .await.map_err(|e| e.to_string())?;

    Ok(())
}
