# 工作空间文件管理详细设计

> 版本：v1.0 | 日期：2026-08-11 | 状态：草稿
> 对应需求：F-10 工作空间文件管理

工作空间文件管理为用户提供工作区内的文件浏览、预览、操作能力，并管理 Agent 生成的文档产物（ai-docs）。每个工作区拥有独立的文件目录（物理隔离），用户可通过文件浏览器面板管理文件，Agent 可通过 `exec_command`（文件读取/目录浏览）和 `apply_patch`（Freeform Lark 语法补丁，文件增删改）工具在沙箱范围内操作文件。AI 生成的代码、报告、文章等产物通过手动一键保存归档到 `ai-docs/` 目录，按类型自动分类并记录来源会话元数据。

本文档在 PRD F-10 需求基础上，给出完整的文件系统布局、文件浏览器组件设计、AI 文档管理机制、文件预览系统、Agent 文件操作集成、文件搜索、导入导出、存储配额、Tauri Commands 的实现级设计。

## 1. 文件管理架构

### 1.1 工作区文件系统布局

每个工作区在 `~/.astro/workspaces/{workspace_id}/` 下拥有独立的文件目录树，工作区之间完全物理隔离：

```text
~/.astro/workspaces/
  ├── {workspace_id_1}/
  │     ├── files/                # 用户自行管理的文件
  │     │     ├── notes/
  │     │     ├── images/
  │     │     └── ...
  │     ├── ai-docs/              # AI 产物归档根目录
  │     │     ├── code/           # 代码文件（.py .rs .ts .sh 等）
  │     │     ├── reports/        # 报告、分析、总结（.md .pdf 等）
  │     │     ├── articles/       # 文章、文案、文档（.md .txt 等）
  │     │     ├── data/           # 数据文件（.json .csv .yaml 等）
  │     │     └── other/          # 无法自动归类的产物
  │     └── knowledge/            # 知识库原始文档（由 RAG 模块管理）
  │
  ├── {workspace_id_2}/
  │     ├── files/
  │     ├── ai-docs/
  │     └── knowledge/
  └── ...
```

### 1.2 WorkspaceFileSystem 核心结构

```rust
// crates/agent-core/src/files/workspace_fs.rs

use std::path::{Path, PathBuf};

/// 工作区文件系统管理器
pub struct WorkspaceFileSystem {
    /// 工作区根目录：~/.astro/workspaces/{workspace_id}/
    root: PathBuf,
    /// 工作区 ID
    workspace_id: String,
}

/// 工作区内的标准子目录
#[derive(Debug, Clone, Copy, PartialEq)]
pub enum WorkspaceDir {
    /// 用户文件目录
    Files,
    /// AI 产物归档目录
    AiDocs,
    /// AI 产物子目录
    AiDocsCode,
    AiDocsReports,
    AiDocsArticles,
    AiDocsData,
    AiDocsOther,
    /// 知识库文档目录
    Knowledge,
}

impl WorkspaceDir {
    pub fn relative_path(&self) -> &'static str {
        match self {
            Self::Files => "files",
            Self::AiDocs => "ai-docs",
            Self::AiDocsCode => "ai-docs/code",
            Self::AiDocsReports => "ai-docs/reports",
            Self::AiDocsArticles => "ai-docs/articles",
            Self::AiDocsData => "ai-docs/data",
            Self::AiDocsOther => "ai-docs/other",
            Self::Knowledge => "knowledge",
        }
    }
}

impl WorkspaceFileSystem {
    pub fn new(base_dir: &Path, workspace_id: &str) -> Self {
        Self {
            root: base_dir.join("workspaces").join(workspace_id),
            workspace_id: workspace_id.to_string(),
        }
    }

    /// 初始化工作区目录结构（创建工作区时调用）
    pub async fn initialize(&self) -> anyhow::Result<()> {
        let dirs = [
            WorkspaceDir::Files,
            WorkspaceDir::AiDocsCode,
            WorkspaceDir::AiDocsReports,
            WorkspaceDir::AiDocsArticles,
            WorkspaceDir::AiDocsData,
            WorkspaceDir::AiDocsOther,
            WorkspaceDir::Knowledge,
        ];
        for dir in &dirs {
            tokio::fs::create_dir_all(self.resolve_dir(*dir)).await?;
        }
        Ok(())
    }

    /// 获取工作区根目录
    pub fn root(&self) -> &Path {
        &self.root
    }

    /// 解析标准子目录的绝对路径
    pub fn resolve_dir(&self, dir: WorkspaceDir) -> PathBuf {
        self.root.join(dir.relative_path())
    }

    /// 将用户传入的相对路径解析为绝对路径，并验证不逃逸工作区
    pub fn resolve_path(&self, relative: &str) -> anyhow::Result<PathBuf> {
        let resolved = self.root.join(relative).canonicalize()
            .unwrap_or_else(|_| self.root.join(relative));

        // 工作区沙箱校验：绝对路径必须以 root 为前缀
        if !resolved.starts_with(&self.root) {
            anyhow::bail!(
                "路径逃逸工作区沙箱：{} 不在 {} 范围内",
                resolved.display(),
                self.root.display()
            );
        }
        Ok(resolved)
    }

    /// 将绝对路径转换为工作区内的相对路径
    pub fn to_relative(&self, absolute: &Path) -> anyhow::Result<String> {
        absolute
            .strip_prefix(&self.root)
            .map(|p| p.to_string_lossy().into_owned())
            .map_err(|_| anyhow::anyhow!("路径不在工作区范围内"))
    }

    /// 删除工作区全部文件（删除工作区时调用）
    pub async fn destroy(&self) -> anyhow::Result<()> {
        if self.root.exists() {
            tokio::fs::remove_dir_all(&self.root).await?;
        }
        Ok(())
    }
}
```

```rust
    Markdown,
    Code,

    Audio,
    Video,
    Data,
    Text,
    Archive,
    Unknown,

            "md" | "markdown" | "mdx" => Self::Markdown,
            "rs" | "py" | "ts" | "tsx" | "js" | "jsx" | "go" | "java"

            | "toml" | "yaml" | "yml" | "html" | "css" | "scss" | "sql"

            "png" | "jpg" | "jpeg" | "gif" | "webp" | "svg" | "bmp" | "ico" => Self::Image,
            "pdf" => Self::Pdf,
            "mp3" | "wav" | "ogg" | "flac" | "aac" | "m4a" => Self::Audio,
            "mp4" | "mov" | "avi" | "webm" | "mkv" => Self::Video,
            "json" | "csv" | "tsv" | "xml" | "parquet" => Self::Data,
            "txt" | "text" | "log" | "ini" | "cfg" | "conf" => Self::Text,
            "zip" | "tar" | "gz" | "bz2" | "7z" | "rar" => Self::Archive,

    }
```

### 2.2 文件列表读取

```rust
// crates/agent-core/src/files/reader.rs

use std::path::Path;
use tokio::fs;

/// 读取目录下的文件列表（单层，不递归）
pub async fn list_directory(
    wfs: &WorkspaceFileSystem,
    relative_dir: &str,
) -> anyhow::Result<Vec<FileEntry>> {
    let abs_dir = wfs.resolve_path(relative_dir)?;

    if !abs_dir.is_dir() {
        anyhow::bail!("路径不是目录: {}", relative_dir);
    }

    let mut entries = Vec::new();
    let mut read_dir = fs::read_dir(&abs_dir).await?;

    while let Some(entry) = read_dir.next_entry().await? {
        let metadata = entry.metadata().await?;
        let name = entry.file_name().to_string_lossy().into_owned();

        // 跳过隐藏文件（以 . 开头）
        if name.starts_with('.') {
            continue;
        }

        let is_dir = metadata.is_dir();
        let extension = if is_dir {
            None
        } else {
            Path::new(&name)
                .extension()
                .and_then(|e| e.to_str())
                .map(|e| e.to_lowercase())
        };

        let file_type = if is_dir {
            FileType::Directory
        } else {
            extension
                .as_deref()
                .map(FileType::from_extension)
                .unwrap_or(FileType::Unknown)
        };

        let modified_at = metadata
            .modified()
            .ok()
            .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
            .map(|d| d.as_millis() as i64);

        let size = if is_dir { None } else { Some(metadata.len()) };

        // 目录的子条目计数（快速统计，不递归）
        let children_count = if is_dir {
            count_dir_entries(&entry.path()).await.ok()
        } else {
            None
        };

        let relative_path = wfs.to_relative(&entry.path())?;

        entries.push(FileEntry {
            path: relative_path,
            name,
            is_dir,
            size,
            modified_at,
            extension,
            file_type,
            children_count,
        });
    }

    // 排序：目录在前，同级按名称字母排序
    entries.sort_by(|a, b| {
        b.is_dir
            .cmp(&a.is_dir)
            .then_with(|| a.name.to_lowercase().cmp(&b.name.to_lowercase()))
    });

    Ok(entries)
}

/// 快速统计目录下的条目数量（不含隐藏文件）
async fn count_dir_entries(dir: &Path) -> anyhow::Result<usize> {
    let mut count = 0;
    let mut read_dir = fs::read_dir(dir).await?;
    while let Some(entry) = read_dir.next_entry().await? {
        let name = entry.file_name();
        if !name.to_string_lossy().starts_with('.') {
            count += 1;
        }
    }
    Ok(count)
}
```

### 2.4 FileBrowser Zustand Store

```typescript
// apps/desktop/src/store/fileBrowser.ts

import { create } from "zustand";
import { invoke } from "@tauri-apps/api/core";
import { listen } from "@tauri-apps/api/event";
import { useWorkspaceStore } from "./workspace";

interface FileBrowserStore {
  entries: FileEntry[];
  currentDir: string;
  selectedPath: string | null;
  viewMode: "tree" | "list" | "grid";
  expandedDirs: Set<string>;
  loading: boolean;

  // Actions
  loadDirectory: (relativePath: string) => Promise<void>;
  toggleExpand: (dirPath: string) => Promise<void>;
  selectFile: (path: string | null) => void;
  setViewMode: (mode: "tree" | "list" | "grid") => void;
  refresh: () => Promise<void>;
  createFile: (name: string, parentDir: string) => Promise<void>;
  createDirectory: (name: string, parentDir: string) => Promise<void>;
  renameEntry: (oldPath: string, newName: string) => Promise<void>;
  deleteEntry: (path: string) => Promise<void>;
  addToKnowledge: (filePath: string) => Promise<void>;

  initListeners: () => () => void;
}

export const useFileBrowserStore = create<FileBrowserStore>((set, get) => ({
  entries: [],
  currentDir: "",
  selectedPath: null,
  viewMode: "tree",
  expandedDirs: new Set<string>(),
  loading: false,

  loadDirectory: async (relativePath: string) => {
    set({ loading: true });
    try {
      const entries = await invoke<FileEntry[]>("list_files", {
        relativePath,
      });
      set({ entries, currentDir: relativePath, loading: false });
    } catch (e) {
      console.error("加载目录失败:", e);
      set({ loading: false });
    }
  },

  toggleExpand: async (dirPath: string) => {
    const { expandedDirs } = get();
    const next = new Set(expandedDirs);
    if (next.has(dirPath)) {
      next.delete(dirPath);
    } else {
      next.add(dirPath);
    }
    set({ expandedDirs: next });
  },

  selectFile: (path) => set({ selectedPath: path }),

  setViewMode: (mode) => set({ viewMode: mode }),

  refresh: async () => {
    const { currentDir, loadDirectory } = get();
    await loadDirectory(currentDir);
  },

  createFile: async (name, parentDir) => {
    await invoke("write_file", {
      relativePath: `${parentDir}/${name}`,
      content: "",
    });
    await get().refresh();
  },

  createDirectory: async (name, parentDir) => {
    await invoke("create_directory", {
      relativePath: `${parentDir}/${name}`,
    });
    await get().refresh();
  },

  renameEntry: async (oldPath, newName) => {
    await invoke("rename_file", { oldPath, newName });
    await get().refresh();
  },

  deleteEntry: async (path) => {
    await invoke("delete_file", { relativePath: path });
    await get().refresh();
  },

  addToKnowledge: async (filePath: string) => {
    await invoke("add_file_to_knowledge", { filePath });
  },

  initListeners: () => {
    const unlisteners: Array<() => void> = [];

    // 监听工作区切换事件 => 重新加载根目录
    listen("workspace_switched", () => {
      get().loadDirectory("");
    }).then((u) => unlisteners.push(u));

    // 监听文件变更事件 => 刷新当前目录
    listen("file_changed", () => {
      get().refresh();
    }).then((u) => unlisteners.push(u));

    return () => unlisteners.forEach((u) => u());
  },
}));
```

```rust
use serde::{Deserialize, Serialize};

/// AI 产物类型
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
#[serde(rename_all = "lowercase")]
pub enum ArtifactType {
    Code,
    Report,
    Article,
    Data,
    Other,
}

impl ArtifactType {
    /// 对应的 ai-docs 子目录名
    pub fn subdir(&self) -> &'static str {
        match self {
            Self::Code => "code",
            Self::Report => "reports",
            Self::Article => "articles",
            Self::Data => "data",
            Self::Other => "other",
        }
    }

    /// 根据文件扩展名推断产物类型
    pub fn from_extension(ext: &str) -> Self {
        match ext.to_lowercase().as_str() {
            "rs" | "py" | "ts" | "tsx" | "js" | "jsx" | "go" | "java"
            | "c" | "cpp" | "h" | "hpp" | "rb" | "sh" | "bash" | "zsh"
            | "swift" | "kt" | "dart" | "lua" | "r" | "zig"
            | "html" | "css" | "scss" | "sql" => Self::Code,
            "json" | "csv" | "tsv" | "xml" | "yaml" | "yml"
            | "parquet" | "toml" => Self::Data,
            _ => Self::Other,
        }
    }

    /// 根据文件内容特征进一步推断（对 .md / .txt 做细分）
    pub fn from_content_heuristic(content: &str, ext: &str) -> Self {
        match ext {
            "md" | "txt" => {
                // 报告特征：包含"分析"、"总结"、"报告"、"结论"等关键词
                let report_keywords = ["分析", "总结", "报告", "结论", "Summary",
                                       "Analysis", "Report", "Conclusion", "## 结果"];
                let has_report_signal = report_keywords.iter()
                    .any(|kw| content.contains(kw));

                if has_report_signal {
                    Self::Report
                } else {
                    Self::Article
                }
            }
            _ => Self::from_extension(ext),
        }
    }
}

/// AI 产物记录
#[derive(Debug, Clone, Serialize, Deserialize)]

pub struct AiArtifact {

    pub file_path: String,

/// 保存 AI 产物的输入参数
#[derive(Debug, Deserialize)]
#[serde(rename_all = "camelCase")]
pub struct SaveArtifactInput {
    /// AI 生成的内容
    pub content: String,
    /// 文件扩展名（如 "rs"、"md"）
    pub extension: String,
    /// AI 推断的标题（用户可在确认弹窗中修改）
    pub title: String,
    /// 来源对话 ID
    pub conversation_id: Option<String>,
    /// 来源消息 ID
    pub message_id: Option<String>,
    /// 用户指定的目标子目录（可选，不填则自动分类）
    pub target_subdir: Option<String>,
    /// 用户指定的文件名（可选，不填则自动生成）
    pub custom_filename: Option<String>,
}

/// ArtifactManager：管理 AI 产物的保存、列表、搜索
pub struct ArtifactManager {
    pool: SqlitePool,
    wfs: Arc<WorkspaceFileSystem>,
}

impl ArtifactManager {
    pub fn new(pool: SqlitePool, wfs: Arc<WorkspaceFileSystem>) -> Self {
        Self { pool, wfs }
    }

    /// 保存 AI 产物到 ai-docs/ 目录并记录元数据
    pub async fn save_artifact(
        &self,
        workspace_id: &str,
        input: SaveArtifactInput,
    ) -> anyhow::Result<AiArtifact> {
        // 1. 推断产物类型
        let artifact_type = if let Some(ref subdir) = input.target_subdir {
            match subdir.as_str() {
                "code" => ArtifactType::Code,
                "reports" => ArtifactType::Report,
                "articles" => ArtifactType::Article,
                "data" => ArtifactType::Data,
                _ => ArtifactType::Other,
            }
        } else {
            ArtifactType::from_content_heuristic(&input.content, &input.extension)
        };

        // 2. 生成文件名
        let file_name = if let Some(custom) = &input.custom_filename {
            custom.clone()
        } else {
            let date = chrono::Local::now().format("%Y-%m-%d");
            let sanitized_title = sanitize_filename(&input.title);
            format!("{}_{}.{}", date, sanitized_title, input.extension)
        };

        // 3. 构建目标路径
        let relative_path = format!("ai-docs/{}/{}", artifact_type.subdir(), file_name);
        let abs_path = self.wfs.resolve_path(&relative_path)?;

        // 4. 确保目标目录存在
        if let Some(parent) = abs_path.parent() {
            tokio::fs::create_dir_all(parent).await?;
        }

        // 5. 处理同名文件冲突（追加序号）
        let (final_path, final_relative) = resolve_name_conflict(&abs_path, &relative_path).await;

        // 6. 写入文件
        tokio::fs::write(&final_path, &input.content).await?;
        let file_size = input.content.len() as i64;

        // 7. 推断编程语言（仅 code 类型）
        let language = if artifact_type == ArtifactType::Code {
            detect_language(&input.extension)
        } else {
            None
        };

        // 8. 写入数据库记录
        let id = uuid::Uuid::new_v4().to_string();
        let now = now_ms();
        let final_name = final_path.file_name()
            .unwrap_or_default()
            .to_string_lossy()
            .into_owned();

        sqlx::query!(
            "INSERT INTO ai_artifacts
             (id, workspace_id, file_path, file_name, artifact_type, file_size,
              conversation_id, message_id, title, language, created_at, updated_at)
             VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
            id, workspace_id, final_relative, final_name,
            artifact_type.subdir(), file_size,
            input.conversation_id, input.message_id,
            input.title, language, now, now
        )
        .execute(&self.pool)
        .await?;

        Ok(AiArtifact {
            id,
            workspace_id: workspace_id.to_string(),
            file_path: final_relative,
            file_name: final_name,
            artifact_type,
            file_size,
            conversation_id: input.conversation_id,
            message_id: input.message_id,
            title: input.title,
            description: None,
            language,
            tags: vec![],
            created_at: now,
            updated_at: now,
        })
    }

    /// 列出工作区的 AI 产物（支持按类型过滤）
    pub async fn list_artifacts(
        &self,
        workspace_id: &str,
        type_filter: Option<&str>,
        limit: i64,
        offset: i64,
    ) -> anyhow::Result<Vec<AiArtifact>> {
        let rows = if let Some(t) = type_filter {
            sqlx::query!(
                "SELECT * FROM ai_artifacts
                 WHERE workspace_id = ? AND artifact_type = ?
                 ORDER BY created_at DESC LIMIT ? OFFSET ?",
                workspace_id, t, limit, offset

            .fetch_all(&self.pool)
            .await?
        } else {
            sqlx::query!(
                "SELECT * FROM ai_artifacts
                 WHERE workspace_id = ?
                 ORDER BY created_at DESC LIMIT ? OFFSET ?",
                workspace_id, limit, offset
            )
            .fetch_all(&self.pool)
            .await?
        };

        Ok(rows.into_iter().map(|r| AiArtifact {
            id: r.id,
            workspace_id: r.workspace_id,
            file_path: r.file_path,
            file_name: r.file_name,
            artifact_type: match r.artifact_type.as_str() {
                "code" => ArtifactType::Code,
                "reports" => ArtifactType::Report,
                "articles" => ArtifactType::Article,
                "data" => ArtifactType::Data,
                _ => ArtifactType::Other,
            },
            file_size: r.file_size,
            conversation_id: r.conversation_id,
            message_id: r.message_id,
            title: r.title,
            description: r.description,
            language: r.language,
            tags: serde_json::from_str(r.tags.as_deref().unwrap_or("[]")).unwrap_or_default(),
            created_at: r.created_at,
            updated_at: r.updated_at,
        }).collect())
    }

    /// 搜索 AI 产物（FTS5 全文检索标题和描述）
    pub async fn search_artifacts(
        &self,
        workspace_id: &str,
        query: &str,
        limit: i64,
    ) -> anyhow::Result<Vec<AiArtifact>> {
        let rows = sqlx::query!(
            "SELECT a.* FROM ai_artifacts_fts f
             JOIN ai_artifacts a ON a.rowid = f.rowid
             WHERE ai_artifacts_fts MATCH ?
               AND a.workspace_id = ?
             ORDER BY bm25(ai_artifacts_fts)
             LIMIT ?",
            query, workspace_id, limit
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| AiArtifact {
            id: r.id,
            workspace_id: r.workspace_id,
            file_path: r.file_path,
            file_name: r.file_name,
            artifact_type: match r.artifact_type.as_str() {
                "code" => ArtifactType::Code,
                "reports" => ArtifactType::Report,
                "articles" => ArtifactType::Article,
                "data" => ArtifactType::Data,
                _ => ArtifactType::Other,
            },
            file_size: r.file_size,
            conversation_id: r.conversation_id,
            message_id: r.message_id,
            title: r.title,
            description: r.description,
            language: r.language,
            tags: serde_json::from_str(r.tags.as_deref().unwrap_or("[]")).unwrap_or_default(),
            created_at: r.created_at,
            updated_at: r.updated_at,
        }).collect())
    }

    /// 删除 AI 产物（同时删除文件和数据库记录）
    pub async fn delete_artifact(
        &self,
        workspace_id: &str,
        artifact_id: &str,
    ) -> anyhow::Result<()> {
        // 查找记录
        let row = sqlx::query!(
            "SELECT file_path FROM ai_artifacts WHERE id = ? AND workspace_id = ?",
            artifact_id, workspace_id
        )
        .fetch_optional(&self.pool)
        .await?
        .ok_or_else(|| anyhow::anyhow!("AI 产物不存在: {}", artifact_id))?;

        // 删除文件
        let abs_path = self.wfs.resolve_path(&row.file_path)?;
        if abs_path.exists() {
            tokio::fs::remove_file(&abs_path).await?;
        }

        // 删除数据库记录（FTS 触发器自动同步）
        sqlx::query!("DELETE FROM ai_artifacts WHERE id = ?", artifact_id)
            .execute(&self.pool)
            .await?;

        Ok(())
    }
}

/// 将标题清理为安全的文件名部分
fn sanitize_filename(title: &str) -> String {
    let sanitized: String = title.chars()
        .map(|c| {
            if c.is_alphanumeric() || c == '-' || c == '_' || c == '.' || c > '\u{007F}' {
                c
            } else {
                '_'
            }
        })
        .collect();
    // 截断到合理长度
    let truncated: String = sanitized.chars().take(60).collect();
    if truncated.is_empty() { "untitled".to_string() } else { truncated }
}

/// 处理同名文件冲突：追加 (1)、(2)... 序号
async fn resolve_name_conflict(path: &Path, relative: &str) -> (PathBuf, String) {
    if !path.exists() {
        return (path.to_path_buf(), relative.to_string());
    }

    let stem = path.file_stem().unwrap_or_default().to_string_lossy();
    let ext = path.extension().unwrap_or_default().to_string_lossy();
    let parent = path.parent().unwrap();

    for i in 1..=99 {
        let new_name = if ext.is_empty() {
            format!("{}_({})", stem, i)
        } else {
            format!("{}_({}).{}", stem, i, ext)
        };
        let new_path = parent.join(&new_name);
        if !new_path.exists() {
            let new_relative = relative.rsplit_once('/')
                .map(|(dir, _)| format!("{}/{}", dir, new_name))
                .unwrap_or(new_name.clone());
            return (new_path, new_relative);
        }
    }

    // 极端情况回退：追加时间戳
    let ts = chrono::Utc::now().timestamp_millis();
    let new_name = format!("{}_{}.{}", stem, ts, ext);
    let new_path = parent.join(&new_name);
    let new_relative = relative.rsplit_once('/')
        .map(|(dir, _)| format!("{}/{}", dir, new_name))
        .unwrap_or(new_name);
    (new_path, new_relative)
}

/// 推断编程语言名称
fn detect_language(ext: &str) -> Option<String> {
    match ext.to_lowercase().as_str() {
        "rs" => Some("rust"),
        "py" => Some("python"),
        "ts" | "tsx" => Some("typescript"),
        "js" | "jsx" => Some("javascript"),
        "go" => Some("go"),
        "java" => Some("java"),
        "c" | "h" => Some("c"),
        "cpp" | "hpp" => Some("cpp"),
        "rb" => Some("ruby"),
        "sh" | "bash" | "zsh" => Some("shell"),
        "swift" => Some("swift"),
        "kt" => Some("kotlin"),
        "sql" => Some("sql"),
        "html" => Some("html"),
        "css" | "scss" => Some("css"),
        _ => None,
    }
    .map(String::from)
}
```

```rust
// crates/agent-core/src/files/preview.rs

use serde::{Deserialize, Serialize};

/// 预览内容类型
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type", rename_all = "lowercase")]
pub enum PreviewContent {
    /// Markdown 渲染预览（返回原始 Markdown 文本，前端渲染）
    Markdown { content: String },
    /// 代码语法高亮预览（返回原始代码 + 语言标识）
    Code { content: String, language: String },
    /// 纯文本预览
    Text { content: String },
    /// 图片预览（返回 base64 编码或本地文件 URI）
    Image { uri: String, width: Option<u32>, height: Option<u32> },
    /// PDF 预览（返回首页的渲染图片 base64）
    Pdf { first_page_image: String, page_count: u32 },
    /// 音频预览（返回文件 URI 和时长）
    Audio { uri: String, duration_secs: Option<f64> },
    /// 视频预览（返回文件 URI 和时长）
    Video { uri: String, duration_secs: Option<f64> },
    /// 不支持预览的文件类型
    Unsupported { message: String },
}

/// 预览选项
#[derive(Debug, Deserialize)]
pub struct PreviewOptions {
    /// 文本类文件的最大读取字节数（默认 512KB）
    pub max_text_bytes: Option<usize>,
    /// 图片最大尺寸（宽，超过则缩放）
    pub max_image_width: Option<u32>,
}

impl Default for PreviewOptions {
    fn default() -> Self {
        Self {
            max_text_bytes: Some(512 * 1024),  // 512KB
            max_image_width: Some(1200),
        }
    }
}

/// 生成文件预览内容
pub async fn generate_preview(
    wfs: &WorkspaceFileSystem,
    relative_path: &str,
    options: &PreviewOptions,
) -> anyhow::Result<PreviewContent> {
    let abs_path = wfs.resolve_path(relative_path)?;

    if !abs_path.exists() {
        anyhow::bail!("文件不存在: {}", relative_path);
    }

    let extension = abs_path
        .extension()
        .and_then(|e| e.to_str())
        .unwrap_or("")
        .to_lowercase();

    let file_type = FileType::from_extension(&extension);
    let max_bytes = options.max_text_bytes.unwrap_or(512 * 1024);

    match file_type {
        FileType::Markdown => {
            let content = read_text_capped(&abs_path, max_bytes).await?;
            Ok(PreviewContent::Markdown { content })
        }

        FileType::Code => {
            let content = read_text_capped(&abs_path, max_bytes).await?;
            let language = detect_language(&extension)
                .unwrap_or_else(|| extension.clone());
            Ok(PreviewContent::Code { content, language })
        }

        FileType::Text | FileType::Data => {
            let content = read_text_capped(&abs_path, max_bytes).await?;
            Ok(PreviewContent::Text { content })
        }

        FileType::Image => {
            // 小图片直接 base64 内联，大图片使用 Tauri asset URI
            let metadata = tokio::fs::metadata(&abs_path).await?;
            let uri = if metadata.len() < 2 * 1024 * 1024 {
                // < 2MB，base64 内联
                let bytes = tokio::fs::read(&abs_path).await?;
                let mime = match extension.as_str() {
                    "png" => "image/png",
                    "jpg" | "jpeg" => "image/jpeg",
                    "gif" => "image/gif",
                    "webp" => "image/webp",
                    "svg" => "image/svg+xml",
                    _ => "application/octet-stream",
                };
                format!("data:{};base64,{}", mime, base64::encode(&bytes))
            } else {
                // 大文件使用 asset 协议
                format!("asset://localhost/{}", abs_path.display())
            };

            Ok(PreviewContent::Image { uri, width: None, height: None })
        }

        FileType::Pdf => {
            // PDF 首页预览：通过 pdf-render 库渲染首页为 PNG
            let (first_page_image, page_count) =
                render_pdf_first_page(&abs_path).await?;
            Ok(PreviewContent::Pdf { first_page_image, page_count })
        }

        FileType::Audio => {
            let uri = format!("asset://localhost/{}", abs_path.display());
            Ok(PreviewContent::Audio { uri, duration_secs: None })
        }

        FileType::Video => {
            let uri = format!("asset://localhost/{}", abs_path.display());
            Ok(PreviewContent::Video { uri, duration_secs: None })
        }

        _ => Ok(PreviewContent::Unsupported {
            message: format!("不支持预览 .{} 格式的文件", extension),
        }),
    }
}

/// 读取文本文件，超出最大字节数时截断并附加提示
async fn read_text_capped(path: &Path, max_bytes: usize) -> anyhow::Result<String> {
    let metadata = tokio::fs::metadata(path).await?;
    let file_size = metadata.len() as usize;

    if file_size <= max_bytes {
        Ok(tokio::fs::read_to_string(path).await?)
    } else {
        let bytes = tokio::fs::read(path).await?;
        let truncated = &bytes[..max_bytes];
        // 在 UTF-8 边界处截断
        let content = String::from_utf8_lossy(truncated).into_owned();
        Ok(format!(
            "{}\n\n--- 文件过大，仅显示前 {} ---",
            content,
            format_size(max_bytes)
        ))
    }
}

fn format_size(bytes: usize) -> String {
    if bytes < 1024 { return format!("{} B", bytes); }
    if bytes < 1024 * 1024 { return format!("{:.1} KB", bytes as f64 / 1024.0); }
    format!("{:.1} MB", bytes as f64 / (1024.0 * 1024.0))
}

/// 渲染 PDF 首页为 base64 PNG（阻塞任务）
async fn render_pdf_first_page(path: &Path) -> anyhow::Result<(String, u32)> {
    let path_owned = path.to_path_buf();
    tokio::task::spawn_blocking(move || {
        let doc = pdf_extract::extract_text(&path_owned)?;
        let page_count = doc.lines().count().max(1) as u32; // 简化：真实实现用 pdf crate
        // 实际实现应使用 pdfium-render 或 mupdf 渲染首页
        Ok((String::new(), page_count))
    })
    .await?
}
```

## 5. Agent 文件操作集成

### 5.1 工具与工作区文件系统的映射

Agent 的 `exec_command`（文件读取/搜索/列目录）和 `apply_patch`（Freeform Lark 语法补丁，替代原 file_ops）工具在执行时被限定在当前工作区的文件沙箱内。原有的 `file_read`/`file_write`/`file_edit` 工具已移除，其功能分别归入 `exec_command` 和 `apply_patch`。路径解析和权限校验在工具执行层统一处理：

```rust
// crates/agent-core/src/tools/ (路径解析与沙箱校验)

/// 文件操作工具上下文：注入工作区文件系统
pub struct FileToolContext {
    pub wfs: Arc<WorkspaceFileSystem>,
    pub workspace_id: String,
}

impl FileToolContext {
    /// 验证路径并解析为绝对路径（核心安全逻辑）
    ///
    /// 规则：
    /// 1. 路径必须在工作区根目录内（沙箱校验）
    /// 2. 禁止访问 knowledge/ 目录（由 RAG 模块管理）
    /// 3. file_write 写入代码/文档时，自动路由到 ai-docs/ 还是 files/ 取决于调用者指定
    pub fn validate_and_resolve(&self, path: &str) -> anyhow::Result<PathBuf> {
        // 沙箱校验
        let resolved = self.wfs.resolve_path(path)?;

        // 禁止直接操作 knowledge/ 目录
        let relative = self.wfs.to_relative(&resolved)?;
        if relative.starts_with("knowledge/") || relative == "knowledge" {
            anyhow::bail!(
                "knowledge/ 目录由知识库模块管理，请使用 knowledge_manage 工具操作"
            );
        }

        Ok(resolved)
    }
}
```

```rust
                .and_then(|e| e.to_str())
                .unwrap_or("txt");

        if let Some(parent) = abs_path.parent() {
            tokio::fs::create_dir_all(parent).await
                .map_err(|e| ToolError::ExecutionFailed(e.to_string()))?;
        }

        // 存储配额检查
        let content_size = params.content.len() as u64;
        check_storage_quota(context, content_size).await?;

        // 写入文件
        tokio::fs::write(&abs_path, &params.content).await
            .map_err(|e| ToolError::ExecutionFailed(e.to_string()))?;

        // 通知前端刷新文件列表
        if let Some(app) = &context.app_handle {
            app.emit("file_changed", serde_json::json!({
                "action": "created",
                "path": target_path,
            })).ok();
        }

        Ok(ToolOutput::success(format!("文件已写入: {}", target_path)))
    }

            format_size(usage.quota_bytes as usize),
            format_size(additional_bytes as usize),

    }
    Ok(())
}
```

### 6.1 与全局搜索的集成

文件搜索作为全局搜索（F-17）的 `FileSearch` 源之一，同时在文件浏览器面板内提供独立的快速过滤功能。

```rust
// crates/agent-core/src/files/search.rs

use grep::regex::RegexMatcher;
use grep::searcher::{Searcher, Sink, SinkMatch};
use ignore::WalkBuilder;
use std::time::{Duration, Instant};

/// 文件搜索引擎：基于 ripgrep 核心库实现
pub struct FileSearchEngine;

impl FileSearchEngine {
    /// 搜索工作区内的文件内容
    ///
    /// 策略：
    /// - 文件名匹配：glob 模式
    /// - 文件内容匹配：ripgrep (grep crate)
    /// - 超时保护：2 秒硬性超时
    /// - 跳过二进制文件和 .gitignore 规则
    pub async fn search(
        wfs: &WorkspaceFileSystem,
        query: &str,
        limit: usize,
    ) -> anyhow::Result<Vec<FileSearchResult>> {
        let root = wfs.root().to_path_buf();
        let query_owned = query.to_string();
        let search_limit = limit;

        // 文件搜索是 CPU 密集型，在阻塞线程中执行
        tokio::task::spawn_blocking(move || {
            let start = Instant::now();
            let timeout = Duration::from_secs(2);
            let mut results = Vec::new();

            // 构建正则匹配器（大小写不敏感）
            let matcher = RegexMatcher::new_line_matcher(&format!("(?i){}", regex::escape(&query_owned)))
                .map_err(|e| anyhow::anyhow!("正则构造失败: {}", e))?;

            // 遍历目录（跳过 .gitignore 中的文件和二进制文件）
            let walker = WalkBuilder::new(&root)
                .hidden(true)                    // 跳过隐藏文件
                .git_ignore(true)                // 遵循 .gitignore
                .max_depth(Some(10))             // 最大递归深度
                .build();

            for entry_result in walker {
                // 超时检查
                if start.elapsed() > timeout {
                    break;
                }
                if results.len() >= search_limit {
                    break;
                }

                let entry = match entry_result {
                    Ok(e) => e,
                    Err(_) => continue,
                };

                let path = entry.path();
                if !path.is_file() {
                    continue;
                }

                // 文件名匹配
                let file_name = path.file_name()
                    .unwrap_or_default()
                    .to_string_lossy();
                let name_match = file_name.to_lowercase()
                    .contains(&query_owned.to_lowercase());

                // 内容匹配
                let mut content_matches = Vec::new();
                let mut searcher = Searcher::new();
                let _ = searcher.search_path(
                    &matcher,
                    path,
                    MatchCollector::new(&mut content_matches, 3), // 每文件最多 3 个匹配行
                );

                if name_match || !content_matches.is_empty() {
                    let relative_path = path.strip_prefix(&root)
                        .map(|p| p.to_string_lossy().into_owned())
                        .unwrap_or_default();

                    results.push(FileSearchResult {
                        path: relative_path,
                        file_name: file_name.into_owned(),
                        name_match,
                        content_snippets: content_matches,
                    });
                }
            }

            Ok(results)
        })
        .await?
    }
}

/// 文件搜索结果
#[derive(Debug, Clone, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct FileSearchResult {
    /// 相对路径
    pub path: String,
    /// 文件名
    pub file_name: String,
    /// 文件名是否匹配
    pub name_match: bool,
    /// 内容匹配片段（行号 + 内容）
    pub content_snippets: Vec<ContentSnippet>,
}

#[derive(Debug, Clone, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct ContentSnippet {
    pub line_number: u64,
    pub content: String,
}

/// grep Sink 实现：收集匹配行
struct MatchCollector<'a> {
    matches: &'a mut Vec<ContentSnippet>,
    max_matches: usize,
}

impl<'a> MatchCollector<'a> {
    fn new(matches: &'a mut Vec<ContentSnippet>, max: usize) -> Self {
        Self { matches, max_matches: max }
    }
}

impl<'a> Sink for MatchCollector<'a> {
    type Error = std::io::Error;

    fn matched(
        &mut self,
        _searcher: &Searcher,
        mat: &SinkMatch<'_>,
    ) -> Result<bool, Self::Error> {
        if self.matches.len() >= self.max_matches {
            return Ok(false); // 停止搜索
        }
        let content = String::from_utf8_lossy(mat.bytes()).trim().to_string();
        // 截断过长的行
        let truncated = if content.len() > 200 {
            format!("{}...", &content[..200])
        } else {
            content
        };
        self.matches.push(ContentSnippet {
            line_number: mat.line_number().unwrap_or(0),
            content: truncated,
        });
        Ok(true)
    }
}
```

```rust
// crates/agent-core/src/files/import_export.rs

/// 导入外部文件/文件夹到工作区
pub async fn import_files(
    wfs: &WorkspaceFileSystem,
    source_paths: &[PathBuf],
    target_dir: &str,       // 目标目录（相对工作区根）
    workspace_id: &str,
    quota_bytes: u64,
) -> anyhow::Result<ImportResult> {
    let target_abs = wfs.resolve_path(target_dir)?;
    tokio::fs::create_dir_all(&target_abs).await?;

    let mut imported_count = 0u64;
    let mut imported_bytes = 0u64;
    let mut skipped = Vec::new();

    // 预检查：计算总大小
    let mut total_size = 0u64;
    for path in source_paths {
        total_size += calculate_path_size(path).await?;
    }

    // 配额检查
    let current_usage = calculate_storage_usage(wfs).await?;
    if current_usage.used_bytes + total_size > quota_bytes {
        anyhow::bail!(
            "导入文件总大小 ({}) 将超出存储配额 (剩余 {})",
            format_size(total_size as usize),
            format_size((quota_bytes - current_usage.used_bytes) as usize)
        );
    }

    for source in source_paths {
        if source.is_dir() {
            // 递归复制目录
            let dir_name = source.file_name()
                .unwrap_or_default()
                .to_string_lossy();
            let dest_dir = target_abs.join(dir_name.as_ref());
            match copy_dir_recursive(source, &dest_dir).await {
                Ok((count, bytes)) => {
                    imported_count += count;
                    imported_bytes += bytes;
                }
                Err(e) => {
                    skipped.push(format!("{}: {}", source.display(), e));
                }
            }
        } else if source.is_file() {
            let file_name = source.file_name()
                .unwrap_or_default()
                .to_string_lossy();
            let dest = target_abs.join(file_name.as_ref());

            // 同名冲突处理
            let (final_dest, _) = resolve_name_conflict(
                &dest,
                &format!("{}/{}", target_dir, file_name),
            ).await;

            match tokio::fs::copy(source, &final_dest).await {
                Ok(bytes) => {
                    imported_count += 1;
                    imported_bytes += bytes;
                }
                Err(e) => {
                    skipped.push(format!("{}: {}", source.display(), e));
                }
            }
        }
    }

    Ok(ImportResult {
        imported_count,
        imported_bytes,
        skipped,
    })
}

#[derive(Debug, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct ImportResult {
    pub imported_count: u64,
    pub imported_bytes: u64,
    pub skipped: Vec<String>,
}

/// 递归复制目录
async fn copy_dir_recursive(src: &Path, dst: &Path) -> anyhow::Result<(u64, u64)> {
    tokio::fs::create_dir_all(dst).await?;
    let mut count = 0u64;
    let mut bytes = 0u64;
    let mut entries = tokio::fs::read_dir(src).await?;

    while let Some(entry) = entries.next_entry().await? {
        let entry_path = entry.path();
        let dest_path = dst.join(entry.file_name());

        if entry_path.is_dir() {
            let (c, b) = Box::pin(copy_dir_recursive(&entry_path, &dest_path)).await?;
            count += c;
            bytes += b;
        } else {
            let b = tokio::fs::copy(&entry_path, &dest_path).await?;
            count += 1;
            bytes += b;
        }
    }

    Ok((count, bytes))
}

/// 计算路径的总大小（文件或目录递归）
async fn calculate_path_size(path: &Path) -> anyhow::Result<u64> {
    if path.is_file() {
        Ok(tokio::fs::metadata(path).await?.len())
    } else if path.is_dir() {
        let mut total = 0u64;
        let mut entries = tokio::fs::read_dir(path).await?;
        while let Some(entry) = entries.next_entry().await? {
            total += Box::pin(calculate_path_size(&entry.path())).await?;
        }
        Ok(total)
    } else {
        Ok(0)
    }
}
```

```rust
    zip: &mut ZipFileWriter<tokio::fs::File>,
    dir: &Path,
    prefix: &str,
) -> anyhow::Result<(u64, u64)> {
    let mut count = 0u64;
    let mut bytes = 0u64;
    let mut entries = tokio::fs::read_dir(dir).await?;

    while let Some(entry) = entries.next_entry().await? {
        let path = entry.path();
        let name = entry.file_name().to_string_lossy().into_owned();
        let zip_path = format!("{}{}", prefix, name);

        if path.is_dir() {
            let (c, b) = Box::pin(add_directory_to_zip(
                zip,
                &path,
                &format!("{}/", zip_path),
            ))
            .await?;
            count += c;
            bytes += b;

            let entry_builder = ZipEntryBuilder::new(
                zip_path.into(),
                Compression::Deflate,
            );
            zip.write_entry_whole(entry_builder, &data).await?;
            count += 1;
            bytes += data.len() as u64;
        }
    }

    Ok((count, bytes))
}
```

## 8. 存储配额

### 8.1 配额策略

每个工作区拥有独立的存储配额，防止单个工作区占用过多磁盘空间：

| 配额项 | 默认值 | 可配置范围 | 说明 |
| ------ | ------ | ---------- | ---- |
| 工作区总存储限制 | 500 MB | 100 MB ~ 10 GB | 包含 files/ + ai-docs/ + knowledge/ |
| 单文件大小限制 | 50 MB | 10 MB ~ 200 MB | Agent file_write 和用户上传均受此限制 |
| 预警阈值 | 80% | 60% ~ 95% | 超过阈值时弹出警告 Toast |

### 8.2 存储用量计算

```rust
// crates/agent-core/src/files/quota.rs

/// 存储用量统计
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "camelCase")]
pub struct StorageUsage {
    /// 已使用字节数
    pub used_bytes: u64,
    /// 配额上限字节数
    pub quota_bytes: u64,
    /// 使用率（0.0 ~ 1.0）
    pub usage_ratio: f64,
    /// 各目录分项
    pub breakdown: StorageBreakdown,
    /// 是否超过预警阈值
    pub warning: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "camelCase")]
pub struct StorageBreakdown {
    pub files_bytes: u64,
    pub ai_docs_bytes: u64,
    pub knowledge_bytes: u64,
}

/// 默认配额：500 MB
pub const DEFAULT_QUOTA_BYTES: u64 = 500 * 1024 * 1024;
/// 预警阈值：80%
pub const WARNING_THRESHOLD: f64 = 0.80;

/// 计算工作区存储用量
pub async fn calculate_storage_usage(
    wfs: &WorkspaceFileSystem,
) -> anyhow::Result<StorageUsage> {
    let files_bytes = dir_size(&wfs.resolve_dir(WorkspaceDir::Files)).await;
    let ai_docs_bytes = dir_size(&wfs.resolve_dir(WorkspaceDir::AiDocs)).await;
    let knowledge_bytes = dir_size(&wfs.resolve_dir(WorkspaceDir::Knowledge)).await;

    let used_bytes = files_bytes + ai_docs_bytes + knowledge_bytes;
    let quota_bytes = DEFAULT_QUOTA_BYTES;
    let usage_ratio = used_bytes as f64 / quota_bytes as f64;

    Ok(StorageUsage {
        used_bytes,
        quota_bytes,
        usage_ratio,
        breakdown: StorageBreakdown {
            files_bytes,
            ai_docs_bytes,
            knowledge_bytes,
        },
        warning: usage_ratio >= WARNING_THRESHOLD,
    })
}

/// 递归计算目录大小（异步）
async fn dir_size(path: &Path) -> u64 {
    if !path.exists() {
        return 0;
    }

    let path_owned = path.to_path_buf();
    tokio::task::spawn_blocking(move || {
        let mut total = 0u64;
        if let Ok(walker) = walkdir::WalkDir::new(&path_owned)
            .into_iter()
            .filter_map(|e| e.ok())
            .try_for_each(|entry| {
                if entry.file_type().is_file() {
                    total += entry.metadata().map(|m| m.len()).unwrap_or(0);
                }
                Ok::<_, std::io::Error>(())
            }) {}
        total
    })
    .await
    .unwrap_or(0)
}

/// 清理建议：列出可安全删除的大文件
pub async fn suggest_cleanup(
    wfs: &WorkspaceFileSystem,
    min_file_size: u64,
) -> anyhow::Result<Vec<CleanupSuggestion>> {
    let root = wfs.root().to_path_buf();
    tokio::task::spawn_blocking(move || {
        let mut suggestions = Vec::new();

        for entry in walkdir::WalkDir::new(&root)
            .into_iter()
            .filter_map(|e| e.ok())
        {
            if !entry.file_type().is_file() {
                continue;
            }
            let size = entry.metadata().map(|m| m.len()).unwrap_or(0);
            if size >= min_file_size {
                let relative = entry.path()
                    .strip_prefix(&root)
                    .map(|p| p.to_string_lossy().into_owned())
                    .unwrap_or_default();

                let modified = entry.metadata()
                    .ok()
                    .and_then(|m| m.modified().ok())
                    .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
                    .map(|d| d.as_millis() as i64);

                suggestions.push(CleanupSuggestion {
                    path: relative,
                    size,
                    modified_at: modified,
                });
            }
        }

        // 按大小降序排列
        suggestions.sort_by(|a, b| b.size.cmp(&a.size));
        Ok(suggestions)
    })
    .await?
}

#[derive(Debug, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct CleanupSuggestion {
    pub path: String,
    pub size: u64,
    pub modified_at: Option<i64>,
}
```

```rust
/// 删除文件或目录
#[command]
pub async fn delete_file(
    app: AppHandle,
    state: State<'_, AppState>,
    relative_path: String,
) -> Result<(), AppError> {
    let wfs = state.active_workspace_fs().await?;
    let abs_path = wfs.resolve_path(&relative_path)
        .map_err(AppError::from)?;

    if abs_path.is_dir() {
        tokio::fs::remove_dir_all(&abs_path)
            .await
            .map_err(|e| AppError::Tool(e.to_string()))?;
    } else {
        tokio::fs::remove_file(&abs_path)
            .await
            .map_err(|e| AppError::Tool(e.to_string()))?;
    }

    // 同步删除 ai_artifacts 记录（如果是 ai-docs 下的文件）
    if relative_path.starts_with("ai-docs/") {
        let workspace_id = state.active_workspace_id().await?;
        sqlx::query!(
            "DELETE FROM ai_artifacts WHERE workspace_id = ? AND file_path = ?",
            workspace_id, relative_path
        )
        .execute(&state.db_pool)
        .await
        .ok(); // 非致命错误
    }

    app.emit("file_changed", serde_json::json!({
        "action": "deleted",
        "path": relative_path,
    })).ok();

    Ok(())
}

/// 重命名文件或目录
#[command]
pub async fn rename_file(
    app: AppHandle,
    state: State<'_, AppState>,
    old_path: String,
    new_name: String,
) -> Result<String, AppError> {
    let wfs = state.active_workspace_fs().await?;
    let old_abs = wfs.resolve_path(&old_path)
        .map_err(AppError::from)?;

    let new_abs = old_abs.parent()
        .ok_or_else(|| AppError::Validation("无法获取父目录".into()))?
        .join(&new_name);

    // 校验新路径也在沙箱内
    let _ = wfs.resolve_path(
        &wfs.to_relative(&new_abs).map_err(AppError::from)?
    ).map_err(AppError::from)?;

    tokio::fs::rename(&old_abs, &new_abs)
        .await
        .map_err(|e| AppError::Tool(e.to_string()))?;

    let new_relative = wfs.to_relative(&new_abs)
        .map_err(AppError::from)?;

    // 同步更新 ai_artifacts 记录
    if old_path.starts_with("ai-docs/") {
        let workspace_id = state.active_workspace_id().await?;
        sqlx::query!(
            "UPDATE ai_artifacts SET file_path = ?, file_name = ?, updated_at = ?
             WHERE workspace_id = ? AND file_path = ?",
            new_relative, new_name, now_ms(), workspace_id, old_path
        )
        .execute(&state.db_pool)
        .await
        .ok();
    }

    app.emit("file_changed", serde_json::json!({
        "action": "renamed",
        "oldPath": old_path,
        "newPath": new_relative,
    })).ok();

    Ok(new_relative)
}

/// 将文件添加到知识库
#[command]
pub async fn add_file_to_knowledge(
    state: State<'_, AppState>,
    file_path: String,
) -> Result<String, AppError> {
    let wfs = state.active_workspace_fs().await?;
    let workspace_id = state.active_workspace_id().await?;

    // 复制文件到 knowledge/ 目录
    let src = wfs.resolve_path(&file_path).map_err(AppError::from)?;
    let file_name = src.file_name()
        .ok_or_else(|| AppError::Validation("无效文件名".into()))?;
    let dst = wfs.resolve_dir(WorkspaceDir::Knowledge).join(file_name);

    tokio::fs::copy(&src, &dst)
        .await
        .map_err(|e| AppError::Tool(e.to_string()))?;

    // 触发 RAG 导入管道
    let knowledge_service = state.knowledge_service.clone();
    let uri = dst.to_string_lossy().into_owned();
    let ext = dst.extension()
        .and_then(|e| e.to_str())
        .unwrap_or("txt");
    let source_type = SourceType::from_extension(ext)
        .unwrap_or(SourceType::Txt);

    let source_id = knowledge_service
        .ingest(&workspace_id, &uri, source_type)
        .await
        .map_err(|e| AppError::Tool(e.to_string()))?;

    Ok(source_id)
}
```
