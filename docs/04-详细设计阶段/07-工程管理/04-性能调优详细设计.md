## 1. 性能目标

### 1.1 核心指标定义

| 操作 | 目标值 | 测量方式 | 降级阈值 |
|------|--------|---------|---------|
| 冷启动到首屏可交互 | < 2s | Tauri `setup()` 到前端 `DOMContentLoaded` + React hydration 完成 | > 3s 触发告警 |
| TTFT（首 Token 延迟） | < 500ms（不含 Provider 网络延迟） | 从 `provider.chat_stream()` 调用到首个 `token_chunk` 事件到达前端 | > 800ms |
| 单条消息渲染 | < 16ms（一帧） | React Profiler `actualDuration`，包含 Markdown 解析 + DOM 更新 | > 32ms |
| 全局搜索响应 | < 200ms | 从 `invoke('global_search')` 到结果返回，含 Tantivy + SQLite FTS | > 500ms |
| 内存峰值 | < 300MB | Tauri `sysinfo` crate 采集 RSS（Resident Set Size） | > 400MB |
| 工作区切换 | < 300ms | 从路由跳转到新工作区首屏渲染完成 | > 500ms |
| 数据库单条查询 | < 5ms | sqlx 查询 `duration_ms`（EXPLAIN 验证索引命中） | > 20ms |
| 数据库列表查询（分页） | < 20ms | 20 条/页典型分页查询 | > 50ms |

## 2. 冷启动优化

### 2.1 三阶段启动架构

系统设计阶段定义了三阶段启动策略（Phase 1/2/3），本节细化每个阶段的初始化任务编排和依赖关系。

#### 2.1.1 Phase 1：核心路径（< 500ms，阻塞首屏）

```rust
// apps/desktop/src-tauri/src/startup.rs

use std::time::Instant;
use tokio::join;
use tracing::instrument;

/// Phase 1：最小可用集合，阻塞 WebView 创建
#[instrument(name = "phase1_critical_path")]
pub async fn phase1_init(app_data_dir: &Path) -> anyhow::Result<Phase1State> {
    let start = Instant::now();

    // 并行执行三个无依赖任务
    let (config_result, db_result, provider_registry_result) = join!(
        // 任务 A：加载用户配置（~30ms）
        load_user_config(app_data_dir),
        // 任务 B：SQLite 连接池预热（~50ms）
        init_db_pool(app_data_dir),
        // 任务 C：Provider Registry 骨架构建（~20ms，仅注册不连接）
        init_provider_registry_skeleton(),
    );

    let config = config_result?;
    let db_pool = db_result?;
    let provider_registry = provider_registry_result?;

    tracing::info!(
        elapsed_ms = start.elapsed().as_millis() as u64,
        "Phase 1 完成"
    );

    Ok(Phase1State { config, db_pool, provider_registry })
}

/// 配置文件加载 —— 使用 toml 同步解析，文件通常 < 2KB
#[instrument(name = "load_config")]
async fn load_user_config(app_data_dir: &Path) -> anyhow::Result<AppConfig> {
    let config_path = app_data_dir.join("config.toml");
    let content = tokio::fs::read_to_string(&config_path).await?;
    let config: AppConfig = toml::from_str(&content)?;
    Ok(config)
}

/// 连接池预热 —— 并行创建 min_connections 个连接 + PRAGMA 初始化
#[instrument(name = "init_db_pool")]
async fn init_db_pool(app_data_dir: &Path) -> anyhow::Result<DbPools> {
    let db_path = app_data_dir.join("sessions/state.db");
    let db_url = format!("sqlite:{}", db_path.display());

    // 读写分离连接池
    let (read_pool, write_pool) = join!(
        SqlitePoolOptions::new()
            .max_connections(4)
            .min_connections(2)     // 预热 2 个读连接
            .acquire_timeout(Duration::from_secs(5))
            .connect_with(
                SqliteConnectOptions::from_str(&db_url)?
                    .read_only(true)
                    .pragma("journal_mode", "WAL")
                    .pragma("cache_size", "-65536")      // 64 MB
                    .pragma("mmap_size", "268435456")     // 256 MB
                    .pragma("synchronous", "NORMAL")
            ),
        SqlitePoolOptions::new()
            .max_connections(1)
            .min_connections(1)     // 预热 1 个写连接
            .connect_with(
                SqliteConnectOptions::from_str(&db_url)?
                    .pragma("journal_mode", "WAL")
                    .pragma("cache_size", "-65536")
                    .pragma("synchronous", "NORMAL")
                    .pragma("wal_autocheckpoint", "1000")
            ),
    );

    Ok(DbPools {
        read: read_pool?,
        write: write_pool?,
    })
}

/// Provider Registry 骨架 —— 仅解析配置中的 provider 列表，不建立网络连接
#[instrument(name = "init_provider_registry")]
async fn init_provider_registry_skeleton() -> anyhow::Result<ProviderRegistry> {
    // 从编译期嵌入的默认 provider 定义初始化（embed! 宏）
    let registry = ProviderRegistry::from_defaults();
    // 此处不进行 API key 验证或模型列表拉取，推迟到首次调用
    Ok(registry)
}
```

#### 2.1.2 Phase 2：功能就绪（< 2s，阻塞首次对话）

```rust
// apps/desktop/src-tauri/src/startup.rs

/// Phase 2：首次对话所需的最小依赖
#[instrument(name = "phase2_functional")]
pub async fn phase2_init(
    phase1: &Phase1State,
) -> anyhow::Result<Phase2State> {
    let start = Instant::now();

    // 并行执行
    let (tools_result, memory_result, session_result) = join!(
        // 任务 D：内置工具注册（~20ms）
        register_builtin_tools(&phase1.config),
        // 任务 E：活跃记忆快照加载（~30ms）
        load_active_memory_snapshot(&phase1.db_pool.read, &phase1.config.active_workspace_id),
        // 任务 F：上次会话恢复（~50ms）
        restore_last_session(&phase1.db_pool.read, &phase1.config.active_workspace_id),
    );

    tracing::info!(
        elapsed_ms = start.elapsed().as_millis() as u64,
        "Phase 2 完成"
    );

    Ok(Phase2State {
        tool_registry: tools_result?,
        memory_snapshot: memory_result?,
        last_session: session_result?,
    })
}

/// 记忆快照加载 —— 仅加载当前工作区的活跃记忆（valid_to IS NULL）
/// 使用覆盖索引避免回表查询
async fn load_active_memory_snapshot(
    pool: &SqlitePool,
    workspace_id: &str,
) -> anyhow::Result<Vec<MemoryEntry>> {
    let entries = sqlx::query_as::<_, MemoryEntry>(
        "SELECT id, category, key, value, half_life_days, retrieval_count, valid_from
         FROM memory_entries
         WHERE workspace_id = ? AND valid_to IS NULL
         ORDER BY retrieval_count DESC
         LIMIT 200"
    )
    .bind(workspace_id)
    .fetch_all(pool)
    .await?;

    Ok(entries)
}
```

#### 2.1.3 Phase 3：后台增强（异步，不阻塞用户交互）

```rust
// apps/desktop/src-tauri/src/startup.rs

/// Phase 3：后台异步任务，不阻塞首屏
pub fn phase3_background(
    phase1: &Phase1State,
    phase2: &Phase2State,
    app_handle: tauri::AppHandle,
) {
    let db_pool = phase1.db_pool.clone();
    let config = phase1.config.clone();

    // 所有后台任务通过 tokio::spawn 并行执行
    tokio::spawn(async move {
        // 任务 G：Tantivy 索引恢复 / 增量更新（~200ms）
        let tantivy_handle = tokio::spawn(restore_tantivy_index(
            &config.cache_dir.join("tantivy"),
            &db_pool.read,
        ));

        // 任务 H：MCP Server 工具 Schema 缓存预热（仅查 SQLite 缓存，不连接）
        let mcp_handle = tokio::spawn(preheat_mcp_schema_cache(&db_pool.read));

        // 任务 I：WASM AOT 缓存验证
        let wasm_handle = tokio::spawn(validate_wasm_aot_cache(
            &config.cache_dir.join("wasm"),
        ));

        // 任务 J：最近会话消息预加载（前 3 个会话的最新 50 条消息）
        let preload_handle = tokio::spawn(preload_recent_conversations(
            &db_pool.read, &config.active_workspace_id, 3, 50,
        ));

        // 等待所有后台任务完成并记录耗时
        let (tantivy, mcp, wasm, preload) = join!(
            tantivy_handle, mcp_handle, wasm_handle, preload_handle
        );

        if let Err(e) = tantivy.unwrap() {
            tracing::warn!("Tantivy 索引恢复失败: {}", e);
        }

        tracing::info!("Phase 3 后台初始化完成");
    });
}
```

```rust
        .setup(|app| {
            let app_handle = app.handle().clone();

            tauri::async_runtime::spawn(async move {
                let app_data_dir = app_handle.path().app_data_dir().unwrap();

                    main.show().unwrap();

            Ok(())
```

## 3. 前端渲染优化

### 3.1 虚拟滚动消息列表

对话消息列表采用 `react-virtuoso` 实现虚拟滚动，相比系统设计中的 `@tanstack/react-virtual`，`react-virtuoso` 提供更好的动态高度支持和自动滚动到底部行为，更适合聊天场景：

```typescript
// apps/desktop/src/components/MessageList.tsx

import React, { useRef, useCallback } from 'react';
import { Virtuoso, VirtuosoHandle } from 'react-virtuoso';
import { MessageItem } from './MessageItem';
import { Message } from '../types';

interface MessageListProps {
  messages: Message[];
  isStreaming: boolean;
}

export const MessageList = React.memo(function MessageList({
  messages,
  isStreaming,
}: MessageListProps) {
  const virtuosoRef = useRef<VirtuosoHandle>(null);

  // 流式输出时自动滚动到底部
  const followOutput = useCallback(
    (isAtBottom: boolean) => {
      if (isStreaming) return 'smooth';   // 流式中始终跟随
      if (isAtBottom) return 'smooth';    // 已在底部时跟随新消息
      return false;                       // 用户上滑浏览时不打断
    },
    [isStreaming]
  );

  return (
    <Virtuoso
      ref={virtuosoRef}
      data={messages}
      followOutput={followOutput}
      overscan={{ main: 200, reverse: 200 }}  // 像素级 overscan，避免快速滚动白屏
      increaseViewportBy={{ top: 400, bottom: 400 }}
      itemContent={(index, message) => (
        <MemoizedMessageItem
          key={message.id}
          message={message}
          isLast={index === messages.length - 1}
        />
      )}
      // 动态高度：Virtuoso 自动测量每条消息实际渲染高度
      // 无需 estimateSize，但提供 defaultItemHeight 加速首次渲染
      defaultItemHeight={150}
      alignToBottom
      initialTopMostItemIndex={messages.length - 1}
    />
  );
});

// 单条消息的 memo 包装，仅 message.id 或 message.content 变化时重渲染
const MemoizedMessageItem = React.memo(
  MessageItem,
  (prev, next) =>
    prev.message.id === next.message.id &&
    prev.message.content === next.message.content &&
    prev.message.status === next.message.status &&
    prev.isLast === next.isLast
);
```

### 3.2 RAF 批量 Token 渲染

系统设计中使用 `setInterval(50ms)` 做 delta batching。前端组件设计中采用 `requestAnimationFrame` 做批量刷新。本节统一为 RAF + 最大间隔双保险策略：

```typescript
// apps/desktop/src/hooks/useStreamingText.ts

import { useState, useRef, useCallback, useEffect } from 'react';

interface StreamingTextState {
  displayText: string;
  appendDelta: (delta: string) => void;
  reset: () => void;
  isStreaming: boolean;
}

export function useStreamingText(): StreamingTextState {
  const [displayText, setDisplayText] = useState('');
  const [isStreaming, setIsStreaming] = useState(false);
  const bufferRef = useRef('');
  const rafRef = useRef<number>(0);
  const lastFlushRef = useRef<number>(0);
  const maxIntervalMs = 50; // 最大刷新间隔，防止 RAF 被节流时延迟过大

  const flush = useCallback(() => {
    if (bufferRef.current.length === 0) return;

    const chunk = bufferRef.current;
    bufferRef.current = '';
    lastFlushRef.current = performance.now();

    setDisplayText(prev => prev + chunk);
  }, []);

  const scheduleFlush = useCallback(() => {
    cancelAnimationFrame(rafRef.current);
    rafRef.current = requestAnimationFrame(() => {
      flush();
    });

    // 双保险：若 RAF 被节流（如窗口最小化），50ms 后强制刷新
    const elapsed = performance.now() - lastFlushRef.current;
    if (elapsed > maxIntervalMs) {
      flush();
    }
  }, [flush]);

  const appendDelta = useCallback((delta: string) => {
    bufferRef.current += delta;
    setIsStreaming(true);
    scheduleFlush();
  }, [scheduleFlush]);

  const reset = useCallback(() => {
    cancelAnimationFrame(rafRef.current);
    bufferRef.current = '';
    lastFlushRef.current = 0;
    setDisplayText('');
    setIsStreaming(false);
  }, []);

  // 清理
  useEffect(() => {
    return () => cancelAnimationFrame(rafRef.current);
  }, []);

  return { displayText, appendDelta, reset, isStreaming };
}
```

消息中的代码块在首次进入可视区域时才执行语法高亮，避免在流式输出期间对不可见的代码块做高亮计算：

```typescript
import React, { useRef, useState, useEffect, useMemo } from 'react';

interface CodeBlockProps {
  code: string;
  language: string;
}

export const CodeBlock = React.memo(function CodeBlock({
  code,
  language,
}: CodeBlockProps) {
  const ref = useRef<HTMLPreElement>(null);
  const [highlighted, setHighlighted] = useState<string | null>(null);
  const [isVisible, setIsVisible] = useState(false);

  // IntersectionObserver 检测可见性
  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.disconnect(); // 一次性触发
        }
      },
      { rootMargin: '100px' } // 提前 100px 触发
    );

    observer.observe(el);
    return () => observer.disconnect();
  }, []);

  // 可见后异步加载 highlight.js 并高亮
  useEffect(() => {
    if (!isVisible || highlighted) return;

    let cancelled = false;

    // 动态导入 highlight.js（代码分割，不计入主 bundle）
    import('highlight.js/lib/core').then(async (hljs) => {
      // 按语言按需加载语法定义
      try {
        const langModule = await import(
          `highlight.js/lib/languages/${mapLanguage(language)}`
        );
        hljs.default.registerLanguage(language, langModule.default);
      } catch {
        // 未知语言，使用纯文本
      }

      if (!cancelled) {
        const result = hljs.default.highlight(code, {
          language: mapLanguage(language),
          ignoreIllegals: true,
        });
        setHighlighted(result.value);
      }
    });

    return () => { cancelled = true; };
  }, [isVisible, code, language, highlighted]);

  return (
    <pre ref={ref} className="code-block">
      <div className="code-block__header">
        <span className="code-block__language">{language}</span>
        <CopyButton text={code} />
      </div>
      {highlighted ? (
        <code
          className={`hljs language-${language}`}
          dangerouslySetInnerHTML={{ __html: highlighted }}
        />
      ) : (
        <code>{code}</code>
      )}
    </pre>
  );
});

/// 语言名映射（用户输入 → highlight.js 语言标识）
function mapLanguage(lang: string): string {
  const aliases: Record<string, string> = {
    ts: 'typescript',
    tsx: 'typescript',
    js: 'javascript',
    jsx: 'javascript',
    rs: 'rust',
    py: 'python',
    rb: 'ruby',
    yml: 'yaml',
    sh: 'bash',
    zsh: 'bash',
  };
  return aliases[lang] ?? lang;
}
```

```rust
// crates/db/src/query_analyzer.rs

use sqlx::SqlitePool;

/// 在 debug 构建中自动验证关键查询是否命中索引
pub struct QueryAnalyzer {
    pool: SqlitePool,
}

impl QueryAnalyzer {
    pub fn new(pool: &SqlitePool) -> Self {
        Self { pool: pool.clone() }
    }

    /// 分析查询计划，返回是否存在全表扫描
    pub async fn check_query_plan(&self, sql: &str) -> anyhow::Result<QueryPlanReport> {
        let explain_sql = format!("EXPLAIN QUERY PLAN {}", sql);
        let rows = sqlx::query_as::<_, ExplainRow>(&explain_sql)
            .fetch_all(&self.pool)
            .await?;

        let has_table_scan = rows.iter().any(|r| {
            r.detail.contains("SCAN TABLE") && !r.detail.contains("VIRTUAL TABLE")
        });

        let has_index_usage = rows.iter().any(|r| {
            r.detail.contains("USING INDEX") || r.detail.contains("USING COVERING INDEX")
        });

        Ok(QueryPlanReport {
            sql: sql.to_string(),
            steps: rows.iter().map(|r| r.detail.clone()).collect(),
            has_table_scan,
            has_index_usage,
        })
    }

    /// 批量检查所有关键查询，用于 CI 集成
    pub async fn check_all_critical_queries(&self) -> Vec<QueryPlanReport> {
        let critical_queries = vec![
            "SELECT * FROM conversations WHERE workspace_id = 'test' ORDER BY updated_at DESC LIMIT 20",
            "SELECT * FROM messages WHERE conversation_id = 'test' ORDER BY created_at ASC LIMIT 50",
            "SELECT * FROM memory_entries WHERE workspace_id = 'test' AND valid_to IS NULL LIMIT 200",
            "SELECT model, SUM(cost_usd) FROM usage_metrics WHERE created_at > 0 GROUP BY model",
        ];

        let mut reports = Vec::new();
        for sql in critical_queries {
            match self.check_query_plan(sql).await {
                Ok(report) => {
                    if report.has_table_scan {
                        tracing::warn!(
                            sql = sql,
                            "关键查询存在全表扫描，需要添加索引"
                        );
                    }
                    reports.push(report);
                }
                Err(e) => tracing::error!(sql = sql, error = %e, "查询计划分析失败"),
            }
        }
        reports
    }
}

#[derive(Debug, sqlx::FromRow)]
struct ExplainRow {
    detail: String,
}

#[derive(Debug)]
pub struct QueryPlanReport {
    pub sql: String,
    pub steps: Vec<String>,
    pub has_table_scan: bool,
    pub has_index_usage: bool,
}
```

### 4.3 常见慢查询与修复

| 慢查询模式 | 原因 | 修复方案 |
|-----------|------|---------|
| `SELECT * FROM messages WHERE content LIKE '%keyword%'` | LIKE 前缀通配符无法使用 B-tree 索引 | 使用 FTS5 全文索引 `messages_fts MATCH 'keyword'` |
| `SELECT COUNT(*) FROM messages WHERE conversation_id = ?` | 即使有索引，COUNT 仍需遍历索引叶节点 | 在 `conversations` 表维护 `message_count` 冗余字段，通过触发器同步 |
| `SELECT * FROM memory_entries ORDER BY weibull_score(...)` | 计算列无法建索引 | 预计算 `cached_weibull_score` 列，每次检索时更新，建索引 `(workspace_id, cached_weibull_score DESC)` |
| `SELECT * FROM usage_metrics WHERE date(created_at/1000,'unixepoch') = '2026-08-09'` | 函数包裹列导致索引失效 | 改用范围查询 `created_at BETWEEN start_ms AND end_ms` |
| `INSERT INTO messages (...) VALUES (...)` 在高频流式写入时变慢 | 每条 token 都触发 fsync | 流式期间将 tokens 缓存在内存，流结束后批量 INSERT（一个事务） |

消息计数触发器示例：

```rust
/// 定期 WAL checkpoint（在应用空闲时执行）
pub async fn periodic_wal_checkpoint(pool: &SqlitePool) {
    let mut interval = tokio::time::interval(Duration::from_secs(300)); // 每 5 分钟

    loop {
        interval.tick().await;

        // PASSIVE checkpoint：不阻塞读写
        match sqlx::query("PRAGMA wal_checkpoint(PASSIVE)")
            .execute(pool)
            .await
        {
            Ok(_) => tracing::debug!("WAL passive checkpoint 完成"),
            Err(e) => tracing::warn!("WAL checkpoint 失败: {}", e),
        }
    }
}

        .ok();
```

### 4.6 Prepared Statement 缓存

sqlx 默认启用 prepared statement 缓存。对于高频查询，显式控制缓存行为：

```rust
// crates/db/src/prepared.rs

use sqlx::sqlite::SqliteStatement;
use std::collections::HashMap;
use parking_lot::RwLock;

/// 高频查询的预编译语句注册表
/// sqlx 的 query/query_as 宏在编译期验证 SQL，运行时自动缓存
/// 此注册表用于监控哪些语句被缓存以及命中率
pub struct PreparedStatementMonitor {
    stats: RwLock<HashMap<String, StatementStats>>,
}

#[derive(Debug, Default, Clone)]
struct StatementStats {
    executions: u64,
    total_duration_us: u64,
    avg_duration_us: u64,
}

impl PreparedStatementMonitor {
    pub fn new() -> Self {
        Self {
            stats: RwLock::new(HashMap::new()),
        }
    }

    /// 记录一次查询执行
    pub fn record(&self, sql_hash: &str, duration_us: u64) {
        let mut stats = self.stats.write();
        let entry = stats.entry(sql_hash.to_string()).or_default();
        entry.executions += 1;
        entry.total_duration_us += duration_us;
        entry.avg_duration_us = entry.total_duration_us / entry.executions;
    }

    /// 获取最慢的 N 个查询（开发者面板使用）
    pub fn slowest_queries(&self, n: usize) -> Vec<(String, StatementStats)> {
        let stats = self.stats.read();
        let mut entries: Vec<_> = stats.iter()
            .map(|(k, v)| (k.clone(), v.clone()))
            .collect();
        entries.sort_by(|a, b| b.1.avg_duration_us.cmp(&a.1.avg_duration_us));
        entries.truncate(n);
        entries
    }
}
```

## 5. 流式响应优化

### 5.1 背压控制架构

Provider 流式输出与 UI 渲染之间存在速率差异：LLM 输出速度可达 50-100 tokens/s，而 UI 每帧只能处理有限更新。需要建立背压（backpressure）机制，避免 token 事件在 Tauri 事件总线上积压。

```text
Provider Stream                Tauri 事件桥                     前端
───────────────       ─────────────────────────       ───────────────
  token 1 ──────┐
  token 2 ──────┤     ┌─────────────────────┐
  token 3 ──────┼────►│ TokenBatcher         │
  token 4 ──────┤     │ 每 50ms 或 10 tokens │──► emit("token_chunk") ──► RAF 刷新
  token 5 ──────┤     │ 取较先到达者         │
  token 6 ──────┘     └─────────────────────┘
```

### 5.2 Token 事件批量合并（Rust 侧）

```rust
// crates/agent-runtime/src/streaming/token_batcher.rs

use std::time::{Duration, Instant};
use tokio::sync::mpsc;

/// Token 批量合并器
/// 策略：累积 token 直到满足以下任一条件后发射：
///   1. 累积 >= max_tokens 个 token
///   2. 距上次发射 >= max_interval
///   3. 流结束
pub struct TokenBatcher {
    max_tokens: usize,
    max_interval: Duration,
}

impl TokenBatcher {
    pub fn new(max_tokens: usize, max_interval: Duration) -> Self {
        Self { max_tokens, max_interval }
    }

    /// 默认配置：10 tokens 或 50ms
    pub fn default_config() -> Self {
        Self::new(10, Duration::from_millis(50))
    }

    /// 将原始 token 流转换为批量 token 流
    pub async fn batch_stream(
        &self,
        mut input: mpsc::Receiver<String>,
        output: mpsc::Sender<String>,
    ) {
        let mut buffer = String::with_capacity(256);
        let mut token_count = 0usize;
        let mut last_emit = Instant::now();

        loop {
            // 计算距离下次强制发射的剩余时间
            let remaining = self.max_interval
                .saturating_sub(last_emit.elapsed());

            tokio::select! {
                // 收到新 token
                token = input.recv() => {
                    match token {
                        Some(t) => {
                            buffer.push_str(&t);
                            token_count += 1;

                            // 达到 token 数阈值，立即发射
                            if token_count >= self.max_tokens {
                                let batch = std::mem::take(&mut buffer);
                                let _ = output.send(batch).await;
                                token_count = 0;
                                last_emit = Instant::now();
                            }
                        }
                        None => {
                            // 流结束，发射剩余缓冲
                            if !buffer.is_empty() {
                                let _ = output.send(buffer).await;
                            }
                            break;
                        }
                    }
                }
                // 超时，强制发射已缓冲内容
                _ = tokio::time::sleep(remaining), if !buffer.is_empty() => {
                    let batch = std::mem::take(&mut buffer);
                    let _ = output.send(batch).await;
                    token_count = 0;
                    last_emit = Instant::now();
                }
            }
        }
    }
}
```

### 5.3 Tauri 事件节流

在 Tauri 事件桥层面添加节流，防止前端事件队列过载：

```rust
// apps/desktop/src-tauri/src/event_bridge.rs

use tauri::Emitter;
use std::sync::atomic::{AtomicU64, Ordering};
use std::time::Instant;

/// Tauri 事件发射器，内置节流逻辑
pub struct ThrottledEmitter {
    app_handle: tauri::AppHandle,
    /// 上次发射时间戳（纳秒），用于全局节流
    last_emit_ns: AtomicU64,
    /// 最小发射间隔
    min_interval: Duration,
}

impl ThrottledEmitter {
    pub fn new(app_handle: tauri::AppHandle, min_interval: Duration) -> Self {
        Self {
            app_handle,
            last_emit_ns: AtomicU64::new(0),
            min_interval,
        }
    }

    /// 发射 token_chunk 事件（带节流）
    pub fn emit_token_chunk(&self, conv_id: &str, delta: &str) -> anyhow::Result<()> {
        let now_ns = Instant::now().elapsed().as_nanos() as u64;
        let last = self.last_emit_ns.load(Ordering::Relaxed);

        // 未达最小间隔，跳过此次发射（数据由 TokenBatcher 合并到下一批）
        if now_ns.saturating_sub(last) < self.min_interval.as_nanos() as u64 {
            return Ok(());
        }

        self.last_emit_ns.store(now_ns, Ordering::Relaxed);

        self.app_handle.emit("token_chunk", serde_json::json!({
            "convId": conv_id,
            "delta": delta,
        }))?;

        Ok(())
    }

    /// 非节流事件（工具调用结果等低频事件直接发射）
    pub fn emit_raw<S: serde::Serialize>(&self, event: &str, payload: S) -> anyhow::Result<()> {
        self.app_handle.emit(event, payload)?;
        Ok(())
    }
}
```

### 5.4 端到端 TTFT 优化

从用户按下回车到首个 token 渲染到屏幕的完整延迟拆解和优化：

```text
TTFT 延迟拆解（目标 < 500ms，不含 Provider 网络 RTT）
────────────────────────────────────────────────────────
[A] 消息入队 + UI 乐观更新              ~5ms
[B] Tauri Command invoke 开销           ~2ms
[C] 上下文构建（ContextBuilder）         ~30-80ms

    └── System Prompt 组装               ~10ms
[D] Provider 请求序列化                  ~5ms
[E] Provider 网络延迟                    ❌ 不计入目标
[F] 首 token 反序列化 + 事件发射         ~5ms
[G] TokenBatcher 缓冲延迟               ~0-50ms（首 token 立即透传）
[H] 前端 RAF 调度延迟                    ~0-16ms

- [C] 记忆检索结果缓存，相同 query 5 分钟内复用
- [G] 首个 token 绕过 batcher 直接发射（isFirst 标记）
```

## 6. 内存管理

### 6.1 会话内存管理策略

仅保留当前活跃会话 + 最近 2 个会话的消息数据在内存中，其余会话卸载到 SQLite：

```rust
// crates/agent-core/src/memory/conversation_cache.rs

use lru::LruCache;
use std::num::NonZeroUsize;
use tokio::sync::RwLock;

/// 会话消息内存缓存
/// 策略：最多缓存 3 个会话的消息（当前 + 2 个最近）
pub struct ConversationMessageCache {
    /// LRU 缓存：conversation_id → 消息列表
    cache: RwLock<LruCache<String, CachedConversation>>,
    /// 单个会话最大缓存消息数
    max_messages_per_conv: usize,
    db_pool: SqlitePool,
}

struct CachedConversation {
    messages: Vec<Message>,
    /// 内存占用估算（字节）
    estimated_bytes: usize,
    /// 是否有未持久化的变更
    dirty: bool,
}

impl ConversationMessageCache {
    pub fn new(db_pool: SqlitePool) -> Self {
        Self {
            cache: RwLock::new(LruCache::new(NonZeroUsize::new(3).unwrap())),
            max_messages_per_conv: 500,  // 超过 500 条时裁剪早期消息
            db_pool,
        }
    }

    /// 获取会话消息，缓存未命中时从 SQLite 加载
    pub async fn get_messages(
        &self,
        conversation_id: &str,
    ) -> anyhow::Result<Vec<Message>> {
        // 缓存命中
        {
            let mut cache = self.cache.write().await;
            if let Some(cached) = cache.get(conversation_id) {
                return Ok(cached.messages.clone());
            }
        }

        // 缓存未命中，从 SQLite 加载最近 N 条消息
        let messages = sqlx::query_as::<_, Message>(
            "SELECT * FROM messages
             WHERE conversation_id = ?
             ORDER BY created_at DESC
             LIMIT ?"
        )
        .bind(conversation_id)
        .bind(self.max_messages_per_conv as i64)
        .fetch_all(&self.db_pool)
        .await?
        .into_iter()
        .rev()  // 恢复时间正序
        .collect::<Vec<_>>();

        let estimated_bytes = messages.iter()
            .map(|m| m.content.len() + 200) // content + 元数据估算
            .sum();

        // 放入缓存（LRU 自动淘汰最久未访问的会话）
        {
            let mut cache = self.cache.write().await;
            cache.put(conversation_id.to_string(), CachedConversation {
                messages: messages.clone(),
                estimated_bytes,
                dirty: false,
            });
        }

        Ok(messages)
    }

    /// 追加流式消息（仅更新内存缓存，延迟持久化）
    pub async fn append_streaming_message(
        &self,
        conversation_id: &str,
        message: Message,
    ) {
        let mut cache = self.cache.write().await;
        if let Some(cached) = cache.get_mut(conversation_id) {
            cached.estimated_bytes += message.content.len() + 200;
            cached.messages.push(message);
            cached.dirty = true;

            // 超过最大消息数时裁剪
            if cached.messages.len() > self.max_messages_per_conv {
                let trim_count = cached.messages.len() - self.max_messages_per_conv;
                cached.messages.drain(0..trim_count);
            }
        }
    }

    /// 获取当前缓存的总内存占用估算
    pub async fn estimated_memory_usage(&self) -> usize {
        let cache = self.cache.read().await;
        cache.iter().map(|(_, v)| v.estimated_bytes).sum()
    }

    /// 强制卸载指定会话的内存缓存
    pub async fn evict(&self, conversation_id: &str) {
        let mut cache = self.cache.write().await;
        cache.pop(conversation_id);
    }
}
```

### 6.2 Embedding 缓存管理

向量 embedding 缓存使用 LRU + 容量上限双重淘汰策略：

```rust
// crates/vector-store/src/embedding_cache.rs

use lru::LruCache;
use parking_lot::Mutex;
use std::num::NonZeroUsize;

/// Embedding 向量缓存
/// 容量上限：100MB（可配置）
/// 淘汰策略：LRU + 容量驱逐
pub struct EmbeddingCache {
    /// content_hash → embedding 向量
    cache: Mutex<LruCache<String, CachedEmbedding>>,
    /// 当前缓存占用（字节）
    current_bytes: Mutex<usize>,
    /// 容量上限（字节）
    max_bytes: usize,
}

struct CachedEmbedding {
    vector: Vec<f32>,
    /// 缓存条目大小（字节）
    byte_size: usize,
}

impl EmbeddingCache {
    /// 创建缓存，默认 100MB 上限
    pub fn new(max_mb: usize) -> Self {
        Self {
            // 最大条目数：假设每个 embedding ~6KB（1536 维 * 4 bytes），
            // 100MB / 6KB ≈ 17000 条
            cache: Mutex::new(LruCache::new(
                NonZeroUsize::new(max_mb * 1024 * 1024 / 6144).unwrap()
            )),
            current_bytes: Mutex::new(0),
            max_bytes: max_mb * 1024 * 1024,
        }
    }

    /// 查找缓存
    pub fn get(&self, content_hash: &str) -> Option<Vec<f32>> {
        let mut cache = self.cache.lock();
        cache.get(content_hash).map(|c| c.vector.clone())
    }

    /// 插入缓存，必要时驱逐旧条目
    pub fn put(&self, content_hash: String, vector: Vec<f32>) {
        let byte_size = vector.len() * std::mem::size_of::<f32>();
        let mut cache = self.cache.lock();
        let mut current = self.current_bytes.lock();

        // 容量超限时持续驱逐 LRU 条目
        while *current + byte_size > self.max_bytes {
            if let Some((_, evicted)) = cache.pop_lru() {
                *current -= evicted.byte_size;
            } else {
                break;
            }
        }

        cache.put(content_hash, CachedEmbedding {
            vector,
            byte_size,
        });
        *current += byte_size;
    }

    /// 获取缓存统计信息（开发者面板使用）
    pub fn stats(&self) -> CacheStats {
        let cache = self.cache.lock();
        let current = self.current_bytes.lock();
        CacheStats {
            entry_count: cache.len(),
            used_bytes: *current,
            max_bytes: self.max_bytes,
            utilization_pct: (*current as f64 / self.max_bytes as f64 * 100.0) as u8,
        }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CacheStats {
    pub entry_count: usize,
    pub used_bytes: usize,
    pub max_bytes: usize,
    pub utilization_pct: u8,
}
```

### 6.3 后台任务内存限制

后台任务（Tantivy 索引、WASM 编译等）通过资源信号量和内存预算约束并发度：

```rust
// crates/agent-core/src/resource/memory_budget.rs

use sysinfo::System;
use tokio::sync::Semaphore;

/// 全局内存预算管理器
pub struct MemoryBudget {
    /// 应用总内存上限（字节）
    max_rss_bytes: usize,
    /// 后台任务并发信号量
    background_semaphore: Semaphore,
    /// 系统信息采集器
    system: parking_lot::Mutex<System>,
}

impl MemoryBudget {
    pub fn new(max_rss_mb: usize) -> Self {
        Self {
            max_rss_bytes: max_rss_mb * 1024 * 1024,
            // 限制后台任务并发数，防止内存爆炸
            background_semaphore: Semaphore::new(3),
            system: parking_lot::Mutex::new(System::new_all()),
        }
    }

    /// 检查当前 RSS 是否超过预算
    pub fn check_memory_pressure(&self) -> MemoryPressure {
        let mut sys = self.system.lock();
        sys.refresh_memory();

        let current_rss = sys.process(sysinfo::get_current_pid().unwrap())
            .map(|p| p.memory() as usize)
            .unwrap_or(0);

        let utilization = current_rss as f64 / self.max_rss_bytes as f64;

        if utilization > 0.9 {
            MemoryPressure::Critical {
                current_mb: current_rss / 1024 / 1024,
                max_mb: self.max_rss_bytes / 1024 / 1024,
            }
        } else if utilization > 0.7 {
            MemoryPressure::Warning {
                current_mb: current_rss / 1024 / 1024,
            }
        } else {
            MemoryPressure::Normal
        }
    }

    /// 后台任务获取执行许可
    /// 在内存压力大时阻塞新的后台任务
    pub async fn acquire_background_permit(&self) -> anyhow::Result<tokio::sync::SemaphorePermit<'_>> {
        match self.check_memory_pressure() {
            MemoryPressure::Critical { current_mb, max_mb } => {
                tracing::warn!(
                    current_mb, max_mb,
                    "内存压力严重，后台任务暂停"
                );
                // 等待内存释放
                tokio::time::sleep(Duration::from_secs(5)).await;
            }
            _ => {}
        }

        Ok(self.background_semaphore.acquire().await?)
    }
}

#[derive(Debug)]
pub enum MemoryPressure {
    Normal,
    Warning { current_mb: usize },
    Critical { current_mb: usize, max_mb: usize },
}
```

```rust
fn bench_vector_search(c: &mut Criterion) {
    let runtime = tokio::runtime::Runtime::new().unwrap();

    for vector_count in [1_000, 10_000, 50_000] {
        let store = runtime.block_on(setup_vector_store(vector_count));
        c.bench_with_input(
            BenchmarkId::new("vector_search_top10", vector_count),
            &vector_count,
            |b, _| {
                b.to_async(&runtime).iter(|| async {
                    black_box(store.search("如何优化 SQL 查询", 10).await.unwrap())
                })
            },
        );
    }
}

criterion_group!(
    benches,
    bench_startup_phases,
    bench_sqlite_queries,
    bench_token_batcher,
    bench_context_compress,
    bench_vector_search,
);
criterion_main!(benches);
```

```bash
# 对比上次结果（自动生成回归报告）

cargo bench --bench performance_bench -- --baseline current
```

### 7.3 CI 性能门禁

性能门禁流程确保每次合入主分支的代码不会引入性能回归：

```text
PR 性能检查流程
────────────────
  ① PR 提交
     │
  ② CI 并行执行
     ├── Rust criterion 基准测试
     │   └── 与 main 分支 baseline 对比
     │       ├── 回归 > 10% → ❌ 阻断合入
     │       ├── 回归 5-10% → ⚠️ 警告 + PR 评论
     │       └── 无回归或改善 → ✅ 通过
     │
     ├── Frontend Lighthouse CI
     │   ├── FCP > 1500ms → ❌ 阻断合入
     │   ├── LCP > 2000ms → ❌ 阻断合入
     │   └── TBT > 200ms → ❌ 阻断合入
     │
     └── Bundle Size Check
         ├── JS 主 bundle > 512KB → ⚠️ 警告
         └── 总资源 > 1MB → ⚠️ 警告
  ③ 全部通过后允许合入
```

Bundle Size 检查脚本：

```bash
#!/bin/bash
# scripts/check-bundle-size.sh

set -e

# 构建前端
pnpm --filter desktop build

# 检查主 bundle 大小
MAIN_BUNDLE=$(find apps/desktop/dist/assets -name "index-*.js" | head -1)
MAIN_SIZE=$(stat -f%z "$MAIN_BUNDLE" 2>/dev/null || stat --printf="%s" "$MAIN_BUNDLE")
MAIN_SIZE_KB=$((MAIN_SIZE / 1024))

echo "Main bundle size: ${MAIN_SIZE_KB}KB"

if [ "$MAIN_SIZE_KB" -gt 512 ]; then
  echo "::warning::Main bundle (${MAIN_SIZE_KB}KB) exceeds 512KB budget"
fi

# 检查总资源大小
TOTAL_SIZE=$(find apps/desktop/dist -type f | xargs stat -f%z 2>/dev/null | awk '{s+=$1}END{print s}' || \
             find apps/desktop/dist -type f -exec stat --printf="%s\n" {} + | awk '{s+=$1}END{print s}')
TOTAL_SIZE_KB=$((TOTAL_SIZE / 1024))

echo "Total assets size: ${TOTAL_SIZE_KB}KB"

if [ "$TOTAL_SIZE_KB" -gt 1024 ]; then
  echo "::warning::Total assets (${TOTAL_SIZE_KB}KB) exceeds 1MB budget"
fi
```

### 7.4 指标仪表盘集成

性能指标通过可观测性系统的 `MetricsPanel` 展示（详见 [05-可观测性详细设计.md](../06-安全与基础设施/05-可观测性详细设计.md) 第 7 节），本节补充性能专用视图：

```typescript
// apps/desktop/src/components/dev/PerformanceTab.tsx

import { useState, useEffect } from 'react';
import { invoke } from '@tauri-apps/api/core';
import { listen } from '@tauri-apps/api/event';

interface PerformanceMetrics {
  startup: {
    phase1Ms: number;
    phase2Ms: number;
    totalMs: number;
  };
  memory: {
    totalRssMb: number;
    embeddingCacheUtilPct: number;
    conversationCacheMb: number;
    pressure: string;
  };
  sqlite: {
    slowestQueries: Array<{ sql: string; avgDurationUs: number; count: number }>;
  };
  streaming: {
    avgTtftMs: number;
    p95TtftMs: number;
    avgTokensPerSec: number;
  };
}

export function PerformanceTab({ workspaceId }: { workspaceId: string }) {
  const [metrics, setMetrics] = useState<PerformanceMetrics | null>(null);

  useEffect(() => {
    // 初始加载
    invoke<PerformanceMetrics>('get_performance_metrics', { workspaceId })
      .then(setMetrics);

    // 订阅实时内存快照
    const unlisten = listen<MemorySnapshot>('memory_snapshot', ({ payload }) => {
      setMetrics(prev => prev ? {
        ...prev,
        memory: {
          totalRssMb: payload.total_rss_mb,
          embeddingCacheUtilPct: payload.embedding_cache.utilization_pct,
          conversationCacheMb: payload.conversation_cache_mb,
          pressure: payload.pressure,
        },
      } : null);
    });

    return () => { unlisten.then(f => f()); };
  }, [workspaceId]);

  if (!metrics) return <div>加载中...</div>;

  return (
    <div className="performance-tab space-y-6 p-6">
      {/* 启动耗时 */}
      <section>
        <h3 className="font-semibold mb-2">启动耗时</h3>
        <div className="grid grid-cols-3 gap-4">
          <MetricCard
            label="Phase 1"
            value={`${metrics.startup.phase1Ms}ms`}
            status={metrics.startup.phase1Ms < 500 ? 'good' : 'warn'}
          />
          <MetricCard
            label="Phase 2"
            value={`${metrics.startup.phase2Ms}ms`}
            status={metrics.startup.phase2Ms < 1500 ? 'good' : 'warn'}
          />
          <MetricCard
            label="总启动"
            value={`${metrics.startup.totalMs}ms`}
            status={metrics.startup.totalMs < 2000 ? 'good' : 'bad'}
          />
        </div>
      </section>

      {/* 内存占用 */}
      <section>
        <h3 className="font-semibold mb-2">内存占用</h3>
        <div className="grid grid-cols-3 gap-4">
          <MetricCard
            label="RSS"
            value={`${metrics.memory.totalRssMb}MB`}
            status={metrics.memory.totalRssMb < 300 ? 'good' : 'bad'}
          />
          <MetricCard
            label="Embedding 缓存"
            value={`${metrics.memory.embeddingCacheUtilPct}%`}
            status={metrics.memory.embeddingCacheUtilPct < 80 ? 'good' : 'warn'}
          />
          <MetricCard
            label="会话缓存"
            value={`${metrics.memory.conversationCacheMb}MB`}
            status="neutral"
          />
        </div>
      </section>

      {/* 流式性能 */}
      <section>
        <h3 className="font-semibold mb-2">流式响应</h3>
        <div className="grid grid-cols-3 gap-4">
          <MetricCard
            label="TTFT (avg)"
            value={`${metrics.streaming.avgTtftMs}ms`}
            status={metrics.streaming.avgTtftMs < 500 ? 'good' : 'warn'}
          />
          <MetricCard
            label="TTFT (P95)"
            value={`${metrics.streaming.p95TtftMs}ms`}
            status={metrics.streaming.p95TtftMs < 800 ? 'good' : 'bad'}
          />
          <MetricCard
            label="输出速度"
            value={`${metrics.streaming.avgTokensPerSec.toFixed(1)} tok/s`}
            status="neutral"
          />
        </div>
      </section>

      {/* 最慢查询 */}
      <section>
        <h3 className="font-semibold mb-2">最慢 SQL 查询（Top 5）</h3>
        <table className="w-full text-sm font-mono">
          <thead>
            <tr className="text-left text-muted-foreground">
              <th>查询</th>
              <th>平均耗时</th>
              <th>执行次数</th>
            </tr>
          </thead>
          <tbody>
            {metrics.sqlite.slowestQueries.map((q, i) => (
              <tr key={i} className="border-t">
                <td className="py-1 truncate max-w-md">{q.sql}</td>
                <td className={q.avgDurationUs > 5000 ? 'text-red-600' : ''}>
                  {(q.avgDurationUs / 1000).toFixed(1)}ms
                </td>
                <td>{q.count}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </section>
    </div>
  );
}

function MetricCard({
  label,
  value,
  status,
}: {
  label: string;
  value: string;
  status: 'good' | 'warn' | 'bad' | 'neutral';
}) {
  const colors = {
    good: 'text-green-600 bg-green-50',
    warn: 'text-yellow-600 bg-yellow-50',
    bad: 'text-red-600 bg-red-50',
    neutral: 'text-foreground bg-muted',
  };

  return (
    <div className={`rounded-lg p-3 ${colors[status]}`}>
      <div className="text-xs opacity-70">{label}</div>
      <div className="text-xl font-bold">{value}</div>
    </div>
  );
}
```
