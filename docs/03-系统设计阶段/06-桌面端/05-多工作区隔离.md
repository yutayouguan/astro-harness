# 多工作区 / 项目隔离

> 阶段：系统设计 | 状态：定稿 | 说明：配置/记忆/Skill/MCP 独立存储

不同任务场景需要不同的配置，工作区提供完整的配置隔离。

---

## 工作区包含的内容

```
工作区
├── 基本信息        → 名称、图标、描述
├── Provider 配置   → 使用哪些模型、API Key 来源
├── Skill 集合      → 激活哪些 Skill
├── 记忆            → 独立的短期/长期记忆
├── MCP Server      → 连接哪些 MCP server
├── 对话历史        → 仅属于本工作区
├── 进化历史        → Skill/Prompt 演化记录
└── 安全配置        → 权限规则、人工介入阈值
```

## 数据模型

```sql
-- migrations/006_workspaces.sql

CREATE TABLE workspaces (
    id          TEXT PRIMARY KEY,
    name        TEXT NOT NULL,
    icon        TEXT NOT NULL DEFAULT '🤖',   -- emoji 或图片路径
    description TEXT,
    is_default  INTEGER NOT NULL DEFAULT 0,
    created_at  INTEGER NOT NULL DEFAULT (unixepoch('now','subsec')*1000),
    updated_at  INTEGER NOT NULL DEFAULT (unixepoch('now','subsec')*1000)
);

-- 工作区配置（结构化列，取代原 config JSON 字段）
CREATE TABLE workspace_configs (
    workspace_id             TEXT PRIMARY KEY REFERENCES workspaces(id) ON DELETE CASCADE,
    default_text_model       TEXT,                          -- 覆盖全局默认文本模型
    default_embedding_model  TEXT,                          -- 覆盖全局默认 embedding 模型
    require_approval         TEXT,                          -- JSON: Vec<GuardRule>
    max_agent_depth          INTEGER,                       -- 子 Agent 最大深度
    evolution_enabled        INTEGER NOT NULL DEFAULT 1,
    auto_approve_evolutions  INTEGER NOT NULL DEFAULT 0,
    memory_namespace         TEXT NOT NULL DEFAULT ''       -- 向量存储的命名空间前缀
);

-- 工作区与对话的关联（conversations 表添加字段）
ALTER TABLE conversations ADD COLUMN workspace_id TEXT
    NOT NULL DEFAULT 'default'
    REFERENCES workspaces(id);

-- 工作区与 Skill 的关联（激活状态）
CREATE TABLE workspace_skills (
    workspace_id TEXT NOT NULL REFERENCES workspaces(id),
    skill_name   TEXT NOT NULL,
    enabled      INTEGER NOT NULL DEFAULT 1,
    PRIMARY KEY (workspace_id, skill_name)
);

-- 工作区级 MCP server 配置
CREATE TABLE workspace_mcp_servers (
    workspace_id TEXT NOT NULL REFERENCES workspaces(id),
    server_name  TEXT NOT NULL,
    config       TEXT NOT NULL,   -- JSON
    enabled      INTEGER NOT NULL DEFAULT 1,
    PRIMARY KEY (workspace_id, server_name)
);
```

## WorkspaceManager

```rust
// crates/agent-core/src/workspace/manager.rs

pub struct WorkspaceManager {
    store: Arc<Store>,
    active: Arc<RwLock<String>>,   // 存储当前活跃工作区 ID
    // 每个工作区有独立的 AgentContext 实例
    contexts: Arc<Mutex<HashMap<String, Arc<AgentContext>>>>,
}

impl WorkspaceManager {
    // 切换工作区（不影响其他区的正在运行任务）
    pub async fn switch(&self, workspace_id: &str) -> anyhow::Result<Arc<AgentContext>> {
        let workspace = self.store.get_workspace(workspace_id).await?;

        // 复用已有 context（若已加载）
        if let Some(ctx) = self.contexts.lock().await.get(workspace_id) {
            *self.active.write().await = workspace_id.to_string();
            return Ok(ctx.clone());
        }

        // 构建新的 AgentContext（继承全局配置，用工作区配置覆盖）
        let ctx = AgentContext::build_for_workspace(&workspace).await?;
        self.contexts.lock().await.insert(workspace_id.to_string(), ctx.clone());
        *self.active.write().await = workspace_id.to_string();
        Ok(ctx)
    }

    pub async fn create(&self, name: &str, icon: &str) -> anyhow::Result<Workspace> {
        let workspace = Workspace {
            id: Uuid::new_v4().to_string(),
            name: name.to_string(),
            icon: icon.to_string(),
            config: WorkspaceConfig::default(),
            is_default: false,
            ..Default::default()
        };
        self.store.save_workspace(&workspace).await?;
        Ok(workspace)
    }

    pub async fn list(&self) -> anyhow::Result<Vec<Workspace>> {
        self.store.list_workspaces().await
    }
}
```

## Tauri Commands

```rust
// apps/desktop/src-tauri/src/commands/workspace.rs

#[command]
pub async fn list_workspaces(
    state: State<'_, AppState>,
) -> Result<Vec<Workspace>, String> {
    state.workspace_manager.list().await.map_err(|e| e.to_string())
}

#[command]
pub async fn create_workspace(
    name: String,
    icon: String,
    state: State<'_, AppState>,
) -> Result<Workspace, String> {
    state.workspace_manager.create(&name, &icon)
        .await.map_err(|e| e.to_string())
}

#[command]
pub async fn switch_workspace(
    workspace_id: String,
    state: State<'_, AppState>,
) -> Result<(), String> {
    state.workspace_manager.switch(&workspace_id).await.map_err(|e| e.to_string())?;
    Ok(())
}

#[command]
pub async fn update_workspace_config(
    workspace_id: String,
    config: WorkspaceConfig,
    state: State<'_, AppState>,
) -> Result<(), String> {
    state.workspace_manager.update_config(&workspace_id, config)
        .await.map_err(|e| e.to_string())
}
```

## 前端工作区切换器

```typescript
// apps/desktop/src/components/shared/WorkspaceSwitcher.tsx

export function WorkspaceSwitcher() {
  const { workspaces, active, switchWorkspace } = useWorkspaceStore();

  return (
    <DropdownMenu>
      <DropdownMenuTrigger>
        <span>{active.icon} {active.name}</span>
        <ChevronDown />
      </DropdownMenuTrigger>
      <DropdownMenuContent>
        {workspaces.map(ws => (
          <DropdownMenuItem
            key={ws.id}
            onClick={() => switchWorkspace(ws.id)}
          >
            {ws.icon} {ws.name}
            {ws.id === active.id && <Check className="ml-auto" />}
          </DropdownMenuItem>
        ))}
        <DropdownMenuSeparator />
        <DropdownMenuItem onClick={() => openCreateDialog()}>
          + 新建工作区
        </DropdownMenuItem>
      </DropdownMenuContent>
    </DropdownMenu>
  );
}
```
