# 快捷键与命令面板

> 阶段：系统设计 | 状态：定稿 | 说明：Cmd+K、全局快捷键

## 概述

Astro Agent 提供两套键盘交互层：**全局快捷键**（通过 Tauri global shortcut plugin 注册系统级热键，应用最小化时也可响应）和**命令面板**（在应用内通过 Cmd+K / Ctrl+K 唤起，提供模糊搜索所有操作的能力）。

---

## 1. 全局快捷键

| 快捷键 | 功能 | 适用范围 |
| ---- | ---- | ---- |
| `Ctrl+Shift+Space` / `Cmd+Shift+Space` | 唤起 / 隐藏应用窗口 | 系统全局 |
| `Ctrl+N` / `Cmd+N` | 新建对话 | 应用内 |
| `Ctrl+Tab` / `Cmd+Tab` | 切换工作区（循环） | 应用内 |
| `Ctrl+K` / `Cmd+K` | 打开命令面板 | 应用内 |
| `Ctrl+,` / `Cmd+,` | 打开设置 | 应用内 |
| `Ctrl+W` / `Cmd+W` | 关闭当前对话 | 应用内 |
| `Escape` | 关闭命令面板 / 取消操作 | 应用内 |
| `Enter` | 发送消息 | 输入框聚焦时 |
| `Shift+Enter` | 换行（不发送） | 输入框聚焦时 |

### Tauri global shortcut 配置

```rust
// src-tauri/src/lib.rs

use tauri_plugin_global_shortcut::{Code, GlobalShortcutExt, Modifiers, Shortcut};

pub fn setup_global_shortcuts(app: &tauri::App) -> Result<(), Box<dyn std::error::Error>> {
    let toggle_shortcut = Shortcut::new(
        Some(Modifiers::SUPER | Modifiers::SHIFT),
        Code::Space,
    );

    app.global_shortcut().on_shortcut(toggle_shortcut, |app, _shortcut, _event| {
        toggle_main_window(app);
    })?;

    Ok(())
}
```

在 `tauri.conf.json` 中声明插件：

```json
{
  "plugins": {
    "global-shortcut": {
      "shortcuts": []
    }
  }
}
```

## 2. 命令面板

### 功能分组

| 分组 | 示例命令 |
| ---- | ---- |
| **对话操作** | 新建对话、清空当前对话、导出对话、归档对话 |
| **工作区切换** | 切换到「代码助手」、切换到「研究助手」 |
| **Skill 执行** | 运行 Web 搜索、运行代码审查 |
| **历史搜索** | 搜索「上周的项目方案」 |
| **设置跳转** | 打开模型设置、打开预算设置、打开快捷键设置 |

### 命令注册系统

```typescript
// src/lib/command-registry.ts

export interface Command {
  /** 唯一标识符 */
  id: string;
  /** 显示名称 */
  label: string;
  /** 分组（用于命令面板分类显示） */
  group: "conversation" | "workspace" | "skill" | "history" | "settings";
  /** 快捷键提示（仅展示用，不触发绑定） */
  shortcut?: string;
  /** 关键词（用于模糊搜索） */
  keywords?: string[];
  /** 执行函数 */
  execute: () => void | Promise<void>;
  /** 图标组件名（来自 lucide-react） */
  icon?: string;
  /** 是否在命令面板中可见 */
  visible?: boolean;
}

class CommandRegistry {
  private commands = new Map<string, Command>();
  private listeners: (() => void)[] = [];

  register(command: Command): () => void {
    this.commands.set(command.id, { visible: true, ...command });
    this.notify();
    // 返回注销函数
    return () => {
      this.commands.delete(command.id);
      this.notify();
    };
  }

  registerMany(commands: Command[]): () => void {
    const unregisters = commands.map((c) => this.register(c));
    return () => unregisters.forEach((fn) => fn());
  }

  getAll(): Command[] {
    return Array.from(this.commands.values()).filter((c) => c.visible !== false);
  }

  search(query: string): Command[] {
    if (!query) return this.getAll();
    const q = query.toLowerCase();
    return this.getAll()
      .filter((cmd) => {
        const labelMatch = cmd.label.toLowerCase().includes(q);
        const keywordMatch = cmd.keywords?.some((k) => k.toLowerCase().includes(q));
        return labelMatch || keywordMatch;
      })
      .sort((a, b) => {
        // 标签前缀匹配排名更高
        const aPrefix = a.label.toLowerCase().startsWith(q) ? 0 : 1;
        const bPrefix = b.label.toLowerCase().startsWith(q) ? 0 : 1;
        return aPrefix - bPrefix;
      });
  }

  subscribe(listener: () => void): () => void {
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter((l) => l !== listener);
    };
  }

  private notify() {
    this.listeners.forEach((l) => l());
  }
}

export const commandRegistry = new CommandRegistry();
```

## 3. CommandPalette 组件

```tsx
// src/components/CommandPalette.tsx

import { useEffect, useState, useCallback } from "react";
import {
  Command,
  CommandDialog,
  CommandEmpty,
  CommandGroup,
  CommandInput,
  CommandItem,
  CommandList,
  CommandSeparator,
} from "@/components/ui/command";
import { commandRegistry } from "@/lib/command-registry";
import type { Command as CommandDef } from "@/lib/command-registry";
import * as Icons from "lucide-react";

interface CommandPaletteProps {
  open: boolean;
  onClose: () => void;
}

const GROUP_LABELS: Record<string, string> = {
  conversation: "对话操作",
  workspace: "工作区",
  skill: "技能",
  history: "历史搜索",
  settings: "设置",
};

export function CommandPalette({ open, onClose }: CommandPaletteProps) {
  const [query, setQuery] = useState("");
  const [commands, setCommands] = useState<CommandDef[]>([]);

  useEffect(() => {
    const update = () => setCommands(commandRegistry.search(query));
    update();
    return commandRegistry.subscribe(update);
  }, [query]);

  const handleSelect = useCallback(
    async (command: CommandDef) => {
      onClose();
      setQuery("");
      await command.execute();
    },
    [onClose]
  );

  // 按 group 分组
  const grouped = commands.reduce<Record<string, CommandDef[]>>((acc, cmd) => {
    (acc[cmd.group] ??= []).push(cmd);
    return acc;
  }, {});

  return (
    <CommandDialog open={open} onOpenChange={onClose}>
      <CommandInput
        placeholder="输入命令或搜索..."
        value={query}
        onValueChange={setQuery}
      />
      <CommandList>
        <CommandEmpty>未找到相关命令</CommandEmpty>

        {Object.entries(grouped).map(([group, cmds], idx) => (
          <div key={group}>
            {idx > 0 && <CommandSeparator />}
            <CommandGroup heading={GROUP_LABELS[group] ?? group}>
              {cmds.map((cmd) => {
                const IconComp = cmd.icon
                  ? (Icons as Record<string, React.ComponentType<{ size?: number }>>)[cmd.icon]
                  : null;

                return (
                  <CommandItem
                    key={cmd.id}
                    value={cmd.id}
                    onSelect={() => handleSelect(cmd)}
                    className="flex items-center gap-2"
                  >
                    {IconComp && <IconComp size={16} className="text-muted-foreground" />}
                    <span className="flex-1">{cmd.label}</span>
                    {cmd.shortcut && (
                      <kbd className="text-xs text-muted-foreground bg-muted px-1.5 py-0.5 rounded font-mono">
                        {cmd.shortcut}
                      </kbd>
                    )}
                  </CommandItem>
                );
              })}
            </CommandGroup>
          </div>
        ))}
      </CommandList>
    </CommandDialog>
  );
}
```

## 4. 全局快捷键监听（前端）

```tsx
// src/hooks/use-keyboard-shortcuts.ts

import { useEffect } from "react";

interface ShortcutHandlers {
  onCommandPalette: () => void;
  onNewConversation: () => void;
  onSettings: () => void;
  onNextWorkspace: () => void;
}

export function useKeyboardShortcuts(handlers: ShortcutHandlers) {
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      const isMac = navigator.platform.includes("Mac");
      const mod = isMac ? e.metaKey : e.ctrlKey;

      // Cmd+K / Ctrl+K — 命令面板
      if (mod && e.key === "k") {
        e.preventDefault();
        handlers.onCommandPalette();
        return;
      }

      // Cmd+N / Ctrl+N — 新建对话
      if (mod && e.key === "n") {
        e.preventDefault();
        handlers.onNewConversation();
        return;
      }

      // Cmd+, / Ctrl+, — 设置
      if (mod && e.key === ",") {
        e.preventDefault();
        handlers.onSettings();
        return;
      }

      // Ctrl+Tab — 切换工作区
      if (e.ctrlKey && e.key === "Tab") {
        e.preventDefault();
        handlers.onNextWorkspace();
        return;
      }
    };

    window.addEventListener("keydown", handler);
    return () => window.removeEventListener("keydown", handler);
  }, [handlers]);
}
```

## 5. 自定义快捷键

用户可在设置中重新绑定快捷键，存储在配置文件中。

### 配置结构

```json
// ~/.astro/keybindings.json

{
  "command_palette": "Ctrl+K",
  "new_conversation": "Ctrl+N",
  "switch_workspace": "Ctrl+Tab",
  "open_settings": "Ctrl+Comma",
  "send_message": "Enter",
  "newline_in_input": "Shift+Enter",
  "export_conversation": "Ctrl+Shift+E"
}
```

### Rust 快捷键加载

```rust
// crates/agent-core/src/config/keybindings.rs

use serde::{Deserialize, Serialize};
use std::collections::HashMap;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct KeyBindings {
    #[serde(flatten)]
    pub bindings: HashMap<String, String>,
}

impl KeyBindings {
    pub fn load_or_default(config_dir: &std::path::Path) -> Self {
        let path = config_dir.join("keybindings.json");
        if path.exists() {
            let content = std::fs::read_to_string(&path).unwrap_or_default();
            serde_json::from_str(&content).unwrap_or_default()
        } else {
            Self::default()
        }
    }

    pub fn save(&self, config_dir: &std::path::Path) -> anyhow::Result<()> {
        let path = config_dir.join("keybindings.json");
        std::fs::write(path, serde_json::to_string_pretty(self)?)?;
        Ok(())
    }
}

impl Default for KeyBindings {
    fn default() -> Self {
        let mut bindings = HashMap::new();
        bindings.insert("command_palette".to_string(), "Ctrl+K".to_string());
        bindings.insert("new_conversation".to_string(), "Ctrl+N".to_string());
        bindings.insert("switch_workspace".to_string(), "Ctrl+Tab".to_string());
        bindings.insert("open_settings".to_string(), "Ctrl+Comma".to_string());
        bindings.insert("send_message".to_string(), "Enter".to_string());
        bindings.insert("newline_in_input".to_string(), "Shift+Enter".to_string());
        Self { bindings }
    }
}
```

## 6. Skill 安装后自动注册命令

每次安装新 Skill 后，自动向命令注册系统注入对应命令：

```typescript
// src/lib/skill-commands.ts

import { commandRegistry } from "./command-registry";
import { invoke } from "@tauri-apps/api/core";

interface SkillMeta {
  id: string;
  name: string;
  description: string;
}

export function registerSkillCommands(skills: SkillMeta[]): () => void {
  return commandRegistry.registerMany(
    skills.map((skill) => ({
      id: `skill.run.${skill.id}`,
      label: `运行：${skill.name}`,
      group: "skill" as const,
      keywords: [skill.name, skill.description, "skill", "技能", "运行"],
      icon: "Zap",
      execute: () => invoke("run_skill", { skillId: skill.id }),
    }))
  );
}
```

## 7. 键盘导航规则

| 场景 | 快捷键 | 行为 |
| ---- | ---- | ---- |
| 命令面板已打开 | `↑` / `↓` | 在命令列表中导航 |
| 命令面板已打开 | `Enter` | 执行当前高亮命令 |
| 命令面板已打开 | `Escape` | 关闭命令面板 |
| 消息列表 | `Tab` | 聚焦下一条消息 |
| 消息列表 | `Shift+Tab` | 聚焦上一条消息 |
| 输入框聚焦 | `Enter` | 发送消息 |
| 输入框聚焦 | `Shift+Enter` | 在消息中换行 |
| 输入框聚焦 | `↑` | 编辑上一条发送的消息 |
